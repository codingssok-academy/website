Step 1 — 캐릭터 클래스

캐릭터의 이름, 직업, HP, 공격력, 레벨을 가진 클래스를 만듭니다. 딕셔너리와 클래스를 함께 사용합니다.

# hero.py — 캐릭터 모듈 JOBS = { "전사": {"hp": 120, "attack": 15}, "마법사": {"hp": 80, "attack": 25}, "궁수": {"hp": 100, "attack": 20}, } class Character: def __init__(self, name, job): if job not in JOBS: raise ValueError(f"없는 직업: {job}") stats = JOBS[job] self.name = name self.job = job self.max_hp = stats["hp"] self.hp = self.max_hp self.attack = stats["attack"] self.lv = 1 self.exp = 0 def is_alive(self): return self.hp > 0 def status(self): return (f"[Lv.{self.lv}] {self.name} ({self.job}) " f"HP:{self.hp}/{self.max_hp} ATK:{self.attack}") def level_up(self): self.lv += 1 self.max_hp += 10 self.attack += 3 self.hp = self.max_hp return f"{self.name} 레벨업! Lv.{self.lv}"

Step 2 — 전투 시스템

랜덤 몬스터와 전투하는 시스템입니다. 데미지에 랜덤 요소를 넣어 매번 다른 결과를 만듭니다.

# battle.py — 전투 모듈 import random MONSTERS = [ {"name": "슬라임", "hp": 30, "attack": 5, "exp": 10}, {"name": "고블린", "hp": 50, "attack": 10, "exp": 20}, {"name": "오크", "hp": 80, "attack": 15, "exp": 35}, {"name": "드래곤", "hp": 150, "attack": 25, "exp": 100}, ] def spawn_monster(): m = random.choice(MONSTERS).copy() m["max_hp"] = m["hp"] return m def calc_damage(attacker_atk): # 80%~120% 랜덤 데미지 return int(attacker_atk * random.uniform(0.8, 1.2)) def battle(hero, monster): print(f"\n--- {monster['name']} 등장! ---") print(f"HP: {monster['hp']} ATK: {monster['attack']}") while hero.is_alive() and monster["hp"] > 0: # 플레이어 공격 dmg = calc_damage(hero.attack) monster["hp"] -= dmg print(f"{hero.name} 공격! {dmg} 데미지") if monster["hp"] <= 0: print(f"{monster['name']} 처치! +{monster['exp']}EXP") hero.exp += monster["exp"] if hero.exp >= hero.lv * 30: hero.exp = 0 print(hero.level_up()) return True # 몬스터 반격 m_dmg = calc_damage(monster["attack"]) hero.hp -= m_dmg print(f"{monster['name']} 반격! {m_dmg} 데미지 (HP:{hero.hp})") print("패배...") return False

Step 3 — 세이브/로드

JSON 파일로 캐릭터 데이터를 저장하고 불러옵니다. 예외 처리로 파일이 없거나 손상된 경우도 대비합니다.

# save_load.py — 세이브/로드 모듈 import json SAVE_FILE = "rpg_save.json" def save_game(hero): data = { "name": hero.name, "job": hero.job, "lv": hero.lv, "hp": hero.hp, "max_hp": hero.max_hp, "attack": hero.attack, "exp": hero.exp, } try: with open(SAVE_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) print("세이브 완료!") except IOError as e: print(f"저장 실패: {e}") def load_game(): try: with open(SAVE_FILE, encoding="utf-8") as f: data = json.load(f) # Character 객체 복원 from hero import Character h = Character(data["name"], data["job"]) h.lv = data["lv"] h.hp = data["hp"] h.max_hp = data["max_hp"] h.attack = data["attack"] h.exp = data["exp"] print(f"로드 완료! {h.status()}") return h except FileNotFoundError: print("세이브 파일이 없습니다") return None except (json.JSONDecodeError, KeyError) as e: print(f"세이브 파일 손상: {e}") return None

Step 4 — 미니 게임 모음

전투 외에 즐길 수 있는 미니 게임들입니다. 승리하면 보상을 받습니다.

# minigame.py — 미니 게임 모듈 import random def number_guess(): """숫자 맞추기 게임 (7번 이내)""" answer = random.randint(1, 100) print("1~100 사이의 숫자를 맞춰보세요! (7번 기회)") for i in range(7): try: guess = int(input(f"[{i+1}/7] 숫자: ")) except ValueError: print("숫자만 입력하세요!") continue if guess == answer: print(f"정답! {i+1}번 만에 맞췄습니다!") return True elif guess < answer: print("UP!") else: print("DOWN!") print(f"실패... 정답은 {answer}이었습니다") return False def rock_paper_scissors(): """가위바위보 3판 2선승""" options = ["가위", "바위", "보"] wins = {("가위","보"), ("바위","가위"), ("보","바위")} player_w, com_w = 0, 0 while player_w < 2 and com_w < 2: user = input("가위/바위/보: ") if user not in options: print("다시 입력하세요") continue com = random.choice(options) print(f"상대: {com}") if (user, com) in wins: player_w += 1 print("이겼다!") elif user == com: print("비겼다!") else: com_w += 1 print("졌다...") return player_w >= 2

Step 5 — 메인 게임 루프

모든 모듈을 합쳐서 메뉴 기반의 완성된 RPG 프로그램을 만듭니다.

# main.py — 메인 게임 from hero import Character, JOBS from battle import spawn_monster, battle from save_load import save_game, load_game from minigame import number_guess, rock_paper_scissors def create_character(): name = input("캐릭터 이름: ") print("직업 선택:", ", ".join(JOBS.keys())) while True: try: job = input("직업: ") return Character(name, job) except ValueError as e: print(e) def main(): print("=== RPG 캐릭터 관리 ===") hero = load_game() if hero is None: hero = create_character() while True: print(f"\n{hero.status()}") print("1.전투 2.미니게임 3.회복 4.저장 5.종료") try: choice = input("선택: ") except (EOFError, KeyboardInterrupt): break if choice == "1": m = spawn_monster() battle(hero, m) elif choice == "2": print("a.숫자맞추기 b.가위바위보") g = input("선택: ") won = number_guess() if g == "a" else rock_paper_scissors() if won: hero.hp = min(hero.hp + 20, hero.max_hp) print("보상: HP +20 회복!") elif choice == "3": hero.hp = hero.max_hp print("HP 완전 회복!") elif choice == "4": save_game(hero) elif choice == "5": save_game(hero) print("게임 종료!") break if __name__ == "__main__": main()
