Step 1 — Song 클래스 설계

Song 클래스는 노래 한 곡의 정보를 담습니다. 제목(title), 아티스트(artist), 재생시간(duration, 초 단위)을 속성으로 갖습니다.

# Step 1: Song 클래스 class Song: def __init__(self, title, artist, duration): self.title = title # 곡 제목 self.artist = artist # 아티스트 self.duration = duration # 재생시간(초) def __str__(self): mins = self.duration // 60 secs = self.duration % 60 return f"{self.title} - {self.artist} ({mins}:{secs:02d})" def to_dict(self): return { "title": self.title, "artist": self.artist, "duration": self.duration, } @classmethod def from_dict(cls, d): return cls(d["title"], d["artist"], d["duration"]) # 테스트 s = Song("Dynamite", "BTS", 199) print(s) # Dynamite - BTS (3:19)

Step 2 — Playlist 클래스: 추가/삭제/검색/셔플

Playlist 클래스는 Song 객체들의 리스트를 관리합니다. random.shuffle()로 곡 순서를 섞고, 제목 키워드로 곡을 검색할 수 있습니다.

# Step 2: Playlist 클래스 import random class Playlist: def __init__(self, name): self.name = name self.songs = [] # Song 객체 리스트 def add(self, song): self.songs.append(song) print(f"추가: {song.title}") def remove(self, title): for i, s in enumerate(self.songs): if s.title == title: self.songs.pop(i) print(f"삭제: {title}") return print(f"'{title}' 곡을 찾을 수 없습니다.") def search(self, keyword): results = [s for s in self.songs if keyword.lower() in s.title.lower()] return results def shuffle(self): random.shuffle(self.songs) print("셔플 완료!") def total_time(self): total = sum(s.duration for s in self.songs) mins = total // 60 secs = total % 60 return f"{mins}분 {secs}초" def show(self): print(f"--- {self.name} ({len(self.songs)}곡) ---") for i, s in enumerate(self.songs, 1): print(f" {i}. {s}") print(f" 총 재생시간: {self.total_time()}")

Step 3 — JSON으로 플레이리스트 저장/불러오기

플레이리스트를 json.dump()로 파일에 저장하고 json.load()로 다시 불러옵니다. Song 클래스의 to_dict()와 from_dict()를 활용합니다.

# Step 3: JSON 저장/불러오기 import json def save_playlist(playlist, filename): data = { "name": playlist.name, "songs": [s.to_dict() for s in playlist.songs], } with open(filename, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) print(f"저장 완료: {filename}") def load_playlist(filename): with open(filename, "r", encoding="utf-8") as f: data = json.load(f) pl = Playlist(data["name"]) for d in data["songs"]: pl.songs.append(Song.from_dict(d)) print(f"불러오기 완료: {len(pl.songs)}곡") return pl
