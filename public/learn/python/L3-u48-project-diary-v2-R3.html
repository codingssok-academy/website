Step 1 — 파일 구조 설계와 할 일 추가

각 할 일은 todos.txt에 한 줄로 저장합니다. 파이프(|) 구분자로 ID, 제목, 우선순위, 마감일, 상태를 구분합니다.

def add_todo(title, priority=2, deadline="없음", filename="todos.txt"): """할 일 추가 — 자동 ID 생성""" new_id = 1 try: with open(filename, "r", encoding="utf-8") as f: lines = f.readlines() if lines: last = lines[-1].split("|", 1) new_id = int(last[0]) + 1 except FileNotFoundError: pass except ValueError: new_id = 1 with open(filename, "a", encoding="utf-8") as f: f.write(f"{new_id}|{title}|{priority}|{deadline}|todo\n") print(f"[추가됨] ID {new_id}: {title} (우선순위 {priority})")

Step 2 — 완료 처리, 삭제, 목록 출력

할 일의 상태를 done으로 변경하거나 항목을 삭제합니다. 목록을 출력할 때 완료 여부를 구분하여 표시합니다.

def complete_todo(todo_id, filename="todos.txt"): """할 일 완료 처리""" try: with open(filename, "r", encoding="utf-8") as f: lines = f.readlines() new_lines, done = [], False for line in lines: parts = line.strip().split("|") if len(parts) == 5 and parts[0] == str(todo_id): parts[4] = "done" new_lines.append("|".join(parts) + "\n") done = True else: new_lines.append(line) if not done: print(f"ID {todo_id}를 찾을 수 없습니다."); return with open(filename, "w", encoding="utf-8") as f: f.writelines(new_lines) print(f"[완료] ID {todo_id}") except FileNotFoundError: print("파일이 없습니다.") def list_todos(filename="todos.txt"): """할 일 목록 출력 (미완료 우선)""" try: with open(filename, "r", encoding="utf-8") as f: lines = [l.strip().split("|") for l in f.readlines()] todos = [p for p in lines if len(p) == 5] pending = [t for t in todos if t[4] == "todo"] done = [t for t in todos if t[4] == "done"] print("\n[미완료]") for t in pending: print(f" [{t[0]}] {t[1]} (P{t[2]}, 마감: {t[3]})") print("\n[완료]") for t in done: print(f" [{t[0]}] {t[1]} (완료)") except FileNotFoundError: print("저장된 할 일이 없습니다.")

Step 3 — 우선순위 정렬과 마감일 D-day

우선순위 숫자가 낮을수록 중요한 일입니다. sorted로 정렬하고, datetime 모듈로 오늘 날짜와 마감일을 비교해 D-day를 계산합니다.

from datetime import datetime def calc_dday(deadline_str): """마감일까지 남은 일수 계산""" if deadline_str == "없음": return "마감 없음" try: deadline = datetime.strptime(deadline_str, "%Y-%m-%d") today = datetime.today().replace(hour=0, minute=0, second=0, microsecond=0) diff = (deadline - today).days if diff < 0: return f"D+{abs(diff)} (기한 초과)" elif diff == 0: return "D-Day (오늘 마감)" return f"D-{diff}" except ValueError: return "날짜 형식 오류" def list_by_priority(filename="todos.txt"): """우선순위 오름차순 정렬 + D-day 표시""" try: with open(filename, "r", encoding="utf-8") as f: raw = [l.strip().split("|") for l in f.readlines()] pending = [p for p in raw if len(p) == 5 and p[4] == "todo"] sorted_todos = sorted(pending, key=lambda x: int(x[2])) print("\n--- 우선순위별 할 일 ---") for t in sorted_todos: dday = calc_dday(t[3]) print(f" P{t[2]} [{t[0]}] {t[1]} ({dday})") except FileNotFoundError: print("할 일 파일이 없습니다.")

핵심 요약

  • 구분자 |로 여러 속성을 한 줄에 저장하면 파일 구조가 단순해진다
  • 상태 필드(todo/done)로 완료 여부를 표시하고 필터링할 수 있다
  • sorted(list, key=lambda x: int(x[2])) — 문자열 숫자를 int로 변환 후 정렬
  • datetime.strptime(s, "%Y-%m-%d") — 날짜 문자열을 datetime 객체로 변환
  • (deadline - today).days — 두 날짜의 차이를 일수로 반환하는 timedelta 속성
  • 상태 변경/삭제 패턴: 전체 읽기 → 해당 줄 가공 → "w" 모드로 재저장
