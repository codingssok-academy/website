(2)문자열 길이 반환 — ___
(3)문자열 복사 — ___

Q2
코드 완성
보통
이름을 입력받아 대문자로 변환 후 "HELLO, [이름]!" 형식으로 출력하세요.
정답 보기
char name[30], msg[50]; scanf("%s", name); for (int i = 0; i < strlen(name); i++) name[i] = toupper(name[i]); sprintf(msg, "HELLO, %s!", name); printf("%s\n", msg);
Q3
종합 문자열 분석
보통
입력 문자열에서 알파벳 수, 숫자 수, 길이를 출력하는 코드를 작성하세요.
정답 보기
#include <stdio.h> #include <string.h> #include <ctype.h> int main() { char s[100]; int alpha=0, digit=0; scanf("%s", s); for (int i=0; i<strlen(s); i++) { if (isalpha(s[i])) alpha++; else if (isdigit(s[i])) digit++; } printf("길이:%lu 알파벳:%d 숫자:%d\n",strlen(s),alpha,digit); }
Q4
strtok 활용
보통
"이름:나이:점수" 형식 문자열 "홍길동:15:95"를 파싱해 각각 출력하세요.
정답 보기
char data[] = "홍길동:15:95"; char *name = strtok(data, ":"); char *age = strtok(NULL, ":"); char *score = strtok(NULL, ":"); printf("이름:%s 나이:%s 점수:%s\n", name, age, score);
Q5
종합 미니 프로그램
보통
안전하게 이름을 입력받아 첫 글자만 대문자, 나머지는 소문자로 바꿔 출력하세요.
정답 보기
#include <stdio.h> #include <string.h> #include <ctype.h> int main() { char name[30]; fgets(name, sizeof(name), stdin); int len = strlen(name); if (name[len-1] == '\n') name[len-1] = '\0'; for (int i=0; i<strlen(name); i++) name[i] = (i==0) ? toupper(name[i]) : tolower(name[i]); printf("%s\n", name); }
핵심 정리

L7 핵심 요약

  • 문자열 = char 배열 + \0. "Hello"는 6바이트입니다.
  • strlen(\0 제외), sizeof(\0 포함)의 차이를 명심합니다.
  • strcmp로 비교, strcpy/strcat으로 복사/연결합니다.
  • ctype.h: isalpha, isdigit, toupper, tolower 등 문자 처리.
  • fgets로 안전하게 입력, gets는 절대 사용 금지입니다.

종합 예제 — 학생 정보 처리기

#include <stdio.h> #include <string.h> #include <ctype.h> #include <stdlib.h> void safe_input(char *b, int n) { fgets(b, n, stdin); int l = strlen(b); if (l>0 && b[l-1]=='\n') b[l-1]='\0'; } int main() { char name[30], score_str[10], badge[60]; printf("이름: "); safe_input(name, sizeof(name)); printf("점수: "); safe_input(score_str, sizeof(score_str)); // 이름 첫 글자 대문자 if (strlen(name) > 0) name[0] = toupper(name[0]); // 점수 변환 int score = atoi(score_str); char *grade = score>=90?"A":score>=80?"B":score>=70?"C":"F"; // 뱃지 문자열 조립 sprintf(badge, "[%s | %d점 | %s등급]", name, score, grade); printf("\n%s\n", badge); printf("이름 길이: %lu\n", strlen(name)); return 0; }
