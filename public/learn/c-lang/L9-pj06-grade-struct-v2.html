구조체 배열로 성적 관리하기

구조체 배열에 여러 과목 점수를 저장하고, 평균·순위·등급을 자동으로 계산하는 성적 관리 프로그램을 만듭니다.

프로젝트 목표

  • 구조체에 과목별 점수 배열을 멤버로 포함합니다.
  • 평균을 자동 계산하는 함수를 구현합니다.
  • 점수 내림차순으로 순위를 매깁니다.
  • 등급(A~F)을 자동으로 부여합니다.

구현 코드

#include <stdio.h> #include <string.h> #define N 4 #define SUBJECTS 3 typedef struct { char name[20]; int scores[SUBJECTS]; // 국어/영어/수학 float avg; int rank; char grade; } Student; void calcAvg(Student *sp) { int sum = 0; for (int i=0;i<SUBJECTS;i++) sum+=sp->scores[i]; sp->avg = (float)sum / SUBJECTS; } char getGrade(float avg) { if (avg>=90) return 'A'; if (avg>=80) return 'B'; if (avg>=70) return 'C'; if (avg>=60) return 'D'; return 'F'; } void calcRanks(Student arr[], int n) { for (int i=0;i<n;i++) { arr[i].rank = 1; for (int j=0;j<n;j++) if (arr[j].avg > arr[i].avg) arr[i].rank++; } } int main() { Student arr[N] = { {"김민준", {85,90,80}}, {"이서연", {95,88,97}}, {"박지호", {72,65,78}}, {"최유진", {60,70,65}} }; for (int i=0;i<N;i++) { calcAvg(&arr[i]); arr[i].grade = getGrade(arr[i].avg); } calcRanks(arr, N); printf("%-8s %4s %4s %4s %6s %4s %2s\n", "이름","국어","영어","수학","평균","등급","순위"); for (int i=0;i<N;i++) printf("%-8s %4d %4d %4d %6.1f %c %d\n", arr[i].name, arr[i].scores[0], arr[i].scores[1], arr[i].scores[2], arr[i].avg, arr[i].grade, arr[i].rank); return 0; }
