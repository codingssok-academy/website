통계 함수 라이브러리
— sum/avg/max/min/count

배열을 받아서 합계, 평균, 최대값, 최소값, 개수를 반환하는 통계 함수 5개로 나만의 라이브러리를 만드는 프로젝트입니다.

이 프로젝트에서 배우는 것

  • 배열과 크기(n)를 매개변수로 받는 함수를 만듭니다.
  • for 루프로 배열 전체를 순회하는 패턴을 익힙니다.
  • 합계, 평균, 최대, 최소 알고리즘을 구현합니다.
  • int/double 반환 함수를 상황에 맞게 선택합니다.
  • 함수 5개를 조합해 통계 보고서를 출력합니다.
PROJECT 07
통계 함수 라이브러리
배열 매개변수 + 5가지 통계 알고리즘

5가지 통계 함수

int sum(int arr[], int n)
배열 n개 원소의 합계
double avg(int arr[], int n)
배열 n개 원소의 평균
int maxVal(int arr[], int n)
배열에서 가장 큰 값
int minVal(int arr[], int n)
배열에서 가장 작은 값
int count(int arr[], int n, int target)
target보다 큰 원소 개수
// 합계 int sum(int arr[], int n) { int total = 0, i; for (i = 0; i < n; i++) total += arr[i]; return total; } // 평균 double avg(int arr[], int n) { return (double)sum(arr, n) / n; // sum 함수 재사용! } // 최대값 int maxVal(int arr[], int n) { int m = arr[0], i; for (i = 1; i < n; i++) if (arr[i] > m) m = arr[i]; return m; } // 최소값 int minVal(int arr[], int n) { int m = arr[0], i; for (i = 1; i < n; i++) if (arr[i] < m) m = arr[i]; return m; }
핵심 패턴: 최대/최소는 첫 번째 원소로 초기화하고, 나머지와 비교합니다. 배열을 함수에 넘길 때는 배열 이름(arr)과 크기(n)를 같이 넘깁니다.

통계 라이브러리 — 전체 코드

/* ====== 통계 함수 라이브러리 ====== */ #include <stdio.h> int sum (int arr[], int n) { int s=0,i; for(i=0;i<n;i++) s+=arr[i]; return s; } double avg (int arr[], int n) { return (double)sum(arr,n)/n; } int maxVal(int arr[], int n) { int m=arr[0],i; for(i=1;i<n;i++) if(arr[i]>m) m=arr[i]; return m; } int minVal(int arr[], int n) { int m=arr[0],i; for(i=1;i<n;i++) if(arr[i]<m) m=arr[i]; return m; } int countAbove(int arr[], int n, int target) { int c=0,i; for(i=0;i<n;i++) if(arr[i]>target) c++; return c; } void printStats(int arr[], int n) { int i; printf("데이터: "); for(i=0;i<n;i++) printf("%d ", arr[i]); printf("\n"); printf("합계 : %d\n", sum(arr,n)); printf("평균 : %.2f\n", avg(arr,n)); printf("최대 : %d\n", maxVal(arr,n)); printf("최소 : %d\n", minVal(arr,n)); printf("80점 초과: %d명\n", countAbove(arr,n,80)); } int main() { int scores[] = {80, 95, 72, 88, 65, 91, 78}; int n = sizeof(scores) / sizeof(scores[0]); printStats(scores, n); return 0; }

int arr[] = {10, 20, 30, 40, 50}; 일 때 sizeof(arr)/sizeof(arr[0])의 결과는?
정답 보기

최대값 알고리즘
쉬움
배열 {5, 2, 8, 1, 9}에서 maxVal 함수가 동작하는 과정을 단계별로 추적하세요. m 값이 어떻게 바뀌나요?
정답 보기
m=5 → 2비교(유지5) → 8비교(m=8) → 1비교(유지8) → 9비교(m=9) → 최종 9 반환

int countRange(int arr[], int n, int lo, int hi) { int c=0, i; for(i=0;i<n;i++) if(arr[i]>=lo && arr[i]<=hi) c++; return c; }
