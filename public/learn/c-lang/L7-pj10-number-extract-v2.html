isdigit — 숫자 필터링 패턴
"abc123def456"에서 숫자만 뽑아내면 "123456"이 된다.
isdigit()으로 각 글자가 숫자인지 확인하고, 숫자만 새 배열에 복사하는 방식이다.
전화번호 정제, 가격 추출, 데이터 파싱 등 실무에서 자주 사용되는 패턴이다.

숫자 추출 구현
기본 — 숫자 문자만 추출
#include <stdio.h> #include <string.h> #include <ctype.h> void extract_digits(char src[], char result[]) { int j = 0; for (int i = 0; src[i]; i++) { if (isdigit(src[i])) { result[j++] = src[i]; } } result[j] = '\0'; } int main() { char input[100], digits[100]; printf("문자열 입력: "); fgets(input, sizeof(input), stdin); input[strcspn(input, "\n")] = '\0'; extract_digits(input, digits); printf("추출된 숫자: %s\n", digits); printf("정수값: %d\n", atoi(digits)); // 문자열 → 정수 return 0; }
전화번호 정제

"010-1234-5678" 또는 "(010) 1234-5678" 형식에서 숫자만 추출하면 "01012345678"이 된다. extract_digits 함수를 그대로 사용한다.

연속된 숫자 그룹 추출 (digit_sum)
// "price: 150, qty: 3" → 숫자들의 합 계산 int sum_all_numbers(char str[]) { int total = 0, num = 0, in_num = 0; for (int i = 0; str[i]; i++) { if (isdigit(str[i])) { num = num * 10 + (str[i] - '0'); // 자릿수 합산 in_num = 1; } else if (in_num) { total += num; // 숫자 그룹 끝 → 누적 num = 0; in_num = 0; } } if (in_num) total += num; // 마지막 숫자 처리 return total; } // "price: 150, qty: 3" → 150 + 3 = 153
자릿수 합산 패턴: num = num * 10 + (c - '0')는 문자를 정수로 직접 변환하는 고전적 패턴이다. '1','5','0' 처리: 0*10+1=1 → 1*10+5=15 → 15*10+0=150.
응용 — 수식 파서
문자열에서 숫자들을 추출해 합산
#include <stdio.h> #include <ctype.h> #include <string.h> int main() { char line[200]; printf("문장 입력 (예: score1=90, score2=85): "); fgets(line, sizeof(line), stdin); int nums[20], cnt = 0; int num = 0, in_num = 0; for (int i = 0; line[i]; i++) { if (isdigit(line[i])) { num = num * 10 + (line[i] - '0'); in_num = 1; } else if (in_num) { nums[cnt++] = num; num = 0; in_num = 0; } } if (in_num) nums[cnt++] = num; int total = 0; printf("추출된 숫자: "); for (int i = 0; i < cnt; i++) { printf("%d ", nums[i]); total += nums[i]; } printf("\n합계: %d, 평균: %.1f\n", total, (float)total/cnt); return 0; }
실행 예시
한계점 주의: "score1=90"에서 1, 90 모두 추출된다. 순수 숫자만 필요하다면 숫자 앞뒤가 알파벳이 아닌지 추가로 확인해야 한다.
