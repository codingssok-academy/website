혈액형 궁합 테스트

switch와 중첩 if로 두 사람의 혈액형 궁합을 판정합니다. 조건문을 실전에서 재미있게 활용해봅시다.

이 프로젝트에서 배우는 것

  • switch 문으로 혈액형을 선택하는 구조를 만들 수 있습니다.
  • switch 안에 중첩 if로 두 가지 조건을 동시에 처리할 수 있습니다.
  • char 타입과 문자 비교('A', 'B')를 사용할 수 있습니다.
  • 점수 계산과 문자열 결합으로 결과를 다양하게 표현할 수 있습니다.
  • 완성된 인터랙티브 프로그램을 처음부터 끝까지 혼자 작성할 수 있습니다.

구현 — switch + 중첩 if 궁합 판정

두 사람의 혈액형을 숫자로 입력받아 궁합을 판정합니다. 숫자 입력 방식부터 완성해봅시다.

1
기본 구조 — switch로 첫 번째 혈액형 분기

b1(첫 번째 혈액형)을 switch로 처리하고, 각 case 안에서 b2(두 번째)를 if로 비교합니다.

#include <stdio.h> int main() { int b1, b2; printf("=== 혈액형 궁합 테스트 ===\n"); printf("1=A형 2=B형 3=O형 4=AB형\n\n"); printf("첫 번째 사람 혈액형 입력: "); scanf("%d", &b1); printf("두 번째 사람 혈액형 입력: "); scanf("%d", &b2); printf("\n--- 결과 ---\n"); switch (b1) { case 1: /* 첫 번째가 A형 */ if (b2 == 1) printf("A + A : 74점 — 꼼꼼함이 맞아요!\n"); else if (b2 == 2) printf("A + B : 48점 — 성격 차이 주의!\n"); else if (b2 == 3) printf("A + O : 88점 — 찰떡 궁합!\n"); else if (b2 == 4) printf("A + AB: 82점 — 서로 잘 이해해요\n"); else printf("잘못된 입력입니다.\n"); break; case 2: /* 첫 번째가 B형 */ if (b2 == 1) printf("B + A : 48점 — 충돌 조심!\n"); else if (b2 == 2) printf("B + B : 90점 — 에너지가 넘쳐요!\n"); else if (b2 == 3) printf("B + O : 65점 — 보통 궁합\n"); else if (b2 == 4) printf("B + AB: 78점 — 자유로운 조합!\n"); else printf("잘못된 입력입니다.\n"); break; case 3: /* 첫 번째가 O형 */ if (b2 == 1) printf("O + A : 88점 — 최고 조합!\n"); else if (b2 == 2) printf("O + B : 65점 — 노력이 필요해요\n"); else if (b2 == 3) printf("O + O : 80점 — 리더 둘의 만남!\n"); else if (b2 == 4) printf("O + AB: 68점 — 배려하면 좋아요\n"); else printf("잘못된 입력입니다.\n"); break; case 4: /* 첫 번째가 AB형 */ if (b2 == 1) printf("AB+ A : 82점 — 합리적 조합!\n"); else if (b2 == 2) printf("AB+ B : 78점 — 창의력 폭발!\n"); else if (b2 == 3) printf("AB+ O : 68점 — 조금 노력 필요\n"); else if (b2 == 4) printf("AB+AB: 95점 — 천재들의 조합!\n"); else printf("잘못된 입력입니다.\n"); break; default: printf("1~4 중에서 입력하세요.\n"); } return 0; }
=== 혈액형 궁합 테스트 ===
1=A형 2=B형 3=O형 4=AB형

두 번째 사람 혈액형 입력: 3

확장 — char 입력 + 점수 계산 + 성격 설명

숫자 대신 문자 'A', 'B', 'O'를 직접 입력받아 더 직관적으로 만들고, 궁합 점수를 변수로 관리하여 성격 설명까지 출력합니다.

char 타입이란?

  • char c = 'A'; — 문자 하나 저장
  • 작은따옴표(' ')로 감쌈
  • scanf(" %c", &c); 앞에 공백 주의
  • 대소문자 다름: 'A' != 'a'

문자 비교 방법

  • if (c == 'A') — A형인지 확인
  • switch(c) — 문자도 switch 가능
  • case 'A': 형태로 사용
  • AB형은 두 글자 → 별도 처리 필요
2
확장 버전 — char 입력 + 점수 변수 + 성격 출력

점수를 int score 변수에 저장하고, 점수 범위에 따라 등급 메시지를 다시 출력합니다. AB형은 소문자 'x'로 입력받습니다.

#include <stdio.h> int main() { char c1, c2; int score = 0; printf("=== 혈액형 궁합 테스트 (문자 버전) ===\n"); printf("A=A형 B=B형 O=O형 x=AB형\n\n"); printf("첫 번째 사람 혈액형: "); scanf(" %c", &c1); /* 공백 하나 넣어야 Enter 무시 */ printf("두 번째 사람 혈액형: "); scanf(" %c", &c2); switch (c1) { case 'A': if (c2 == 'A') { score = 74; printf("꼼꼼함을 공유하는 사이\n"); } else if (c2 == 'B') { score = 48; printf("자유로운 B에 A가 답답함\n"); } else if (c2 == 'O') { score = 88; printf("A의 꼼꼼함 + O의 리더십!\n"); } else if (c2 == 'x') { score = 82; printf("AB의 이해력이 A를 편하게 함\n"); } else { printf("A B O x 중 하나를 입력하세요\n"); } break; case 'B': if (c2 == 'A') { score = 48; printf("A의 규칙이 B에겐 부담\n"); } else if (c2 == 'B') { score = 90; printf("자유로운 영혼 둘의 만남!\n"); } else if (c2 == 'O') { score = 65; printf("O가 B를 이끌어주면 OK\n"); } else if (c2 == 'x') { score = 78; printf("개성파 둘, 자유롭게 지냄\n"); } else { printf("A B O x 중 하나를 입력하세요\n"); } break; case 'O': if (c2 == 'A') { score = 88; printf("O의 넓은 마음이 A를 안아줌\n"); } else if (c2 == 'B') { score = 65; printf("둘 다 개성 강해서 노력 필요\n"); } else if (c2 == 'O') { score = 80; printf("리더 둘이 만나 시너지!\n"); } else if (c2 == 'x') { score = 68; printf("O가 배려하면 잘 맞아요\n"); } else { printf("A B O x 중 하나를 입력하세요\n"); } break; case 'x': /* AB형 */ if (c2 == 'A') { score = 82; printf("AB의 합리성이 A와 잘 맞음\n"); } else if (c2 == 'B') { score = 78; printf("창의력 둘, 아이디어 폭발!\n"); } else if (c2 == 'O') { score = 68; printf("가끔 충돌, 배려가 열쇠\n"); } else if (c2 == 'x') { score = 95; printf("지성파 둘의 완벽한 조합!\n"); } else { printf("A B O x 중 하나를 입력하세요\n"); } break; default: printf("A B O x 중 하나를 입력하세요.\n"); } /* 점수 범위에 따른 등급 출력 */ if (score >= 85) printf("궁합 점수: %d점 [최고 조합]\n", score); else if (score >= 70) printf("궁합 점수: %d점 [좋은 궁합]\n", score); else if (score >= 50) printf("궁합 점수: %d점 [보통 궁합]\n", score); else if (score > 0) printf("궁합 점수: %d점 [노력이 필요해요]\n", score); return 0; }
첫 번째 사람 혈액형: x

전체 코드 정리 + 도전 문제

지금까지 배운 내용을 하나로 합쳐 완성된 프로그램을 만들어봅시다. 두 버전의 핵심 개념을 정리하고 응용 문제에 도전합니다.

3
완성 버전 — 두 방식 통합 (숫자 + char)

모드를 먼저 선택하게 해서 숫자 입력과 문자 입력 중 하나를 고를 수 있는 완성 프로그램입니다.

#include <stdio.h> int main() { int mode; int b1, b2, score = 0; char c1, c2; printf("입력 방식 선택 (1=숫자, 2=문자): "); scanf("%d", &mode); if (mode == 1) { /* ── 숫자 모드 ── */ printf("1=A 2=B 3=O 4=AB\n"); printf("첫 번째: "); scanf("%d", &b1); printf("두 번째: "); scanf("%d", &b2); printf("\n"); switch (b1) { case 1: score = (b2==1)?74:(b2==2)?48:(b2==3)?88:(b2==4)?82:0; break; case 2: score = (b2==1)?48:(b2==2)?90:(b2==3)?65:(b2==4)?78:0; break; case 3: score = (b2==1)?88:(b2==2)?65:(b2==3)?80:(b2==4)?68:0; break; case 4: score = (b2==1)?82:(b2==2)?78:(b2==3)?68:(b2==4)?95:0; break; default: printf("1~4 입력 오류\n"; } } else { /* ── 문자 모드 ── */ printf("A=A형 B=B형 O=O형 x=AB형\n"); printf("첫 번째: "); scanf(" %c", &c1); printf("두 번째: "); scanf(" %c", &c2); printf("\n"); switch (c1) { case 'A': score=(c2=='A')?74:(c2=='B')?48:(c2=='O')?88:(c2=='x')?82:0; break; case 'B': score=(c2=='A')?48:(c2=='B')?90:(c2=='O')?65:(c2=='x')?78:0; break; case 'O': score=(c2=='A')?88:(c2=='B')?65:(c2=='O')?80:(c2=='x')?68:0; break; case 'x': score=(c2=='A')?82:(c2=='B')?78:(c2=='O')?68:(c2=='x')?95:0; break; default: printf("A B O x 입력 오류\n"); } } /* 공통: 점수 + 등급 출력 */ if (score > 0) { printf("궁합 점수: %d점\n", score); if (score >= 85) printf("등급: 최고 조합 ()\n"); else if (score >= 70) printf("등급: 좋은 궁합 ()\n"); else if (score >= 50) printf("등급: 보통 궁합 ()\n"); else printf("등급: 노력 필요 ()\n"); } return 0; }

switch (b1) { case 1: printf("A형\n"); /* break 없음! */ case 2: printf("B형\n"); break; case 3: printf("O형\n"); break; }

도전 과제

  • 대소문자 모두 허용하기 — 'a' 입력해도 A형으로 처리 (toupper() 함수 활용)
  • 궁합 점수를 별() 개수로 시각화해서 출력하기
  • 5명 혈액형을 배열에 저장하고 최고 궁합 쌍을 자동으로 찾기

핵심 정리

  • switch(변수)는 값이 여러 개일 때 if-else보다 깔끔하게 분기한다
  • 각 case 끝에 break를 빠뜨리면 다음 case도 실행되는 fall-through가 발생한다
  • switch 안에 if를 넣는 중첩 구조로 2차원 조건을 처리할 수 있다
  • char 타입은 작은따옴표(' ')로 감싸고, scanf에서 " %c"처럼 앞에 공백을 넣는다
  • 점수를 int 변수에 저장하면 나중에 등급 판정, 합산 등 다양하게 재활용할 수 있다
