  • 반환값 1/0으로 참/거짓을 표현하는 패턴을 익힙니다.
  • 여러 검증 함수를 조합해 종합 판정을 합니다.
  • 실제 로그인 시스템의 비밀번호 검증 원리를 이해합니다.

ctype.h — 문자 판별 함수

ctype.h를 include하면 문자의 종류를 쉽게 판별할 수 있습니다.

isdigit(c) 숫자(0-9)이면 참
isupper(c) 대문자(A-Z)이면 참
islower(c) 소문자(a-z)이면 참
isalpha(c) 영문자(a-z, A-Z)이면 참
// 길이 체크: 8글자 이상이면 1 반환 int checkLength(char pw[]) { return strlen(pw) >= 8; } // 숫자 포함 체크 int hasDigit(char pw[]) { int i; for (i = 0; pw[i] != '\0'; i++) if (isdigit(pw[i])) return 1; return 0; } // 대문자 포함 체크 int hasUpper(char pw[]) { int i; for (i = 0; pw[i] != '\0'; i++) if (isupper(pw[i])) return 1; return 0; } // 종합 검증 int validatePassword(char pw[]) { return checkLength(pw) && hasDigit(pw) && hasUpper(pw); }
'\0' (널 문자): C 문자열의 끝을 나타내는 특수 문자입니다. pw[i] != '\0' 조건으로 문자열 끝까지 반복합니다.

비밀번호 검증 — 전체 코드

/* ====== 비밀번호 검증 함수 ====== */ #include <stdio.h> #include <string.h> #include <ctype.h> int checkLength(char pw[]) { return strlen(pw) >= 8; } int hasDigit(char pw[]) { int i; for(i=0; pw[i]; i++) if(isdigit(pw[i])) return 1; return 0; } int hasUpper(char pw[]) { int i; for(i=0; pw[i]; i++) if(isupper(pw[i])) return 1; return 0; } int validatePassword(char pw[]) { return checkLength(pw) && hasDigit(pw) && hasUpper(pw); } void showReport(char pw[]) { printf("비밀번호: %s\n", pw); printf("길이(8+): %s\n", checkLength(pw) ? "OK" : "NG"); printf("숫자포함: %s\n", hasDigit(pw) ? "OK" : "NG"); printf("대문자: %s\n", hasUpper(pw) ? "OK" : "NG"); printf("종합: %s\n\n", validatePassword(pw) ? "통과!" : "실패"); } int main() { showReport("abc"); // 모두 실패 showReport("abcdefgh"); // 길이OK, 나머지NG showReport("Abc12345"); // 모두 통과 return 0; }
