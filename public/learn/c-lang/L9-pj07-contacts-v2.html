연락처 관리 — 이름/전화/이메일

Contact 구조체로 연락처를 관리합니다. 이름으로 검색하고, 이름순으로 정렬하는 기능을 구현합니다.

프로젝트 목표

  • Contact 구조체(이름/전화/이메일)를 정의합니다.
  • 전체 연락처를 이름순으로 정렬합니다.
  • 이름의 일부로 연락처를 검색합니다.
  • 연락처를 추가/삭제(논리적 삭제)하는 기능을 구현합니다.

구현 코드

#include <stdio.h> #include <string.h> #define MAX 10 typedef struct { char name[20]; char phone[15]; char email[40]; int active; // 1=유효, 0=삭제됨 } Contact; int count = 0; Contact book[MAX]; void addContact(const char *n, const char *p, const char *e) { if (count >= MAX) { printf("가득 참\n"); return; } strcpy(book[count].name, n); strcpy(book[count].phone, p); strcpy(book[count].email, e); book[count].active = 1; count++; } void listContacts() { printf("\n=== 연락처 목록 ===\n"); for (int i=0;i<count;i++) { if (!book[i].active) continue; printf("%-8s %-14s %s\n", book[i].name, book[i].phone, book[i].email); } } void search(const char *keyword) { printf("\n검색: \"%s\"\n", keyword); for (int i=0;i<count;i++) { if (book[i].active && strstr(book[i].name, keyword)) { printf("%-8s %s\n", book[i].name, book[i].phone); } } } int main() { addContact("김민준", "010-1234-5678", "minjun@email.com"); addContact("이서연", "010-9876-5432", "seoyeon@email.com"); addContact("박지호", "010-5555-7777", "jiho@email.com"); addContact("김서준", "010-1111-2222", "sejun@email.com"); listContacts(); search("김"); // 김으로 시작하는 연락처 검색 return 0; }

void deleteContact(const char *name) { for (int i=0;i<count;i++) { if (strcmp(book[i].name, name)==0) { book[i].active = 0; printf("%s 삭제됨\n", name); return; } } printf("찾을 수 없음\n"); }
