종합 예제 — 학교 성적 관리

#include <stdio.h> #include <string.h> #define N 4 enum Grade { A, B, C, D, F }; typedef struct { char name[20]; int grade; // 학년 float avg; enum Grade letter; } Student; enum Grade calcGrade(float avg) { if (avg >= 90) return A; if (avg >= 80) return B; if (avg >= 70) return C; if (avg >= 60) return D; return F; } void printStudent(Student *sp) { const char *letters[] = {"A","B","C","D","F"}; printf("%-8s %d학년 %.1f점 %s등급\n", sp->name, sp->grade, sp->avg, letters[sp->letter]); } int main() { Student arr[N] = { {"김민준",2,88.5f}, {"이서연",1,95.0f}, {"박지호",3,72.0f}, {"최유진",2,61.5f} }; for (int i=0; i<N; i++) arr[i].letter = calcGrade(arr[i].avg); printf("== 성적표 ==\n"); for (int i=0; i<N; i++) printStudent(&arr[i]); return 0; }
