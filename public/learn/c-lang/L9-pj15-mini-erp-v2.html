미니 ERP — 직원/부서/급여 관리

Level 9의 마지막 프로젝트입니다. 직원(Employee) 구조체에 부서명을 포함하고, 부서별 통계와 급여 관리 기능을 구현합니다.

프로젝트 목표

  • Employee 구조체(이름/부서/연차/급여)를 정의합니다.
  • 부서별 평균 급여를 계산합니다.
  • 연차에 따른 급여 인상을 일괄 적용합니다.
  • 전체 인건비를 계산하고 가장 높은 급여 직원을 찾습니다.

구현 코드

#include <stdio.h> #include <string.h> #define N 7 typedef struct { char name[20]; char dept[10]; // "개발", "기획", "영업" int years; // 근무 연수 int salary; // 월 급여 (만원) } Employee; void printEmp(const Employee *e) { printf("%-6s %-4s %2d년 %d만원\n", e->name, e->dept, e->years, e->salary); } void listAll(const Employee arr[], int n) { printf("%-6s %-4s %s %s\n", "이름", "부서", "연차", "급여"); for (int i = 0; i < n; i++) printEmp(&arr[i]); } void deptStats(const Employee arr[], int n, const char *dept) { int total = 0, count = 0; for (int i = 0; i < n; i++) if (strcmp(arr[i].dept, dept) == 0) { total += arr[i].salary; count++; } if (count) printf([%s] 인원: %d명, 평균 급여: %d만원\n", dept, count, total/count); } void raiseByYears(Employee arr[], int n) { // 연차 5년 이상: 10% 인상, 미만: 5% 인상 for (int i = 0; i < n; i++) { float rate = (arr[i].years >= 5) ? 0.10f : 0.05f; int raise = (int)(arr[i].salary * rate); arr[i].salary += raise; printf("%s: +%d만원 (%.0f%% 인상)\n", arr[i].name, raise, rate * 100); } } void findTopEarner(const Employee arr[], int n) { int top = 0; for (int i = 1; i < n; i++) if (arr[i].salary > arr[top].salary) top = i; printf("최고 연봉자: %s (%d만원)\n", arr[top].name, arr[top].salary); } int main() { Employee staff[N] = { {"김민준", "개발", 5, 550}, {"이서연", "기획", 3, 420}, {"박지호", "영업", 7, 620}, {"최유진", "개발", 2, 380}, {"정하준", "개발", 8, 700}, {"한소연", "영업", 4, 460}, {"윤서준", "기획", 6, 510} }; printf("=== 직원 목록 ===\n"); listAll(staff, N); printf("\n=== 부서별 통계 ===\n"); deptStats(staff, N, "개발"); deptStats(staff, N, "기획"); deptStats(staff, N, "영업"); int totalCost = 0; for (int i = 0; i < N; i++) totalCost += staff[i].salary; printf("월 인건비 합계: %d만원\n", totalCost); findTopEarner(staff, N); printf("\n=== 급여 인상 ===\n"); raiseByYears(staff, N); return 0; }
