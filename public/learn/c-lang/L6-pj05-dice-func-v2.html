  • checkWin(player, computer)으로 승패를 판단합니다.

  • 반복문으로 여러 판을 진행하는 프로그램을 만듭니다.

rand() — 난수 생성

rand()는 0부터 큰 수까지 무작위 숫자를 만듭니다. % 6 + 1을 써서 1~6 범위로 만듭니다. srand(time(0))으로 매번 다른 결과를 얻습니다.

#include <stdio.h> #include <stdlib.h> // rand(), srand() #include <time.h> // time() // 주사위 굴리기 함수 int rollDice() { return rand() % 6 + 1; // 1~6 반환 } // 승패 판정 함수: 1=승, -1=패, 0=무승부 int checkWin(int player, int computer) { if (player > computer) return 1; if (player < computer) return -1; return 0; }
srand(time(0)) 이란? 매번 프로그램을 실행할 때 다른 난수를 얻으려면 난수의 씨앗(seed)을 현재 시간으로 설정합니다. main 시작 부분에 한 번만 호출합니다.

주사위 게임 — 전체 코드

/* ====== 주사위 게임 함수 ====== */ #include <stdio.h> #include <stdlib.h> #include <time.h> int rollDice() { return rand() % 6 + 1; } int checkWin(int player, int computer) { if (player > computer) return 1; if (player < computer) return -1; return 0; } void printResult(int result) { if (result == 1) printf("[승리!] 내가 이겼다!\n"); else if (result == -1) printf("[패배] 컴퓨터가 이겼다...\n"); else printf("[무승부] 비겼다!\n"); } int main() { srand(time(0)); int wins = 0, losses = 0, draws = 0; int i, player, computer, result; printf("=== 주사위 게임 (5판) ===\n"); for (i = 1; i <= 5; i++) { player = rollDice(); computer = rollDice(); result = checkWin(player, computer); printf("%d판: 나=%d 컴=%d ", i, player, computer); printResult(result); if (result == 1) wins++; else if (result == -1) losses++; else draws++; } printf("결과: %d승 %d패 %d무\n", wins, losses, draws); return 0; }
