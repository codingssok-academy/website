간단 해시 테이블
체이닝 충돌 해결

key-value 쌍을 O(1)에 저장/검색하는 해시 테이블을 구현합니다. 충돌(collision)은 연결 리스트로 해결하는 체이닝(chaining) 방식을 사용합니다.

이 프로젝트에서 만드는 것

  • 해시 함수로 키를 버킷 인덱스로 변환합니다.
  • 각 버킷은 연결 리스트로 충돌을 처리합니다.
  • key-value 쌍을 삽입하는 ht_set() 함수를 만듭니다.
  • 키로 값을 검색하는 ht_get() 함수를 만듭니다.
  • 해시 테이블 전체를 해제합니다.

해시 테이블 구조

해시 함수는 키 문자열의 모든 문자 ASCII 값을 더해 버킷 수로 나눈 나머지를 인덱스로 사용합니다. 같은 인덱스에 여러 항목이 충돌하면 연결 리스트로 연결합니다.

#include <stdio.h> #include <stdlib.h> #include <string.h> #define BUCKET_SIZE 8 typedef struct Entry { char key[32]; char val[64]; struct Entry *next; } Entry; Entry *table[BUCKET_SIZE]; // 버킷 배열 int hash(const char *key) { unsigned int h = 0; while(*key) h += (unsigned char)(*key++); return h % BUCKET_SIZE; } void ht_set(const char*key, const char*val) { int idx = hash(key); /* 기존 키 업데이트 */ for(Entry*e=table[idx]; e; e=e->next) if(strcmp(e->key,key)==0) { strcpy(e->val,val); return; } /* 새 항목 앞에 추가 */ Entry *e=(Entry*)malloc(sizeof(Entry)); strcpy(e->key,key); strcpy(e->val,val); e->next=table[idx]; table[idx]=e; } const char* ht_get(const char*key) { int idx = hash(key); for(Entry*e=table[idx]; e; e=e->next) if(strcmp(e->key,key)==0) return e->val; return NULL; } void ht_free() { for(int i=0;i<BUCKET_SIZE;i++) { Entry*e=table[i],*nx; while(e){nx=e->next;free(e);e=nx;} table[i]=NULL; } } int main() { ht_set("name","홍길동"); ht_set("age","17"); ht_set("city","Seoul"); ht_set("age","18"); // 업데이트 printf("name=%s\n", ht_get("name")); printf("age =%s\n", ht_get("age")); const char*r=ht_get("email"); printf("email=%s\n", r?r:"없음"); ht_free(); return 0; }

실행 결과

name=홍길동 age =18 email=없음
