미니 RPG 전투 — 턴제 배틀 시스템

조건문과 변수 관리를 모두 녹여낸 종합 프로젝트. 용사와 몬스터가 번갈아 싸우는 턴제 전투 게임을 만들어보자.

어려움 — 조건문 종합 + 변수 관리

이 프로젝트에서 배우는 것

  • 여러 변수를 동시에 선언하고 관리할 수 있습니다.
  • switch문으로 플레이어 행동을 분기할 수 있습니다.
  • while 반복 + if 조건으로 게임 루프를 구성할 수 있습니다.
  • HP 관리(공격·방어·회복)를 코드로 표현할 수 있습니다.
  • rand()로 랜덤 값을 생성해 크리티컬 히트를 구현할 수 있습니다.

기본 구현 — 변수 설계부터 시작

RPG 전투 게임을 만들기 전에, 어떤 데이터가 필요한지 먼저 설계한다. 프로그래밍은 코딩보다 설계가 먼저다.

캐릭터 변수명 초기값 역할
용사 hero_hp 100 용사 체력. 0 이하면 게임 오버
용사 hero_atk 20 고정 공격력 (기본값)
용사 defense_mode 0 방어 모드 여부 (0=일반, 1=방어)
몬스터 monster_hp 80 몬스터 체력. 0 이하면 승리
몬스터 monster_atk 15 고정 공격력 (기본값)

게임 흐름 — while 루프 + switch 분기

mini_rpg_basic.c
/* 미니 RPG 전투 — 기본 버전 */ #include <stdio.h> int main() { /* ── 캐릭터 스탯 선언 ── */ int hero_hp = 100; /* 용사 체력 */ int hero_atk = 20; /* 용사 공격력 */ int monster_hp = 80; /* 몬스터 체력 */ int monster_atk = 15; /* 몬스터 공격력 */ int action; /* 플레이어 선택 입력값 */ int turn = 1; /* 현재 턴 번호 */ printf("=== 미니 RPG 전투 시작! ===\n\n"); /* ── 게임 메인 루프 ── */ while (hero_hp > 0 && monster_hp > 0) { printf("--- 턴 %d ---\n", turn); printf("[용사] HP: %d | [몬스터] HP: %d\n", hero_hp, monster_hp); printf("행동을 선택하세요: 1=공격 2=방어 3=회복\n"); printf("입력 > "); scanf("%d", &action); /* ── switch: 행동 분기 ── */ switch (action) { case 1: /* 공격 */ monster_hp -= hero_atk; printf("용사의 공격! 몬스터에게 %d 데미지!\n", hero_atk); break; case 2: /* 방어 */ printf("용사가 방어 태세를 취했다!\n"); break; case 3: /* 회복 */ hero_hp += 20; if (hero_hp > 100) hero_hp = 100; /* 최대 HP 제한 */ printf("용사가 포션을 사용! HP +20 (현재: %d)\n", hero_hp); break; default: printf("잘못된 입력입니다. 턴을 소비합니다.\n"); } /* ── 몬스터 반격 (몬스터가 살아있을 때만) ── */ if (monster_hp > 0) { if (action == 2) { /* 방어 중이면 데미지 절반 */ hero_hp -= monster_atk / 2; printf("몬스터 공격! 방어로 %d 데미지만 받음.\n", monster_atk / 2); } else { hero_hp -= monster_atk; printf("몬스터의 공격! 용사에게 %d 데미지!\n", monster_atk); } } turn++; printf("\n"); } /* ── 결과 판정 ── */ if (monster_hp <= 0) printf("*** 승리! 몬스터를 물리쳤다! ***\n"); else printf("*** 게임 오버. 용사가 쓰러졌다... ***\n"); return 0; }

핵심 포인트 3가지
1. while (hero_hp > 0 && monster_hp > 0) — 두 조건 모두 참일 때만 루프 유지
2. 방어 중이면 monster_atk / 2 — 정수 나눗셈으로 절반 데미지
3. 회복 후 if (hero_hp > 100) hero_hp = 100; — HP 상한 제한 필수

mini_rpg_advanced.c — 직업 선택 부분
#include <stdio.h> #include <stdlib.h> /* rand(), srand() */ #include <time.h> /* time() */ int main() { srand(time(NULL)); /* 랜덤 시드 초기화 */ int hero_hp, hero_atk, crit_bonus; int job; printf("직업을 선택하세요: 1=전사 2=마법사 3=궁수\n입력 > "); scanf("%d", &job); switch (job) { case 1: /* 전사 */ hero_hp = 120; hero_atk = 25; crit_bonus = 10; printf("전사를 선택했습니다! HP:%d ATK:%d\n", hero_hp, hero_atk); break; case 2: /* 마법사 */ hero_hp = 70; hero_atk = 35; crit_bonus = 10; printf("마법사를 선택했습니다! HP:%d ATK:%d\n", hero_hp, hero_atk); break; case 3: /* 궁수: 크리티컬 확률 2배 */ hero_hp = 90; hero_atk = 20; crit_bonus = 20; printf("궁수를 선택했습니다! HP:%d ATK:%d\n", hero_hp, hero_atk); break; default: hero_hp = 100; hero_atk = 20; crit_bonus = 10; printf("기본 직업으로 시작합니다.\n"); }

크리티컬 히트 — rand()로 확률 판정

크리티컬 히트 + 아이템 코드 (공격 case 1 부분 교체)
case 1: { /* 공격 + 크리티컬 히트 */ int roll = rand() % 100; /* 0~99 난수 생성 */ int dmg; if (roll < crit_bonus) { /* crit_bonus% 확률로 크리티컬 */ dmg = hero_atk * 2; /* 2배 데미지 */ printf(" 크리티컬 히트! %d 데미지!\n", dmg); } else { dmg = hero_atk; printf("용사의 공격! %d 데미지!\n", dmg); } monster_hp -= dmg; break; } /* ── 아이템 사용 (case 3 교체) ── */ case 3: { if (item_count > 0) { hero_hp += 30; if (hero_hp > max_hp) hero_hp = max_hp; item_count--; printf("포션 사용! HP +30 (남은 포션: %d개)\n", item_count); } else { printf("포션이 없습니다!\n"); } break; }
아이템 변수 추가 (게임 시작 전 선언)
int item_count = 3; — 포션 개수
int max_hp = hero_hp; — 직업별 최대 HP 저장 (회복 상한용)
Q
확인 문제 — 방어 + 크리티컬 동시 적용
중간

몬스터가 크리티컬 히트(30% 확률)를 칠 수 있도록 몬스터 반격 코드를 수정하라. 용사가 방어 중이면 크리티컬도 절반만 받는다.

내 답안
힌트 보기
몬스터 반격 직전에 int mroll = rand() % 100;으로 판정한다.
int mdmg = (mroll < 30) ? monster_atk * 2 : monster_atk;
그 다음 if (action == 2) mdmg /= 2;로 방어 처리.

전체 완성 코드

mini_rpg_final.c
#include <stdio.h> #include <stdlib.h> #include <time.h> int main() { srand(time(NULL)); /* ── 스탯 선언 ── */ int hero_hp, hero_atk, max_hp, crit_bonus; int monster_hp = 80; int monster_atk = 15; int item_count = 3; int action, turn = 1; int job; /* ── 직업 선택 ── */ printf("=== 미니 RPG ===\n직업 선택: 1=전사 2=마법사 3=궁수\n> "); scanf("%d", &job); switch (job) { case 1: hero_hp=120; hero_atk=25; crit_bonus=10; printf("전사!\n"); break; case 2: hero_hp=70; hero_atk=35; crit_bonus=10; printf("마법사!\n"); break; case 3: hero_hp=90; hero_atk=20; crit_bonus=20; printf("궁수!\n"); break; default: hero_hp=100; hero_atk=20; crit_bonus=10; } max_hp = hero_hp; printf("HP:%d ATK:%d 포션:%d개\n\n", hero_hp, hero_atk, item_count); /* ── 게임 루프 ── */ while (hero_hp > 0 && monster_hp > 0) { printf("--- 턴 %d | [용사] HP:%d / [몬스터] HP:%d ---\n", turn, hero_hp, monster_hp); printf("1=공격 2=방어 3=포션(%d개) > ", item_count); scanf("%d", &action); /* 용사 행동 */ switch (action) { case 1: { int roll = rand() % 100; int dmg = (roll < crit_bonus) ? hero_atk * 2 : hero_atk; if (roll < crit_bonus) printf(" 크리티컬! 몬스터에게 %d 데미지!\n", dmg); else printf("공격! 몬스터에게 %d 데미지!\n", dmg); monster_hp -= dmg; break; } case 2: printf("방어 태세!\n"); break; case 3: if (item_count > 0) { hero_hp += 30; if (hero_hp > max_hp) hero_hp = max_hp; item_count--; printf("포션 사용! HP+30 → %d (남은:%d)\n", hero_hp, item_count); } else { printf("포션 없음!\n"); } break; default: printf("잘못된 입력.\n"); } /* 몬스터 반격 */ if (monster_hp > 0) { int mroll = rand() % 100; int mdmg = (mroll < 30) ? monster_atk * 2 : monster_atk; if (action == 2) mdmg /= 2; hero_hp -= mdmg; printf("몬스터 공격%s → %d 데미지!\n", (mroll < 30) ? " (크리티컬!)" : "", mdmg); } turn++; printf("\n"); } /* 결과 */ printf("=== %s ===\n", (monster_hp <= 0) ? "승리! 몬스터를 물리쳤다!" : "게임 오버. 용사가 쓰러졌다..."); printf("총 %d턴 진행\n", turn - 1); return 0; }

이 프로젝트에서 사용한 기술 총정리

  • 변수 관리: hero_hp, monster_hp, item_count 등 다수의 변수를 동시 추적
  • while 루프: 두 조건 &&으로 게임 종료 시점 자동 판단
  • switch문: 직업 선택, 플레이어 행동 분기 — 2회 사용
  • rand() % 100: 0~99 난수로 확률 이벤트(크리티컬) 구현
  • 삼항 연산자: ? :로 크리티컬/일반 데미지 간결하게 표현
  • HP 상한 제한: 회복 후 if (hp > max_hp) hp = max_hp; 패턴

도전 과제

Lv.1 쉬움

몬스터 체력바 표시

몬스터 HP를 숫자 대신 # 기호로 표시하라.
(예: ######## 80/80)
for 루프를 사용해 HP/10 개수만큼 출력.

Lv.2 보통

스킬 게이지 시스템

공격할 때마다 스킬 포인트(SP)가 10 쌓인다. SP가 30 이상이면 행동 선택에 4=필살기 추가. 필살기는 공격력 3배, SP 30 소모.

Lv.2 보통

몬스터 2단계 분노

몬스터 HP가 40 이하로 떨어지면 공격력이 25로 증가하고 "몬스터가 분노했다!" 메시지 출력. if문 하나로 구현.

Lv.3 어려움

연속 전투 (보스전)

몬스터를 처치하면 HP를 20% 회복하고 더 강한 몬스터(HP+20, ATK+5)와 재전투. 3번 승리 시 최종 보스 등장.

Lv.3 어려움

장비 강화 시스템

전투 승리 시 골드 획득. 전투 전 상점에서 장비 구매(switch 메뉴)로 공격력 또는 최대 HP 업그레이드.

Lv.4 심화

2인 협동 모드

용사 A(공격형)와 용사 B(지원형) 두 캐릭터의 행동을 번갈아 입력. 두 캐릭터 중 한 명이라도 쓰러지면 게임 오버.

이 프로젝트를 완성했다면 — 변수, 조건문, 반복문, switch, rand()를 모두 사용한 것이다. 지금까지 배운 C언어 핵심 개념을 전부 녹여낸 종합 프로젝트를 완주한 걸 축하한다.
