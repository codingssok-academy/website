동적 연락처
추가/삭제/검색

연락처를 동적으로 추가/삭제하는 프로그램입니다. realloc으로 공간을 늘리고, 삭제 시 배열을 앞으로 당깁니다.

이 프로젝트에서 만드는 것

  • Contact 구조체 배열을 동적 관리합니다.
  • 연락처 추가 시 realloc으로 확장합니다.
  • 이름으로 검색하여 전화번호를 출력합니다.
  • 삭제 시 배열을 앞으로 당겨 빈자리를 채웁니다.
  • 전체 연락처 목록을 출력합니다.

완성 코드

#include <stdio.h> #include <stdlib.h> #include <string.h> typedef struct { char name[30]; char phone[20]; } Contact; Contact *list = NULL; int cnt = 0; void addContact(char*name, char*phone) { Contact *tmp=(Contact*)realloc(list, sizeof(Contact)*(cnt+1)); if(!tmp) return; list=tmp; strcpy(list[cnt].name, name); strcpy(list[cnt].phone, phone); cnt++; printf("%s 추가 완료\n", name); } void searchContact(char*name) { for(int i=0;i<cnt;i++) if(strcmp(list[i].name,name)==0){ printf("%s: %s\n",list[i].name,list[i].phone); return; } printf("%s 없음\n", name); } void deleteContact(char*name) { for(int i=0;i<cnt;i++) if(strcmp(list[i].name,name)==0){ for(int j=i;j<cnt-1;j++) list[j]=list[j+1]; cnt--; printf("%s 삭제\n",name); return; } printf("%s 없음\n",name); } void printAll() { printf("== 연락처 (%d명) ==\n",cnt); for(int i=0;i<cnt;i++) printf("%d. %s %s\n",i+1,list[i].name,list[i].phone); } int main() { addContact("홍길동","010-1111-2222"); addContact("김철수","010-3333-4444"); addContact("박영희","010-5555-6666"); printAll(); searchContact("김철수"); deleteContact("홍길동"); printAll(); free(list); list=NULL; return 0; }
