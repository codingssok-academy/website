문자 카운터
— 모음/자음/숫자/공백

입력된 문자열에서 모음, 자음, 숫자, 공백의 개수를 각각 세는 프로그램을 만듭니다.

사용 함수: fgets, strlen, isalpha, isdigit, isspace, toupper

핵심 개념: ctype.h 함수, 문자 분류, for 루프

프로젝트 설명

영어 문장을 입력받아 각 문자를 분류합니다. 모음(a,e,i,o,u), 자음(알파벳 중 모음 제외), 숫자, 공백으로 나눠 개수를 셉니다.

핵심 알고리즘 — 모음 판별
문자를 소문자로 변환 후 a,e,i,o,u인지 확인합니다.
// 모음 판별 함수 int is_vowel(char c) { c = tolower(c); return c=='a'||c=='e'||c=='i'||c=='o'||c=='u'; }

완성 코드

#include <stdio.h> #include <string.h> #include <ctype.h> int is_vowel(char c) { c = tolower(c); return c=='a'||c=='e'||c=='i'||c=='o'||c=='u'; } int main() { char str[100]; int vowel=0, consonant=0, digit=0, space=0, other=0; printf("문장 입력: "); fgets(str, sizeof(str), stdin); str[strlen(str)-1] = '\0'; for (int i = 0; i < (int)strlen(str); i++) { char c = str[i]; if (isalpha(c)) { if (is_vowel(c)) vowel++; else consonant++; } else if (isdigit(c)) digit++; else if (isspace(c)) space++; else other++; } printf("모음:%d 자음:%d 숫자:%d 공백:%d 기타:%d\n", vowel, consonant, digit, space, other); printf("총 문자 수: %lu\n", strlen(str)); return 0; }
