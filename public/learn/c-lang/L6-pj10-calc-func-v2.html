/* 연산 계층 — 순수한 계산만 */ double add (double a, double b) { return a + b; } double sub (double a, double b) { return a - b; } double mul (double a, double b) { return a * b; } double divFunc(double a, double b) { if (b == 0) { printf("0으로 나눌 수 없습니다!\n"); return 0; } return a / b; } /* 메뉴 계층 — 입출력만 */ void showMenu() { printf("\n1.덧셈 2.뺄셈 3.곱셈 4.나눗셈 5.히스토리 0.종료\n선택: "); } double history[20]; int histCount = 0; void saveHistory(double result) { if (histCount < 20) history[histCount++] = result; } void showHistory() { int i; printf("계산 히스토리:\n"); for (i = 0; i < histCount; i++) printf("%d. %.2f\n", i+1, history[i]); }
전역 변수 사용 주의: history와 histCount는 전역 변수로 선언했습니다. 전역 변수는 함수 밖에서 선언해 모든 함수가 공유합니다. 꼭 필요한 경우에만 씁니다.

계산기 함수 버전 — 전체 코드

/* ====== 계산기 함수 버전 (완성) ====== */ #include <stdio.h> double add(double a, double b) { return a+b; } double sub(double a, double b) { return a-b; } double mul(double a, double b) { return a*b; } double dv(double a, double b) { if(b==0){ printf("오류\n"); return 0; } return a/b; } double hist[20]; int hc=0; void saveHist(double r){ if(hc<20) hist[hc++]=r; } void showHist(){ int i; for(i=0;i<hc;i++) printf("%d: %.2f\n",i+1,hist[i]); } void process(int ch) { double a, b, r; if (ch>=1 && ch<=4) { printf("두 수 입력: "); scanf("%lf %lf",&a,&b); } switch(ch) { case 1: r=add(a,b); printf("= %.2f\n",r); saveHist(r); break; case 2: r=sub(a,b); printf("= %.2f\n",r); saveHist(r); break; case 3: r=mul(a,b); printf("= %.2f\n",r); saveHist(r); break; case 4: r=dv(a,b); printf("= %.2f\n",r); saveHist(r); break; case 5: showHist(); break; case 0: printf("종료!\n"); break; } } int main() { int ch; do { printf("1.덧 2.빼 3.곱 4.나 5.히스토리 0.종료: "); scanf("%d",&ch); process(ch); } while(ch != 0); return 0; }

계층 구조 이해

연산 함수만 따로 테스트 가능, 메뉴만 바꿔도 연산 함수 재사용 가능, 코드 읽기 쉬워짐

쉬움

saveHist, showHist 등 여러 함수에서 공유해야 하므로 전역으로 선언합니다. 지역 변수는 해당 함수 안에서만 접근 가능합니다.

핵심 정리

  • 계층 설계: main → 메뉴 함수 → 연산 함수 순서
