#include <stdio.h> #include <string.h> #include <ctype.h> void analyze(char text[]) { // === 1. 글자 종류별 통계 === int upper=0, lower=0, digit=0, space=0, punct=0; for (int i=0; text[i]; i++) { if (isupper(text[i])) upper++; else if (islower(text[i])) lower++; else if (isdigit(text[i])) digit++; else if (isspace(text[i])) space++; else punct++; } // === 2. 문장 수 (., !, ? 기준) === int sentences = 0; for (int i=0; text[i]; i++) if (text[i]=='.' || text[i]=='!' || text[i]=='?') sentences++; // === 3. 단어 수 + 최장 단어 (strtok) === char copy[500]; strcpy(copy, text); // strtok이 원본 수정하므로 복사 int word_cnt=0, total_len=0, max_len=0; char longest[50] = ""; char *tok = strtok(copy, " \t\n,.!?;:"); while (tok != NULL) { int wlen = strlen(tok); word_cnt++; total_len += wlen; if (wlen > max_len) { max_len = wlen; strcpy(longest, tok); } tok = strtok(NULL, " \t\n,.!?;:"); } // === 결과 출력 === printf("=== 문장 분석 결과 ===\n"); printf("전체 길이 : %d\n", (int)strlen(text)); printf("단어 수 : %d\n", word_cnt); printf("문장 수 : %d\n", sentences); printf("평균 단어길이: %.1f\n", word_cnt?(float)total_len/word_cnt:0); printf("가장 긴 단어: \"%s\" (%d글자)\n", longest, max_len); printf("대문자/소문자/숫자/공백/기타: %d/%d/%d/%d/%d\n", upper, lower, digit, space, punct); } int main() { char text[500]; printf("텍스트 입력: "); fgets(text, sizeof(text), stdin); text[strcspn(text, "\n")] = '\0'; analyze(text); return 0; }

Q1. 분석 결과 예측

"Hello World!" 를 analyze() 함수에 전달하면 단어 수와 문장 수는?

정답 보기
단어 수: 2 ("Hello", "World") / 문장 수: 1 ('!')
Q2. strtok 복사 이유

코드에서 strtok을 호출하기 전에 strcpy(copy, text)로 복사하는 이유는?

정답 보기
strtok은 원본 문자열을 수정(구분자를 '\0'으로 덮어씀)한다. 원본을 보존해야 하므로 복사본에 strtok을 적용한다.

정답 보기
