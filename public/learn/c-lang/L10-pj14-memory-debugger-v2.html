메모리 디버거
주소 추적 + 누수 리포트

malloc/free를 래핑하여 할당 주소와 크기를 추적하는 진화된 메모리 디버거를 만듭니다. 프로그램 종료 시 해제되지 않은 블록의 주소와 크기를 출력합니다.

이 프로젝트에서 만드는 것

  • AllocRecord 구조체로 포인터/크기/파일/라인을 기록합니다.
  • dbg_malloc()이 할당 시 레코드를 등록합니다.
  • dbg_free()가 해제 시 레코드를 삭제합니다.
  • dbg_report()가 미해제 레코드를 모두 출력합니다.
  • 매크로로 malloc/free를 자동 교체합니다.

완성 코드

#include <stdio.h> #include <stdlib.h> #include <string.h> #define MAX_RECORDS 256 typedef struct { void *ptr; size_t size; const char *file; int line; } AllocRecord; AllocRecord records[MAX_RECORDS]; int rec_count = 0; void* dbg_malloc(size_t size, const char*file, int line) { void*p = malloc(size); if(p && rec_count < MAX_RECORDS) { records[rec_count++] = (AllocRecord){p, size, file, line}; printf("[M] %p %zu바이트 (%s:%d)\n", p, size, file, line); } return p; } void dbg_free(void*ptr, const char*file, int line) { if(!ptr) return; for(int i=0;i<rec_count;i++) { if(records[i].ptr==ptr) { printf("[F] %p (%s:%d)\n", ptr, file, line); records[i]=records[--rec_count]; // 마지막과 교환 후 제거 free(ptr); return; } } printf("[!] 미등록 포인터 해제: %p (%s:%d)\n", ptr, file, line); free(ptr); } void dbg_report() { printf("\n=== 메모리 누수 리포트 ===\n"); if(rec_count==0) { printf("누수 없음!\n"); return; } size_t total=0; for(int i=0;i<rec_count;i++) { printf(" 누수: %p %zu바이트 (%s:%d)\n", records[i].ptr, records[i].size, records[i].file, records[i].line); total += records[i].size; } printf("합계: %d개 %zu바이트 미해제\n", rec_count, total); } /* 매크로로 자동 교체 */ #define malloc(s) dbg_malloc(s, __FILE__, __LINE__) #define free(p) dbg_free(p, __FILE__, __LINE__) int main() { int *a = (int*)malloc(40); char *b = (char*)malloc(100); double *c = (double*)malloc(80); free(a); // b, c 미해제 → 누수! dbg_report(); return 0; }
