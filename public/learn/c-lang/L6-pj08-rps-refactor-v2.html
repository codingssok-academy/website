가위(1)무패승
바위(2)승무패
보(3)패승무

3개 핵심 함수

// 1. 사용자 입력 받기 int getChoice() { int c; printf("1=가위 2=바위 3=보: "); scanf("%d", &c); return c; } // 2. 승패 판정 (1=승, -1=패, 0=무) int judge(int player, int computer) { if (player == computer) return 0; // 이기는 경우: 가위>보, 바위>가위, 보>바위 if ((player==1 && computer==3) || (player==2 && computer==1) || (player==3 && computer==2)) return 1; return -1; } // 3. 결과 출력 void printResult(int result) { if (result == 1) printf("당신이 이겼습니다!\n"); else if (result == -1) printf("컴퓨터가 이겼습니다.\n"); else printf("무승부입니다.\n"); }

가위바위보 — 전체 코드

/* ====== 가위바위보 리팩토링 ====== */ #include <stdio.h> #include <stdlib.h> #include <time.h> int getChoice() { int c; printf("1=가위 2=바위 3=보 0=종료: "); scanf("%d",&c); return c; } int getComputer() { return rand()%3+1; } int judge(int p, int c) { if(p==c) return 0; if((p==1&&c==3)||(p==2&&c==1)||(p==3&&c==2)) return 1; return -1; } char* getName(int n) { if(n==1) return "가위"; if(n==2) return "바위"; return "보"; } void printResult(int p, int c, int result) { printf("나: %s 컴: %s ", getName(p), getName(c)); if (result== 1) printf("[승리!]\n"); else if (result==-1) printf("[패배]\n"); else printf("[무승부]\n"); } int main() { srand(time(0)); int player, computer, result, wins=0, losses=0, draws=0; while (1) { player = getChoice(); if (player == 0) break; computer = getComputer(); result = judge(player, computer); printResult(player, computer, result); if(result== 1) wins++; else if(result==-1) losses++; else draws++; } printf("최종: %d승 %d패 %d무\n", wins, losses, draws); return 0; }

핵심 정리

  • 입력(getChoice) → 처리(judge) → 출력(printResult) 3단계 분리
  • judge: 같으면 0, 이기면 1, 지면 -1 반환
  • char* 반환으로 문자열 리터럴을 넘길 수 있음
  • while(1) + break로 무한 반복 메뉴 구현
