동적 문자열 입력
길이를 모를 때

길이를 미리 모르는 문자열을 동적으로 처리합니다. strlen로 길이를 측정하고 malloc으로 정확한 크기만 할당합니다.

이 프로젝트에서 만드는 것

  • 임시 버퍼로 문자열을 읽어옵니다.
  • strlen으로 실제 길이를 측정합니다.
  • malloc(len+1)로 딱 맞는 크기를 할당합니다.
  • strcpy로 동적 메모리에 복사합니다.
  • 사용 후 free로 해제합니다.

동적 문자열 — 왜 필요한가?

고정 크기 문자 배열은 미리 큰 공간을 잡아야 합니다. 동적 할당을 사용하면 실제 문자열 길이에 딱 맞는 공간만 사용할 수 있습니다.

문자열 끝에는 반드시 '\0'(null 문자)이 필요합니다. 그래서 malloc(strlen(s) + 1)처럼 +1 합니다.
#include <stdio.h> #include <stdlib.h> #include <string.h> int main() { char buf[1024]; printf("문자열 입력: "); fgets(buf, sizeof(buf), stdin); // 줄바꿈 제거 int len = strlen(buf); if (buf[len-1] == '\n') buf[len-1] = '\0', len--; // 딱 맞는 크기로 동적 할당 char *str = (char*)malloc(len + 1); if (str == NULL) return 1; strcpy(str, buf); printf("입력: %s\n", str); printf("길이: %d\n", (int)strlen(str)); printf("메모리: %d바이트\n", len+1); free(str); str = NULL; return 0; }

여러 문자열 동적 배열

#include <stdio.h> #include <stdlib.h> #include <string.h> int main() { int n; printf("이름 개수: "); scanf("%d", &n); getchar(); // 개행 문자 제거 // n개의 문자열 포인터 배열 char **names = (char**)malloc(sizeof(char*) * n); if (!names) return 1; char buf[256]; for (int i=0; i<n; i++) { printf("%d번 이름: ", i+1); fgets(buf, sizeof(buf), stdin); int len = strlen(buf); if (buf[len-1]=='\n') buf[--len]='\0'; names[i] = (char*)malloc(len+1); strcpy(names[i], buf); } printf("\n입력된 이름:\n"); for (int i=0; i<n; i++) printf("%d. %s\n", i+1, names[i]); for (int i=0; i<n; i++) { free(names[i]); names[i]=NULL; } free(names); names=NULL; return 0; }
이름 개수: 3 1번 이름: 홍길동 2번 이름: 김철수 3번 이름: 박영희 입력된 이름: 1. 홍길동 2. 김철수 3. 박영희
