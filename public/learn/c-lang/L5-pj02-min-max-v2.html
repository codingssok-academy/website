최대값 최소값 찾기

N개의 숫자를 입력받아 최대값과 최소값, 그리고 그 위치(인덱스)를 출력하는 프로그램을 만들어봅니다.

이 프로젝트에서 배우는 것

  • N개의 값을 배열에 입력받는 방법을 익힙니다.
  • 최대값과 그 위치를 동시에 추적하는 알고리즘을 만듭니다.
  • 최소값 탐색 알고리즘도 함께 구현합니다.
  • 인덱스와 실제 위치(1번~N번)의 차이를 이해합니다.

N개 입력받기 — 동적 크기 배열

이번에는 개수 N을 사용자가 직접 입력합니다. N을 먼저 받고, N개의 값을 배열에 저장합니다. 배열 크기는 충분히 크게 잡아야 합니다.

int n; printf("개수 입력: "); scanf("%d", &n); int arr[100]; // 최대 100개까지 입력 가능 for (int i = 0; i < n; i++) { printf("%d번째 숫자: ", i + 1); scanf("%d", &arr[i]); }

최대/최소 동시 탐색

하나의 for문에서 최대값과 최소값을 동시에 찾을 수 있습니다. 각각의 비교 조건을 if로 구분합니다.

int max = arr[0], maxIdx = 0; int min = arr[0], minIdx = 0; for (int i = 1; i < n; i++) { if (arr[i] > max) { max = arr[i]; maxIdx = i; } if (arr[i] < min) { min = arr[i]; minIdx = i; } } printf("최대값: %d (위치: %d번째)\n", max, maxIdx + 1); printf("최소값: %d (위치: %d번째)\n", min, minIdx + 1);

완성 코드

// 최대값 최소값 찾기 — 완성 코드 #include <stdio.h> int main() { int n, arr[100]; printf("개수 입력 (최대 100): "); scanf("%d", &n); for (int i = 0; i < n; i++) { printf("%d번째: ", i + 1); scanf("%d", &arr[i]); } int max = arr[0], maxIdx = 0; int min = arr[0], minIdx = 0; for (int i = 1; i < n; i++) { if (arr[i] > max) { max = arr[i]; maxIdx = i; } if (arr[i] < min) { min = arr[i]; minIdx = i; } } printf("\n최대값: %d → %d번째\n", max, maxIdx + 1); printf("최소값: %d → %d번째\n", min, minIdx + 1); printf("범위(max-min): %d\n", max - min); return 0; }

정답 보기
int pos=0, neg=0; for(int i=0;i<n;i++){ if(arr[i]>0) pos++; else if(arr[i]<0) neg++; } printf("양수:%d 음수:%d\n",pos,neg);

핵심 정리

  • N을 먼저 입력받고 arr[100] 크기로 넉넉하게 선언
  • max/min 초기값은 항상 arr[0]으로 설정
  • i=1부터 시작하는 이유: arr[0]은 이미 초기값으로 사용
  • 위치 출력 시 maxIdx+1 (인덱스는 0부터이므로)
  • 범위 = max - min으로 데이터의 분포를 파악

심화 — 인덱스 배열 활용

같은 최대값이 여러 개 있을 때 모든 위치를 출력하는 방법도 있습니다. 별도 인덱스 배열을 만들어 저장합니다.

// 중복 최대값 위치 모두 찾기 int maxPositions[100]; int cnt = 0; for (int i = 0; i < n; i++) { if (arr[i] == max) { maxPositions[cnt++] = i + 1; // 1번부터 세는 위치 } } printf("최대값 %d의 위치: ", max); for (int i = 0; i < cnt; i++) { printf("%d번째 ", maxPositions[i]); } printf("\n");
