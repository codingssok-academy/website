병렬 배열로 학생 데이터 관리

구조체 없이 이름 배열과 점수 배열을 같은 인덱스로 짝을 맞춥니다.

// 병렬 배열: 같은 인덱스가 한 학생 char names[10][20]; // 이름 최대 10명, 각 20글자 int scores[10]; // 점수 int n = 0; // 실제 학생 수 // 입력 함수 void inputScores() { printf("학생 수: "); scanf("%d", &n); int i; for (i = 0; i < n; i++) { printf("%d번 이름: ", i+1); scanf("%s", names[i]); printf("%d번 점수: ", i+1); scanf("%d", &scores[i]); } }

포인터 매개변수로 여러 값 반환

함수는 값을 하나만 반환합니다. 여러 값(평균, 최대, 최소)을 돌려주려면 포인터를 씁니다.

// 포인터로 여러 값 반환 void calcStats(double *avg, int *maxS, int *minS) { int sum = 0, i; *maxS = scores[0]; *minS = scores[0]; for (i = 0; i < n; i++) { sum += scores[i]; if (scores[i] > *maxS) *maxS = scores[i]; if (scores[i] < *minS) *minS = scores[i]; } *avg = (double)sum / n; }
포인터 매개변수: *avg처럼 별표(*)가 붙으면 포인터입니다. 함수를 호출할 때 &a(주소)를 넘기면, 함수 안에서 *avg = 값으로 원본 변수를 직접 바꿀 수 있습니다.

함수형 성적 시스템 — 전체 코드

/* ====== 함수형 성적 시스템 ====== */ #include <stdio.h> #include <string.h> #define MAX 10 char names[MAX][20]; int scores[MAX]; int n; void inputScores() { printf("학생 수: "); scanf("%d",&n); int i; for(i=0;i<n;i++){ printf("%d번 이름: ",i+1); scanf("%s",names[i]); printf("%d번 점수: ",i+1); scanf("%d",&scores[i]); } } char getGrade(int s) { if(s>=90)return'A'; if(s>=80)return'B'; if(s>=70)return'C'; if(s>=60)return'D'; return'F'; } void printReport() { int sum=0,mx=scores[0],mn=scores[0],i; printf("%-8s %4s %2s\n","이름","점수","등급"); printf("-----------------\n"); for(i=0;i<n;i++){ printf("%-8s %4d %c\n",names[i],scores[i],getGrade(scores[i])); sum+=scores[i]; if(scores[i]>mx) mx=scores[i]; if(scores[i]<mn) mn=scores[i]; } printf("평균:%.1f 최고:%d 최저:%d\n",(double)sum/n,mx,mn); } int main() { inputScores(); printReport(); return 0; }
