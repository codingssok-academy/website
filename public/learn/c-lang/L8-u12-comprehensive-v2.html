L8 종합 예제 — 학생 성적 관리

포인터의 여러 기법을 한 예제에서 사용해봅니다.

#include <stdio.h> // 포인터 매개변수로 합계와 평균 계산 void calc_stats(const int *scores, int n, int *sum, float *avg) { *sum = 0; for (int i=0; i<n; i++) *sum += scores[i]; *avg = (float)*sum / n; } // 포인터로 최대값 위치 반환 int* find_max(int *arr, int n) { int *max = arr; for (int i=1; i<n; i++) if (arr[i] > *max) max = arr+i; return max; } int main() { int s[] = {85, 92, 78, 96, 88}; int sum; float avg; calc_stats(s, 5, &sum, &avg); printf("합계=%d, 평균=%.1f\n", sum, avg); int *pmax = find_max(s, 5); printf("최고점=%d (인덱스=%td)\n", *pmax, pmax-s); return 0; } // 합계=439, 평균=87.8 // 최고점=96 (인덱스=3)

void minmax(int a, int b, int *big, int *small){ if(a>b){ *big=a; *small=b; } else { *big=b; *small=a; } } // 호출: minmax(10,3,&mx,&mn); → mx=10, mn=3
