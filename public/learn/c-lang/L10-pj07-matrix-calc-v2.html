동적 행렬 곱셈
int** 이중 포인터

크기를 입력받아 동적 2D 배열로 행렬을 만들고 행렬 곱셈을 수행합니다. int** 이중 포인터 활용의 실전 예제입니다.

이 프로젝트에서 만드는 것

  • 행렬 크기를 입력받아 int** 로 동적 할당합니다.
  • 행렬 A, B에 값을 입력받습니다.
  • 행렬 곱셈 C = A × B를 계산합니다.
  • 결과 행렬 C를 출력합니다.
  • 세 행렬을 모두 안전하게 해제합니다.

행렬 곱셈 공식

A(m×k) × B(k×n) = C(m×n) 에서 C[i][j] = A[i][0]*B[0][j] + A[i][1]*B[1][j] + ... + A[i][k-1]*B[k-1][j]

#include <stdio.h> #include <stdlib.h> int** allocMatrix(int r, int c) { int **m = (int**)malloc(sizeof(int*)*r); for(int i=0;i<r;i++) m[i]=(int*)malloc(sizeof(int)*c); return m; } void freeMatrix(int**m, int r) { for(int i=0;i<r;i++) { free(m[i]); m[i]=NULL; } free(m); } void multiply(int**A,int**B,int**C,int m,int k,int n) { for(int i=0;i<m;i++) for(int j=0;j<n;j++){ C[i][j]=0; for(int p=0;p<k;p++) C[i][j]+=A[i][p]*B[p][j]; } } void printMatrix(int**m, int r, int c) { for(int i=0;i<r;i++){ for(int j=0;j<c;j++) printf("%5d",m[i][j]); printf("\n"); } } int main() { int m=2,k=2,n=2; int **A=allocMatrix(m,k), **B=allocMatrix(k,n), **C=allocMatrix(m,n); // A=[1,2;3,4] B=[5,6;7,8] A[0][0]=1;A[0][1]=2;A[1][0]=3;A[1][1]=4; B[0][0]=5;B[0][1]=6;B[1][0]=7;B[1][1]=8; multiply(A,B,C,m,k,n); printf("결과:\n"); printMatrix(C,m,n); freeMatrix(A,m); freeMatrix(B,k); freeMatrix(C,m); return 0; }
