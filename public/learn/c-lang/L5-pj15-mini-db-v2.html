미니 데이터베이스

병렬 배열(이름/점수)로 여러 레코드를 관리하고, 메뉴를 통해 검색·정렬·삭제 기능을 구현합니다.

이 프로젝트에서 배우는 것

  • 병렬 배열로 여러 필드를 가진 레코드를 관리합니다.
  • 메뉴 기반 프로그램 구조를 설계합니다.
  • 이름으로 검색하는 선형 탐색을 구현합니다.
  • 레코드를 삭제할 때 배열을 앞으로 당기는 방법을 익힙니다.
  • 이름 비교에 strcmp 함수를 사용합니다.

병렬 배열 설계 & 검색

이름 배열과 점수 배열을 같은 인덱스로 연결합니다. 현재 레코드 수를 n으로 관리합니다.

#include <stdio.h> #include <string.h> #define MAX 50 char names[MAX][20]; // 이름 (최대 19글자+null) int scores[MAX]; // 점수 int n = 0; // 현재 레코드 수 // 이름으로 검색 — 인덱스 반환 (없으면 -1) int search(char *name) { for (int i=0;i<n;i++) if (strcmp(names[i], name) == 0) return i; return -1; }

추가 & 삭제 구현

// 추가 void addRecord(char *name, int score) { if (n >= MAX) { printf("가득 찼습니다!\n"); return; } strcpy(names[n], name); scores[n] = score; n++; } // 삭제 (인덱스 기준) void deleteRecord(int idx) { for (int i=idx; i<n-1; i++) { strcpy(names[i], names[i+1]); scores[i] = scores[i+1]; } n--; }

메뉴 기반 완성 코드

void printAll() { if(n==0){printf("데이터 없음\n");return;} printf("%-12s %6s\n","이름","점수"); for(int i=0;i<n;i++) printf("%-12s %6d\n", names[i], scores[i]); } void sortByScore() { // 버블 정렬 (내림차순) for(int i=0;i<n-1;i++) for(int j=0;j<n-1-i;j++) if(scores[j]<scores[j+1]){ int ts=scores[j];scores[j]=scores[j+1];scores[j+1]=ts; char tn[20]; strcpy(tn,names[j]); strcpy(names[j],names[j+1]); strcpy(names[j+1],tn); } } int main() { addRecord("김민준",88); addRecord("이서연",94); addRecord("박도현",72); int choice; do { printf("\n1.목록 2.추가 3.검색 4.정렬 5.삭제 0.종료: "); scanf("%d", &choice); if(choice==1) { printAll(); } else if(choice==2) { char nm[20]; int sc; printf("이름 점수: "); scanf("%s %d",nm,&sc); addRecord(nm,sc); } else if(choice==3) { char nm[20]; printf("검색 이름: "); scanf("%s",nm); int idx=search(nm); if(idx>=0) printf("%s: %d점\n",names[idx],scores[idx]); else printf("없음\n"); } else if(choice==4) { sortByScore(); printAll(); } else if(choice==5) { char nm[20]; printf("삭제 이름: "); scanf("%s",nm); int idx=search(nm); if(idx>=0){deleteRecord(idx);printf("삭제됨\n");} else printf("없음\n"); } } while(choice!=0); return 0; }
