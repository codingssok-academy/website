메모리 풀
고정 블록 할당기

큰 메모리를 한 번에 확보해두고 고정 크기 블록 단위로 나눠 쓰는 간단한 메모리 풀(Pool)을 구현합니다. malloc 호출 횟수를 줄여 성능을 높이는 원리를 이해합니다.

이 프로젝트에서 만드는 것

  • 고정 크기 블록을 가진 메모리 풀 구조체를 설계합니다.
  • 풀 초기화 시 한 번에 메모리를 malloc으로 확보합니다.
  • pool_alloc()으로 블록 하나를 대여합니다.
  • pool_free()로 블록을 반납하고 재사용 가능하게 합니다.
  • 풀 전체를 한 번의 free()로 해제합니다.

메모리 풀 구조 설계

메모리 풀은 미리 큰 메모리를 확보해두고, 블록 단위로 나눠 쓰는 방식입니다. free() 호출 대신 "반납" 표시만 해두고 재활용합니다.

#include <stdio.h> #include <stdlib.h> #include <string.h> #define POOL_BLOCK_SIZE 64 // 블록 하나 크기 (바이트) #define POOL_COUNT 8 // 블록 개수 typedef struct { char *buf; // 실제 데이터 메모리 int used[POOL_COUNT]; // 사용 여부 (1=사용중) } MemPool; /* 풀 초기화 — 한 번만 malloc */ int pool_init(MemPool *p) { p->buf = (char*)malloc(POOL_BLOCK_SIZE * POOL_COUNT); if (!p->buf) return 0; memset(p->used, 0, sizeof(p->used)); printf("풀 초기화: %d바이트 확보\n", POOL_BLOCK_SIZE * POOL_COUNT); return 1; } /* 블록 대여 — 빈 슬롯 탐색 */ void* pool_alloc(MemPool *p) { for (int i = 0; i < POOL_COUNT; i++) { if (!p->used[i]) { p->used[i] = 1; printf(" alloc: 블록[%d] 대여\n", i); return p->buf + (i * POOL_BLOCK_SIZE); } } printf(" alloc: 풀 가득!\n"); return NULL; } /* 블록 반납 — 포인터로 슬롯 번호 계산 */ void pool_free(MemPool *p, void *ptr) { int idx = ((char*)ptr - p->buf) / POOL_BLOCK_SIZE; if (idx < 0 || idx >= POOL_COUNT) { printf("잘못된 포인터\n"); return; } p->used[idx] = 0; printf(" free: 블록[%d] 반납\n", idx); } /* 풀 전체 해제 */ void pool_destroy(MemPool *p) { free(p->buf); p->buf = NULL; printf("풀 해제 완료\n"); } int main() { MemPool pool; if (!pool_init(&pool)) return 1; char *a = (char*)pool_alloc(&pool); char *b = (char*)pool_alloc(&pool); char *c = (char*)pool_alloc(&pool); sprintf(a, "Hello"); sprintf(b, "World"); printf("a=%s b=%s\n", a, b); pool_free(&pool, b); // b 반납 char *d = (char*)pool_alloc(&pool); // 블록[1] 재사용 sprintf(d, "Reused!"); printf("d=%s\n", d); pool_destroy(&pool); return 0; }
