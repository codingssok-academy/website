동적 도서관 관리 시스템

C언어 교재의 마지막 프로젝트입니다. 지금까지 배운 모든 개념을 통합하여 실전 수준의 도서관 관리 시스템을 완성합니다.

통합하는 핵심 개념

  • 구조체 + 동적 할당 (malloc, realloc, free)
  • 포인터 배열과 문자열 처리
  • 연결 리스트로 대출 이력 관리
  • 함수 포인터로 정렬 기준 선택
  • 파일 입출력으로 데이터 저장/불러오기
  • 메모리 누수 없이 완전한 해제

시스템 구조 설계

Book 구조체와 LoanLog 연결 리스트를 결합한 완전한 도서관 시스템입니다.

#include <stdio.h> #include <stdlib.h> #include <string.h> /* ── 구조체 정의 ── */ typedef struct { int id; char title[64]; char author[32]; int year; int loaned; // 0=보유 1=대출중 } Book; typedef struct LoanLog { int book_id; char date[12]; // "2025-03-27" int returned; struct LoanLog *next; } LoanLog; typedef struct { Book **books; // 동적 포인터 배열 int count; int capacity; LoanLog *log_head; // 대출 이력 연결 리스트 } Library; /* ── 라이브러리 초기화 ── */ Library* lib_create() { Library *lib = (Library*)malloc(sizeof(Library)); lib->capacity = 4; lib->books = (Book**)malloc(sizeof(Book*) * lib->capacity); lib->count = 0; lib->log_head = NULL; return lib; }

핵심 함수 구현

/* ── 도서 추가 (capacity 초과 시 2배 확장) ── */ void lib_add_book(Library *lib, int id, const char*title, const char*author, int year) { if (lib->count == lib->capacity) { lib->capacity *= 2; lib->books = (Book**)realloc(lib->books, sizeof(Book*)*lib->capacity); } Book *b = (Book*)malloc(sizeof(Book)); b->id=id; b->loaned=0; b->year=year; strcpy(b->title,title); strcpy(b->author,author); lib->books[lib->count++] = b; } /* ── 대출 처리 + 이력 추가 ── */ int lib_loan(Library *lib, int id, const char*date) { for(int i=0;i<lib->count;i++) { if(lib->books[i]->id==id) { if(lib->books[i]->loaned) { printf("이미 대출중\n"); return 0; } lib->books[i]->loaned = 1; /* 이력 앞에 삽입 */ LoanLog *log=(LoanLog*)malloc(sizeof(LoanLog)); log->book_id=id; log->returned=0; strcpy(log->date,date); log->next=lib->log_head; lib->log_head=log; printf("대출: [%d] %s (%s)\n",id,lib->books[i]->title,date); return 1; } } printf("도서 없음: %d\n", id); return 0; } /* ── 반납 처리 ── */ void lib_return(Library *lib, int id) { for(int i=0;i<lib->count;i++) if(lib->books[i]->id==id) { lib->books[i]->loaned=0; break; } for(LoanLog*l=lib->log_head;l;l=l->next) if(l->book_id==id && !l->returned){ l->returned=1; break; } printf("반납 완료: %d\n", id); } /* ── 함수 포인터로 정렬 (제목/연도 기준) ── */ int cmp_title(const void*a, const void*b) { return strcmp((*(Book**)a)->title, (*(Book**)b)->title); } int cmp_year(const void*a, const void*b) { return (*(Book**)a)->year - (*(Book**)b)->year; } /* ── 목록 출력 ── */ void lib_print(Library *lib) { printf("== 도서 목록 (%d권) ==\n", lib->count); for(int i=0;i<lib->count;i++) { Book*b=lib->books[i]; printf(" [%d] %s / %s / %d년 / %s\n", b->id, b->title, b->author, b->year, b->loaned ? "대출중" : "보유"); } } /* ── 대출 이력 출력 ── */ void lib_print_log(Library *lib) { printf("== 대출 이력 ==\n"); for(LoanLog*l=lib->log_head;l;l=l->next) printf(" book_id=%d %s %s\n", l->book_id, l->date, l->returned?"반납완료":"대출중"); } /* ── 전체 해제 (2단계: 노드 → 배열 → 구조체) ── */ void lib_destroy(Library *lib) { for(int i=0;i<lib->count;i++) free(lib->books[i]); free(lib->books); LoanLog*l=lib->log_head,*nx; while(l){nx=l->next;free(l);l=nx;} free(lib); printf("라이브러리 해제 완료\n"); } /* ── main ── */ int main() { Library *lib = lib_create(); lib_add_book(lib,1,"C Programming","Ritchie",1978); lib_add_book(lib,2,"Algorithms","Cormen",2009); lib_add_book(lib,3,"Clean Code","Martin",2008); lib_add_book(lib,4,"The Pragmatic","Hunt",1999); lib_add_book(lib,5,"Design Patterns","GoF",1994); lib_print(lib); lib_loan(lib,1,"2025-03-10"); lib_loan(lib,3,"2025-03-15"); lib_loan(lib,1,"2025-03-20"); // 이미 대출중 lib_return(lib,1); lib_print_log(lib); qsort(lib->books, lib->count, sizeof(Book*), cmp_title); printf("-- 제목순 정렬 --\n"); lib_print(lib); lib_destroy(lib); return 0; }
