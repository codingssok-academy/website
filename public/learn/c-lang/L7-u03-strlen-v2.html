strlen 동작원리
— \0까지 세기

strlen이 \0을 만날 때까지 문자를 세는 원리를 이해하고, 직접 구현해봅니다.

이 유닛에서 배우는 것

  • strlen이 \0을 만날 때까지 세는 원리를 설명할 수 있습니다.
  • strlen의 반환값이 \0을 제외한 글자 수임을 알 수 있습니다.
  • strlen과 sizeof의 차이를 구분할 수 있습니다.
  • strlen과 같은 함수를 직접 while 루프로 구현할 수 있습니다.
  • strlen을 활용해 문자열 마지막 문자에 접근할 수 있습니다.

strlen — \0을 만날 때까지 세기

strlen은 문자열의 첫 번째 문자부터 시작해서 \0을 만나면 멈추고, 그때까지 센 개수를 반환합니다. \0 자체는 세지 않습니다.

#include <stdio.h> #include <string.h> int main() { char s[] = "Hello"; printf("strlen: %lu\n", strlen(s)); // 5 printf("sizeof: %lu\n", sizeof(s)); // 6 (\0 포함) // 마지막 문자 접근 printf("마지막: %c\n", s[strlen(s) - 1]); // o return 0; }

strlen 직접 구현

strlen의 원리를 이해하면 직접 만들 수 있습니다. \0을 만날 때까지 카운터를 증가시키면 됩니다.

// strlen을 직접 구현한 함수 int my_strlen(char str[]) { int count = 0; while (str[count] != '\0') { count++; } return count; } // 포인터 버전 (나중에 배울 내용) int my_strlen2(char *p) { int count = 0; while (*p++ != '\0') count++; return count; }

strlen 활용 패턴

strlen은 문자열을 다루는 많은 곳에서 사용됩니다. 대표적인 활용 패턴을 익혀봅니다.

#include <stdio.h> #include <string.h> int main() { char s[20] = "Hello"; // 1. 문자열 길이로 반복 for (int i = 0; i < (int)strlen(s); i++) { printf("%c ", s[i]); // H e l l o } printf("\n"); // 2. 마지막 문자 확인 printf("마지막: %c\n", s[strlen(s) - 1]); // o // 3. 빈 문자열 체크 char empty[] = ""; if (strlen(empty) == 0) printf("빈 문자열\n"); return 0; }

핵심 정리

핵심 요약

  • strlen은 \0을 만날 때까지 문자를 세어 반환합니다.
  • \0은 세지 않으므로 strlen("Hello") = 5입니다.
  • sizeof는 \0 포함 전체 크기, strlen은 글자 수만 셉니다.
  • 마지막 문자: s[strlen(s)-1], 빈 문자열: strlen(s)==0
  • while(s[i] != '\0') i++; 로 직접 구현 가능합니다.

전체 예제 — strlen 종합 활용

#include <stdio.h> #include <string.h> // 직접 구현한 strlen int my_strlen(char s[]) { int i = 0; while (s[i] != '\0') i++; return i; } int main() { char word[50]; printf("단어 입력: "); scanf("%s", word); int len = strlen(word); printf("표준 strlen: %d\n", len); printf("직접 구현: %d\n", my_strlen(word)); printf("첫 글자: %c\n", word[0]); printf("마지막: %c\n", word[len-1]); // 각 문자와 인덱스 출력 for (int i = 0; i < len; i++) printf("[%d]=%c ", i, word[i]); printf("\n"); return 0; }
