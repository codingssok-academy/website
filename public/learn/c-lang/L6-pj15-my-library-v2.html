헤더 파일 구조 + 유틸 함수 10개

헤더 파일(.h)이란?

여러 .c 파일에서 공통으로 쓸 함수를 묶어둔 파일입니다. #include "myutil.h" 한 줄로 모든 함수를 불러옵니다.

/* myutil.h — 나만의 유틸리티 라이브러리 */ /* #ifndef 가드: 중복 include 방지 */ #ifndef MYUTIL_H #define MYUTIL_H #include <stdio.h> #include <string.h> #include <stdlib.h> /* 함수 선언 + 구현 모두 헤더에 */ int max2(int a, int b) { return a > b ? a : b; } int min2(int a, int b) { return a < b ? a : b; } int clamp(int v, int lo, int hi) { if(v < lo) return lo; if(v > hi) return hi; return v; } /* ... 나머지 함수들 ... */ #endif /* MYUTIL_H */

myutil.h — 완전한 구현

/* ====== myutil.h — 나만의 유틸 라이브러리 ====== */ #ifndef MYUTIL_H #define MYUTIL_H #include <stdio.h> #include <stdlib.h> #include <time.h> /* 1. 두 수 중 큰 값 */ int max2(int a, int b) { return a>b?a:b; } /* 2. 두 수 중 작은 값 */ int min2(int a, int b) { return a<b?a:b; } /* 3. 범위 제한 */ int clamp(int v, int lo, int hi) { if(v<lo)return lo; if(v>hi)return hi; return v; } /* 4. 절대값 */ int abs2(int n) { return n>=0?n:-n; } /* 5. 소수 판별 */ int isPrime(int n) { if(n<2) return 0; int i; for(i=2;i*i<=n;i++) if(n%i==0) return 0; return 1; } /* 6. 팩토리얼 (재귀) */ int factorial(int n) { return n<=1?1:n*factorial(n-1); } /* 7. 배열 합계 */ int arrSum(int a[], int n) { int s=0,i; for(i=0;i<n;i++) s+=a[i]; return s; } /* 8. 배열 평균 */ double arrAvg(int a[], int n) { return (double)arrSum(a,n)/n; } /* 9. 구분선 출력 */ void printLine(int n) { int i; for(i=0;i<n;i++) printf("-"); printf("\n"); } /* 10. 범위 난수 */ int randRange(int lo, int hi) { return rand()%(hi-lo+1)+lo; } #endif
/* main.c — myutil.h 사용 예시 */ #include <stdio.h> #include "myutil.h" // 나만의 라이브러리! int main() { srand(time(0)); printf("max(10,20)=%d\n", max2(10,20)); printf("clamp(150,0,100)=%d\n", clamp(150,0,100)); printf("isPrime(17)=%d\n", isPrime(17)); printf("5!=%d\n", factorial(5)); printf("rand(1~6)=%d\n", randRange(1,6)); printLine(30); return 0; }

연습 문제 + Level 6 마무리

1
#ifndef 가드 역할
쉬움
#ifndef MYUTIL_H / #define MYUTIL_H / ... / #endif 구조의 역할은?
정답 보기
같은 헤더를 여러 번 include해도 내용이 중복 처리되지 않도록 막는 "포함 가드"입니다. 처음 include 때만 실행됩니다.
2
clamp 함수 활용
쉬움
다음 결과를 예측하세요.
① clamp(50, 0, 100)
② clamp(-5, 0, 100)
③ clamp(150, 0, 100)
정답 보기
① 50 (범위 내) ② 0 (하한으로 고정) ③ 100 (상한으로 고정)

나만의 함수 추가

정답 보기
