다익스트라DPDAGBronze

[문제] N개의 정점과 M개의 간선(양의 가중치)이 있는 방향 그래프가 주어집니다.
시작 정점 S에서 도착 정점 E까지의 최단 거리와,
그 최단 거리로 갈 수 있는 서로 다른 경로의 수를 구하세요.
경로 수는 매우 클 수 있으므로 1,000,000,007로 나눈 나머지를 출력합니다.

핵심 아이디어: 다익스트라 알고리즘으로 최단 거리를 구하면서,
동시에 각 정점까지의 최단 경로 수를 DP로 세는 방법입니다.
dist[v] = 시작점에서 v까지의 최단 거리
cnt[v] = 시작점에서 v까지 최단 거리로 가는 경로 수
초기값: dist[S] = 0, cnt[S] = 1

상태 전이 규칙:
간선 (u, v, w)를 relaxation 할 때:
1) dist[u] + w < dist[v]이면: dist[v] = dist[u] + w, cnt[v] = cnt[u] (새로운 최단 경로 발견)
2) dist[u] + w == dist[v]이면: cnt[v] += cnt[u] (같은 거리의 경로 추가)
3) dist[u] + w > dist[v]이면: 무시

#include <stdio.h> #include <stdlib.h> #include <limits.h> #define MAXN 100001 #define MOD 1000000007 typedef struct { int to, w, next; } Edge; Edge edges[200002]; int head[MAXN], ecnt; void addEdge(int u, int v, int w) { edges[ecnt] = (Edge){v, w, head[u]}; head[u] = ecnt++; } long long dist[MAXN]; long long cnt[MAXN]; /* 간단한 우선순위 큐 (배열 기반 min-heap) */ typedef struct { int v; long long d; } Node; Node heap[400002]; int hsize; void push(Node x) { heap[++hsize] = x; int i = hsize; while (i > 1 && heap[i].d < heap[i/2].d) { Node t = heap[i]; heap[i] = heap[i/2]; heap[i/2] = t; i /= 2; } } Node pop() { Node ret = heap[1]; heap[1] = heap[hsize--]; int i = 1; while (2*i <= hsize) { int c = 2*i; if (c+1 <= hsize && heap[c+1].d < heap[c].d) c++; if (heap[i].d <= heap[c].d) break; Node t = heap[i]; heap[i] = heap[c]; heap[c] = t; i = c; } return ret; } int main() { int N, M, S, E; scanf("%d %d %d %d", &N, &M, &S, &E); for (int i = 0; i <= N; i++) head[i] = -1; for (int i = 0; i < M; i++) { int u, v, w; scanf("%d %d %d", &u, &v, &w); addEdge(u, v, w); } for (int i = 0; i <= N; i++) { dist[i] = LLONG_MAX; cnt[i] = 0; } dist[S] = 0; cnt[S] = 1; hsize = 0; push((Node){S, 0}); while (hsize > 0) { Node cur = pop(); if (cur.d > dist[cur.v]) continue; /* 이미 처리된 노드 */ for (int e = head[cur.v]; e != -1; e = edges[e].next) { int nv = edges[e].to; long long nd = cur.d + edges[e].w; if (nd < dist[nv]) { dist[nv] = nd; cnt[nv] = cnt[cur.v]; /* 새 최단 → 경로 수 덮어쓰기 */ push((Node){nv, nd}); } else if (nd == dist[nv]) { cnt[nv] = (cnt[nv] + cnt[cur.v]) % MOD; /* 동일 거리 → 경로 수 합산 */ } } } printf("최단 거리: %lld\n", dist[E]); printf("경로 수: %lld\n", cnt[E]); return 0; }

[문제] N개의 점이 수직선 위에 있습니다. 각 점의 좌표는 최대 109까지 가능합니다.
Q개의 쿼리가 주어집니다:
1) 1 x : 좌표 x에 점 하나를 추가합니다.
2) 2 l r : 좌표 [l, r] 범위에 있는 점의 개수를 출력합니다.
좌표 범위가 109이므로, 좌표 압축 후 세그먼트 트리를 사용합니다.

좌표 압축(Coordinate Compression):
큰 범위의 좌표를 작은 인덱스로 매핑하는 기법입니다.
1) 모든 쿼리를 미리 읽어서 등장하는 좌표를 수집합니다.
2) 좌표를 정렬하고 중복을 제거합니다.
3) 이진 탐색으로 원래 좌표 → 압축 인덱스 변환합니다.

Ccoord_compress_segtree.c
#include <stdio.h> #include <stdlib.h> #define MAXQ 200001 int tree[800004]; int coords[MAXQ], csz; int qtype[MAXQ], qa[MAXQ], qb[MAXQ]; int cmp(const void* a, const void* b) { return (*(int*)a - *(int*)b); } int compress(int val) { int lo = 0, hi = csz - 1; while (lo <= hi) { int mid = (lo + hi) / 2; if (coords[mid] == val) return mid; else if (coords[mid] < val) lo = mid + 1; else hi = mid - 1; } return lo; } void update(int node, int s, int e, int idx) { if (s == e) { tree[node]++; return; } int mid = (s + e) / 2; if (idx <= mid) update(2*node, s, mid, idx); else update(2*node+1, mid+1, e, idx); tree[node] = tree[2*node] + tree[2*node+1]; } int query(int node, int s, int e, int l, int r) { if (r < s || e < l) return 0; if (l <= s && e <= r) return tree[node]; int mid = (s + e) / 2; return query(2*node, s, mid, l, r) + query(2*node+1, mid+1, e, l, r); } int main() { int Q; scanf("%d", &Q); csz = 0; for (int i = 0; i < Q; i++) { scanf("%d", &qtype[i]); if (qtype[i] == 1) { scanf("%d", &qa[i]); coords[csz++] = qa[i]; } else { scanf("%d %d", &qa[i], &qb[i]); coords[csz++] = qa[i]; coords[csz++] = qb[i]; } } /* 좌표 압축: 정렬 후 중복 제거 */ qsort(coords, csz, sizeof(int), cmp); int unique = 0; for (int i = 0; i < csz; i++) if (i == 0 || coords[i] != coords[i-1]) coords[unique++] = coords[i]; csz = unique; for (int i = 0; i < Q; i++) { if (qtype[i] == 1) { int idx = compress(qa[i]); update(1, 0, csz-1, idx); } else { int li = compress(qa[i]); int ri = compress(qb[i]); printf("%d\n", query(1, 0, csz-1, li, ri)); } } return 0; }

[문제] N개의 나무가 일렬로 서 있고, 각 나무의 높이가 주어집니다.
절단기의 높이 H를 설정하면, H보다 높은 나무는 H 위의 부분이 잘립니다.
적어도 M 미터의 나무를 가져가야 할 때,
절단기의 높이 H의 최댓값을 구하세요.

매개변수 탐색(Parametric Search):
"최적의 H를 구하라" → "H를 정해두고, 조건을 만족하는지 검사"로 바꿉니다.
H에 대해 이분 탐색을 수행합니다:
- check(H): 높이 H로 자를 때 얻는 나무 총량 >= M인지 확인
- 조건 만족 → H를 더 높여본다 (lo = mid + 1)
- 조건 불만족 → H를 낮춘다 (hi = mid - 1)

#include <stdio.h> #define MAXN 1000001 long long trees[MAXN]; int N; long long M; /* check: 높이 H로 자를 때 M 이상 얻을 수 있는가? */ int check(long long H) { long long total = 0; for (int i = 0; i < N; i++) { if (trees[i] > H) total += trees[i] - H; } return total >= M; } int main() { scanf("%d %lld", &N, &M); long long maxH = 0; for (int i = 0; i < N; i++) { scanf("%lld", &trees[i]); if (trees[i] > maxH) maxH = trees[i]; } long long lo = 0, hi = maxH, ans = 0; while (lo <= hi) { long long mid = (lo + hi) / 2; if (check(mid)) { ans = mid; /* 조건 만족 → 더 높여본다 */ lo = mid + 1; } else { hi = mid - 1; /* 조건 불만족 → 낮춘다 */ } } printf("%lld\n", ans); return 0; }

트리 DPBFS트리 지름Bronze

[문제] N개의 정점으로 이루어진 가중치 트리가 주어집니다.
트리의 지름(Diameter)을 구하세요.
트리의 지름 = 가장 먼 두 정점 사이의 거리입니다.

방법 1 — 2번 BFS:
1) 임의의 정점에서 BFS → 가장 먼 정점 u를 찾는다
2) u에서 BFS → 가장 먼 정점 v를 찾는다
3) dist(u, v)가 트리의 지름이다
시간 복잡도: O(N)

방법 2 — 트리 DP:
각 정점 v에 대해 depth[v] = v를 루트로 하는 서브트리에서 가장 깊은 경로 길이
정점 v를 지나는 최장 경로 = v의 자식들 중 가장 긴 두 depth의 합
모든 정점에 대해 이 값의 최댓값이 트리의 지름입니다.

Ctree_diameter.c
#include <stdio.h> #include <string.h> #define MAXN 100001 typedef struct { int to, w, next; } Edge; Edge edges[200002]; int head[MAXN], ecnt; int dist[MAXN], visited[MAXN]; int queue[MAXN]; void addEdge(int u, int v, int w) { edges[ecnt] = (Edge){v, w, head[u]}; head[u] = ecnt++; edges[ecnt] = (Edge){u, w, head[v]}; head[v] = ecnt++; } /* BFS로 start에서 가장 먼 정점과 거리 반환 */ int bfs(int start, int N, int* farDist) { memset(visited, 0, sizeof(int) * (N + 1)); memset(dist, 0, sizeof(int) * (N + 1)); int front = 0, rear = 0; queue[rear++] = start; visited[start] = 1; int farthest = start; *farDist = 0; while (front < rear) { int u = queue[front++]; for (int e = head[u]; e != -1; e = edges[e].next) { int v = edges[e].to; if (!visited[v]) { visited[v] = 1; dist[v] = dist[u] + edges[e].w; queue[rear++] = v; if (dist[v] > *farDist) { *farDist = dist[v]; farthest = v; } } } } return farthest; } int main() { int N; scanf("%d", &N); memset(head, -1, sizeof(head)); ecnt = 0; for (int i = 0; i < N - 1; i++) { int u, v, w; scanf("%d %d %d", &u, &v, &w); addEdge(u, v, w); } int d1, d2; int u = bfs(1, N, &d1); /* 1단계: 임의 정점에서 가장 먼 정점 u */ int v = bfs(u, N, &d2); /* 2단계: u에서 가장 먼 정점 v */ printf("트리의 지름: %d\n", d2); return 0; }

SECTION 381Bronze | 종합 실전
문자열 + DP — 와일드카드 매칭
DP문자열와일드카드Bronze

[문제] 패턴 문자열 P와 텍스트 문자열 T가 주어집니다.
패턴에는 다음 특수 문자가 포함될 수 있습니다:
- ? : 정확히 하나의 임의 문자와 매칭
- * : 0개 이상의 임의 문자열과 매칭
P가 T 전체와 매칭되는지 판별하세요.

DP 정의:
dp[i][j] = P[0..i-1]이 T[0..j-1]과 매칭되는지 여부 (true/false)
초기 조건:
- dp[0][0] = true (빈 패턴 ↔ 빈 텍스트)
- dp[i][0]: P[0..i-1]이 모두 *인 경우만 true
전이:
- P[i-1] == T[j-1] 또는 P[i-1] == '?' → dp[i][j] = dp[i-1][j-1]
- P[i-1] == '*' → dp[i][j] = dp[i-1][j] || dp[i][j-1]
  (* 가 0개 매칭: dp[i-1][j], 1개 이상 매칭: dp[i][j-1])
- 그 외 → dp[i][j] = false

플로우 + DP — 최소 비용 최대 유량 (MCMF) 개념

최소 비용 최대 유량(MCMF, Min-Cost Max-Flow):
네트워크 플로우 문제에서 각 간선에 용량(capacity)과 비용(cost)이 있을 때,
최대 유량을 보내면서 총 비용을 최소화하는 문제입니다.

알고리즘 개요:
1) 소스 S에서 싱크 T까지 최소 비용 경로를 찾는다 (벨만-포드 또는 SPFA)
2) 그 경로를 따라 가능한 만큼 유량을 보낸다
3) 역간선의 비용은 -cost로 설정한다
4) 더 이상 증가 경로가 없을 때까지 반복한다

벨만-포드를 사용하는 이유:
역간선의 비용이 음수이므로 다익스트라를 직접 쓸 수 없습니다.
벨만-포드는 음의 가중치를 처리할 수 있습니다.
(SPFA는 벨만-포드의 최적화 버전으로, 실전에서 주로 사용합니다)

Cmcmf_concept.c
#include <stdio.h> #include <string.h> #include <limits.h> #define MAXV 201 #define MAXE 10001 #define INF 1000000000 typedef struct { int to, cap, cost, next; } Edge; Edge edges[MAXE]; int head[MAXV], ecnt; int dist[MAXV], inQueue[MAXV], prevv[MAXV], preve[MAXV]; int que[MAXV * 100]; void initGraph() { memset(head, -1, sizeof(head)); ecnt = 0; } void addEdge(int u, int v, int cap, int cost) { edges[ecnt] = (Edge){v, cap, cost, head[u]}; head[u] = ecnt++; edges[ecnt] = (Edge){u, 0, -cost, head[v]}; head[v] = ecnt++; /* 정방향: cap, +cost / 역방향: 0, -cost */ } /* SPFA로 최소 비용 경로 탐색 */ int spfa(int S, int T, int V) { for (int i = 0; i < V; i++) dist[i] = INF; memset(inQueue, 0, sizeof(inQueue)); dist[S] = 0; inQueue[S] = 1; int front = 0, rear = 0; que[rear++] = S; while (front < rear) { int u = que[front++]; inQueue[u] = 0; for (int e = head[u]; e != -1; e = edges[e].next) { int v = edges[e].to; if (edges[e].cap > 0 && dist[u] + edges[e].cost < dist[v]) { dist[v] = dist[u] + edges[e].cost; prevv[v] = u; preve[v] = e; if (!inQueue[v]) { inQueue[v] = 1; que[rear++] = v; } } } } return dist[T] < INF; } /* MCMF 메인 */ void mcmf(int S, int T, int V, int* totalFlow, int* totalCost) { *totalFlow = 0; *totalCost = 0; while (spfa(S, T, V)) { /* 경로의 최소 잔여 용량 구하기 */ int minCap = INF; for (int v = T; v != S; v = prevv[v]) if (edges[preve[v]].cap < minCap) minCap = edges[preve[v]].cap; /* 경로를 따라 유량 흘리기 */ for (int v = T; v != S; v = prevv[v]) { edges[preve[v]].cap -= minCap; edges[preve[v] ^ 1].cap += minCap; /* 역간선 */ } *totalFlow += minCap; *totalCost += minCap * dist[T]; } } int main() { initGraph(); /* 예시: S=0, T=3, 정점 4개 */ addEdge(0, 1, 3, 1); /* 용량 3, 비용 1 */ addEdge(0, 2, 2, 5); /* 용량 2, 비용 5 */ addEdge(1, 2, 1, 2); /* 용량 1, 비용 2 */ addEdge(1, 3, 2, 3); /* 용량 2, 비용 3 */ addEdge(2, 3, 3, 1); /* 용량 3, 비용 1 */ int flow, cost; mcmf(0, 3, 4, &flow, &cost); printf("최대 유량: %d\n최소 비용: %d\n", flow, cost); return 0; }

[문제] N개의 목장이 일렬로 있고, 각 목장의 비용이 주어집니다.

전체 비용의 합을 최소화하세요.

DP 정의:
dp[i][k] = 처음 i개의 목장을 k개의 그룹으로 나눌 때의 최소 비용
전이:
dp[i][k] = min(dp[j][k-1] + cost(j+1, i)) (j = k-1 ~ i-1)
여기서 cost(a, b) = 구간 [a, b]의 max - min
초기: dp[i][1] = cost(1, i)

#include <stdio.h> #include <limits.h> #define MAXN 501 #define MAXK 501 int a[MAXN]; int dp[MAXN][MAXK]; int cost(int l, int r) { int mn = INT_MAX, mx = INT_MIN; for (int i = l; i <= r; i++) { if (a[i] < mn) mn = a[i]; if (a[i] > mx) mx = a[i]; } return mx - mn; } int main() { int N, K; scanf("%d %d", &N, &K); for (int i = 1; i <= N; i++) scanf("%d", &a[i]); /* 초기화: INF */ for (int i = 0; i <= N; i++) for (int k = 0; k <= K; k++) dp[i][k] = INT_MAX / 2; dp[0][0] = 0; for (int k = 1; k <= K; k++) { for (int i = k; i <= N; i++) { for (int j = k - 1; j < i; j++) { int c = dp[j][k-1] + cost(j+1, i); if (c < dp[i][k]) dp[i][k] = c; } } } printf("%d\n", dp[N][K]); return 0; }

#include <stdio.h> #include <stdlib.h> #define MAXN 100001 typedef struct { int s, e, w; } Interval; Interval intervals[MAXN]; long long dp[MAXN]; int cmp(const void* a, const void* b) { return ((Interval*)a)->e - ((Interval*)b)->e; } /* 이분 탐색: intervals[idx].s 이전에 끝나는 마지막 구간 */ int findLast(int idx) { int lo = 0, hi = idx - 1, ans = -1; while (lo <= hi) { int mid = (lo + hi) / 2; if (intervals[mid].e <= intervals[idx].s) { ans = mid; lo = mid + 1; } else { hi = mid - 1; } } return ans; } int main() { int N; scanf("%d", &N); for (int i = 0; i < N; i++) scanf("%d %d %d", &intervals[i].s, &intervals[i].e, &intervals[i].w); qsort(intervals, N, sizeof(Interval), cmp); /* 가중치 DP */ dp[0] = intervals[0].w; for (int i = 1; i < N; i++) { long long include = intervals[i].w; int prev = findLast(i); if (prev != -1) include += dp[prev]; dp[i] = (include > dp[i-1]) ? include : dp[i-1]; } printf("%lld\n", dp[N-1]); return 0; }

[문제] 문자열 A를 문자열 B로 변환하는 최소 연산 횟수를 구하세요.
허용 연산:
- 삽입: 한 위치에 문자 하나를 삽입
- 삭제: 문자 하나를 삭제
- 교체: 문자 하나를 다른 문자로 교체
이것이 유명한 편집 거리(Edit Distance, Levenshtein Distance) 문제입니다.

DP 정의:
dp[i][j] = A[0..i-1]을 B[0..j-1]로 변환하는 최소 연산 수
초기: dp[0][j] = j (삽입만), dp[i][0] = i (삭제만)
전이:
- A[i-1] == B[j-1] → dp[i][j] = dp[i-1][j-1]
- 그 외 → dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
  (삭제, 삽입, 교체 중 최소)
