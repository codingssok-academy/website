LIS(Longest Increasing Subsequence)와 세그먼트 트리
LIS는 가장 긴 증가하는 부분 수열의 길이를 구하는 고전적인 문제입니다.
기본 DP는 O(N²)이지만, 세그먼트 트리를 활용하면 O(N log N)으로 풀 수 있습니다.
핵심 아이디어: dp[i] = a[i]를 마지막 원소로 하는 LIS 길이라 하면,
dp[i] = max(dp[j] | a[j] < a[i]) + 1입니다.
이 구간 최댓값 쿼리를 세그먼트 트리로 O(log N)에 처리합니다.

좌표 압축이 필요한 이유
값의 범위가 10&sup9;처럼 클 때, 세그먼트 트리 크기를 줄이기 위해 좌표 압축합니다.
원소를 정렬하여 순위(rank)를 매긴 뒤, rank를 인덱스로 사용합니다.
N개 원소이므로 트리 크기는 O(N)이 됩니다.

알고리즘 흐름
1단계: 배열의 모든 값을 정렬하여 좌표 압축 (중복 제거)
2단계: 세그먼트 트리 초기화 (구간 최댓값 트리, 초기값 0)
3단계: 왼쪽에서 오른쪽으로 순회하며:
  - rank = compress(a[i])를 구한다
  - query(1, rank-1)로 a[i]보다 작은 값들 중 최대 dp를 구한다
  - dp[i] = query 결과 + 1
  - update(rank, dp[i])로 세그트리에 반영
4단계: 전체 dp의 최댓값이 LIS 길이

#include <stdio.h> #include <stdlib.h> #define MAXN 200005 int a[MAXN], sorted[MAXN], n; int tree[4 * MAXN]; // 구간 최댓값 세그먼트 트리 int cmp(const void *x, const void *y) { return *(int*)x - *(int*)y; } /* 좌표 압축: lower_bound 이진 탐색 */ int compress(int val, int sz) { int lo = 0, hi = sz - 1, mid; while (lo < hi) { mid = (lo + hi) / 2; if (sorted[mid] < val) lo = mid + 1; else hi = mid; } return lo + 1; // 1-indexed } /* 구간 최댓값 쿼리 */ int query(int node, int s, int e, int l, int r) { if (r < s || e < l) return 0; if (l <= s && e <= r) return tree[node]; int mid = (s + e) / 2; int lv = query(node*2, s, mid, l, r); int rv = query(node*2+1, mid+1, e, l, r); return lv > rv ? lv : rv; } /* 포인트 업데이트 */ void update(int node, int s, int e, int idx, int val) { if (s == e) { if (val > tree[node]) tree[node] = val; return; } int mid = (s + e) / 2; if (idx <= mid) update(node*2, s, mid, idx, val); else update(node*2+1, mid+1, e, idx, val); tree[node] = tree[node*2] > tree[node*2+1] ? tree[node*2] : tree[node*2+1]; } int unique(int arr[], int sz) { int cnt = 0; for (int i = 0; i < sz; i++) if (i == 0 || arr[i] != arr[i-1]) arr[cnt++] = arr[i]; return cnt; } int main(void) { scanf("%d", &n); for (int i = 0; i < n; i++) { scanf("%d", &a[i]); sorted[i] = a[i]; } qsort(sorted, n, sizeof(int), cmp); int sz = unique(sorted, n); int ans = 0; for (int i = 0; i < n; i++) { int rank = compress(a[i], sz); int best = (rank > 1) ? query(1, 1, sz, 1, rank - 1) : 0; int dp_i = best + 1; update(1, 1, sz, rank, dp_i); if (dp_i > ans) ans = dp_i; } printf("%d\n", ans); return 0; }

DAG 최장 경로 문제
DAG(Directed Acyclic Graph)에서는 위상 정렬 순서대로 DP를 수행하여 최장 경로를 구할 수 있습니다.
일반 그래프에서 최장 경로는 NP-Hard이지만, DAG에서는 O(V+E)입니다.
dp[v] = 정점 v에서 끝나는 최장 경로의 길이 (간선 수 기준)입니다.

경로 복원
prev[v] 배열을 두어, dp[v]가 갱신될 때 어떤 정점에서 왔는지 기록합니다.
최장 경로의 끝점에서 prev를 따라가면 실제 경로를 역추적할 수 있습니다.
역추적한 결과를 뒤집으면 시작점부터 끝점까지의 경로가 됩니다.

알고리즘 단계
1단계: 인접 리스트와 진입 차수(in-degree) 구성
2단계: 진입 차수가 0인 정점을 큐에 넣고 위상 정렬
3단계: 위상 정렬 순서대로 dp 갱신:
  각 정점 u의 인접 정점 v에 대해, dp[v] = max(dp[v], dp[u] + 1)
  갱신 시 prev[v] = u 기록
4단계: dp 최댓값을 가진 정점에서 prev를 역추적하여 경로 복원

" -> " : "\n"); return 0; }

Suffix Array란?
문자열 S의 모든 접미사(suffix)를 사전순으로 정렬한 배열입니다.
예: S = "banana"의 접미사들: banana, anana, nana, ana, na, a
사전순: a, ana, anana, banana, na, nana → SA = [5, 3, 1, 0, 4, 2]
SA를 O(N log² N) 또는 O(N log N)에 구축할 수 있습니다.

LCP 배열
LCP[i] = SA 상에서 인접한 두 접미사 SA[i-1]와 SA[i]의 최장 공통 접두사 길이입니다.
Kasai 알고리즘으로 O(N)에 구축 가능합니다.
LCP 배열을 이용하면 서로 다른 부분 문자열 수를 구할 수 있습니다.

서로 다른 부분 문자열 수
전체 부분 문자열 수: N*(N+1)/2 (길이 N인 문자열)
중복되는 것: LCP[i]의 합
따라서 서로 다른 부분 문자열 수 = N*(N+1)/2 - sum(LCP[i])

구간 쿼리를 위한 세그먼트 트리
구간 [l, r]에 대한 서로 다른 부분 문자열 수를 여러 번 묻는 경우,
오프라인 쿼리 + 세그먼트 트리를 조합하여 효율적으로 해결할 수 있습니다.
이는 SA + LCP + 구간 min 쿼리(RMQ)의 조합입니다.

#include <stdio.h> #include <string.h> #include <stdlib.h> #define MAXN 100005 char s[MAXN]; int sa[MAXN], rk[MAXN], tmp[MAXN], lcp[MAXN]; int n; /* O(N log^2 N) SA 구축 */ int sa_k; int sa_cmp(const void *a, const void *b) { int i = *(int*)a, j = *(int*)b; if (rk[i] != rk[j]) return rk[i] - rk[j]; int ri = (i + sa_k < n) ? rk[i + sa_k] : -1; int rj = (j + sa_k < n) ? rk[j + sa_k] : -1; return ri - rj; } void build_sa(void) { for (int i = 0; i < n; i++) { sa[i] = i; rk[i] = s[i]; } for (sa_k = 1; sa_k < n; sa_k *= 2) { qsort(sa, n, sizeof(int), sa_cmp); tmp[sa[0]] = 0; for (int i = 1; i < n; i++) tmp[sa[i]] = tmp[sa[i-1]] + (sa_cmp(&sa[i-1], &sa[i]) ? 1 : 0); memcpy(rk, tmp, n * sizeof(int)); } } /* Kasai 알고리즘: O(N) LCP 구축 */ void build_lcp(void) { int inv[MAXN]; for (int i = 0; i < n; i++) inv[sa[i]] = i; int k = 0; for (int i = 0; i < n; i++) { if (inv[i] == 0) { k = 0; continue; } int j = sa[inv[i] - 1]; while (i + k < n && j + k < n && s[i+k] == s[j+k]) k++; lcp[inv[i]] = k; if (k) k--; } } int main(void) { scanf("%s", s); n = strlen(s); build_sa(); build_lcp(); long long total = (long long)n * (n + 1) / 2; for (int i = 1; i < n; i++) total -= lcp[i]; printf("서로 다른 부분 문자열 수: %lld\n", total); printf("SA: "); for (int i = 0; i < n; i++) printf("%d ", sa[i]); printf("\nLCP: "); for (int i = 0; i < n; i++) printf("%d ", lcp[i]); printf("\n"); return 0; }

N개의 직사각형이 주어질 때, 합집합의 넓이를 구하는 문제입니다.

알고리즘 개요
1단계: 각 직사각형의 왼쪽 변과 오른쪽 변을 이벤트로 만든다
  - 왼쪽 변: (x1, y1, y2, +1) — 커버 시작
  - 오른쪽 변: (x2, y1, y2, -1) — 커버 종료
2단계: 이벤트를 x좌표로 정렬
3단계: y좌표를 좌표 압축
4단계: x좌표 순으로 스위핑하며:
  - 현재 이벤트의 y 구간에 대해 세그트리에 +1 또는 -1
  - 세그트리에서 "커버된 y 길이" 쿼리
  - 넓이 += 커버된 y 길이 * (다음 x - 현재 x)

세그먼트 트리의 역할
각 노드가 y 구간을 나타냅니다.
cnt[node]: 해당 구간이 완전히 커버된 횟수
len[node]: 해당 구간에서 1번 이상 커버된 실제 길이
cnt > 0이면 len = 구간 전체 길이,

카탈란 수 (Catalan Number)
C(n) = C(2n, n) / (n+1) = (2n)! / ((n+1)! * n!)
재귀식: C(n) = sum(C(i) * C(n-1-i), i=0..n-1), C(0) = 1
처음 몇 개: 1, 1, 2, 5, 14, 42, 132, 429, 1430, ...

카탈란 수가 등장하는 문제들
1. 올바른 괄호열의 수: n쌍의 괄호로 만들 수 있는 올바른 괄호열의 수 = C(n)
2. 이진 트리의 수: n개의 내부 노드를 가진 이진 트리의 구조 수 = C(n)
3. 격자 경로 수: (0,0)에서 (n,n)까지 대각선 위로 넘지 않는 경로 수 = C(n)
4. 볼록 다각형 삼각분할: (n+2)-각형의 삼각분할 수 = C(n)
5. 스택 가능 순열: n개 원소의 스택 정렬 가능 순열 수 = C(n)

DP로 카탈란 수 계산하기
dp[0] = 1
dp[i] = sum(dp[j] * dp[i-1-j], j=0..i-1)
시간 복잡도: O(N²)

BIT(Binary Indexed Tree) 하나의 한계
기본 BIT는 포인트 업데이트 + 구간 합 쿼리를 O(log N)에 처리합니다.
구간 업데이트 + 포인트 쿼리도 차분 배열 + BIT로 가능합니다.
하지만 구간 업데이트 + 구간 합 쿼리는 BIT 하나로는 안 됩니다.

BIT 2개를 이용한 공식
[1, x] 구간 합 = (x+1) * sum(bit1, x) - sum(bit2, x)
[l, r]에 v를 더하는 업데이트:
  bit1에: add(l, v), add(r+1, -v)
  bit2에: add(l, v*l), add(r+1, -v*(r+1))
구간 합 [l, r] = prefix(r) - prefix(l-1)

유도 과정
[l, r]에 v를 더한 후 prefix(x)를 구하면:
  x < l: 변화 없음
  l ≤ x ≤ r: v*(x-l+1) = v*(x+1) - v*l
  x > r: v*(r-l+1) = v*(r+1) - v*l
이를 bit1, bit2로 분리하면 위 공식이 됩니다.

마르코프 체인 (Markov Chain)
상태 집합과 전이 확률로 정의되는 확률 과정입니다.
"현재 상태가 다음 상태를 결정" — 과거 이력과 무관 (마르코프 성질).
전이 행렬 T[i][j] = 상태 i에서 상태 j로 갈 확률.

N 스텝 후 확률 분포
초기 확률 벡터를 p0라 하면,
N스텝 후 확률 벡터 = p0 * T^N
T^N을 행렬 거듭제곱으로 O(S³ log N)에 계산합니다 (S = 상태 수).

행렬 거듭제곱
정수 거듭제곱과 같은 원리: T^N = T^(N/2) * T^(N/2)
N이 홀수면 T^N = T^(N-1) * T

볼록 다각형 삼각분할
볼록 N-각형을 (N-2)개의 삼각형으로 분할합니다.
각 삼각형의 비용 = 세 꼭짓점의 좌표로 정의되는 함수 (예: 둘레 길이).
총 비용을 최소화하는 삼각분할을 구합니다.

구간 DP
dp[i][j] = 꼭짓점 i부터 j까지의 다각형을 삼각분할하는 최소 비용
전이: dp[i][j] = min(dp[i][k] + dp[k][j] + cost(i, k, j)) for k in (i+1 .. j-1)
기저: j - i < 2이면 dp[i][j] = 0 (삼각형을 이룰 수 없음)
답: dp[0][N-1]

비용 함수 예: 삼각형 둘레
cost(i, k, j) = dist(P[i], P[k]) + dist(P[k], P[j]) + dist(P[i], P[j])
여기서 dist는 두 점 사이의 유클리드 거리입니다.

최소 경로 커버 (Minimum Path Cover)
DAG에서 모든 정점을 커버하는 최소 개수의 경로를 구하는 문제입니다.
정리: 최소 경로 커버 수 = N - 최대 이분 매칭
각 정점 v를 v_out(왼쪽)과 v_in(오른쪽)으로 분할하여 이분 그래프를 만듭니다.
간선 (u, v)가 있으면 u_out → v_in 간선을 추가합니다.

이분 매칭 (Hungarian / Hopcroft-Karp)
간단한 구현은 증가 경로(augmenting path) 기반입니다.
각 왼쪽 정점에서 DFS로 증가 경로를 찾습니다.
시간 복잡도: O(V * E). Hopcroft-Karp는 O(E * sqrt(V)).

왜 N - 최대 매칭인가?
매칭이 없으면: 각 정점이 별도 경로 → N개 경로
매칭 하나 추가: 두 정점을 하나의 경로로 합침 → 경로 1개 감소
따라서 최대 매칭 M이면 경로 수 = N - M

Heavy-Light Decomposition (HLD)
트리의 간선을 heavy edge와 light edge로 분류합니다.
heavy edge: 자식 서브트리 크기가 가장 큰 간선
light edge: 나머지
heavy edge들이 연결된 경로를 "heavy chain"이라 합니다.
임의의 정점에서 루트까지 최대 O(log N)개의 heavy chain을 거칩니다.

HLD + 세그먼트 트리
각 heavy chain을 세그먼트 트리의 연속 구간으로 매핑합니다.
DFS 순서로 번호를 매기되, heavy child를 먼저 방문합니다.
그러면 같은 chain의 정점들이 연속 번호를 받습니다.
경로 쿼리 = O(log N)개 chain의 세그트리 쿼리 = O(log² N)

Lazy Propagation과 결합
트리 경로에 값을 더하고, 경로 합을 구하려면:
  1. 경로를 O(log N)개 chain 구간으로 분해
  2. 각 구간에 세그트리 range update (Lazy)
  3. 쿼리도 마찬가지로 range query

(최적 분할점 단조성)		O(N log N)분할 정복으로 탐색 범위 축소
Aliens 트릭
(WQS Binary Search)		dp[K][N]에서 K를
라그랑주 승수로 제거		O(N log N * log C)K 제약을 비용으로 변환

CHT 판별 기준
전이식을 변형하여 dp[i] = min/max(m[j] * x[i] + b[j]) 형태가 되는지 확인합니다.
m[j]와 b[j]가 j에만 의존하고, x[i]가 i에만 의존하면 CHT 적용 가능.
m[j]가 단조이면 스택으로 O(N), 아니면 Li Chao Tree로 O(N log N).

분할 정복 최적화 판별 기준
dp[i] = min(dp[j] + C(j+1, i))에서
C가 사각 부등식(quadrangle inequality)을 만족하면
최적 분할점 opt[i]가 단조 증가합니다.
C(a,c) + C(b,d) ≤ C(a,d) + C(b,c) (a ≤ b ≤ c ≤ d)

Aliens 트릭 판별 기준
dp[K][N] (정확히 K개 선택)에서 K를 바꿔가며 풀기 어려울 때,
"하나 선택할 때마다 비용 lambda를 지불"하는 문제로 변환합니다.
lambda를 이진 탐색하여 최적 K를 찾습니다.
조건: dp[K][N]이 K에 대해 볼록(convex) 함수여야 합니다.

Ccht_example.c
#include <stdio.h> #define MAXN 500005 /* Li Chao Tree (간소화) */ typedef struct { long long m, b; } Line; Line seg[MAXN * 4]; int has[MAXN * 4]; long long eval(Line l, long long x) { return l.m * x + l.b; } void add_line(int nd, int s, int e, Line new_l) { int mid = (s + e) / 2; if (!has[nd]) { seg[nd] = new_l; has[nd] = 1; return; } int left_better = eval(new_l, s) < eval(seg[nd], s); int mid_better = eval(new_l, mid) < eval(seg[nd], mid); if (mid_better) { Line tmp = seg[nd]; seg[nd] = new_l; new_l = tmp; } if (s == e) return; if (left_better != mid_better) add_line(nd*2, s, mid, new_l); else add_line(nd*2+1, mid+1, e, new_l); } long long query_min(int nd, int s, int e, int x) { if (!has[nd]) return 1e18; long long res = eval(seg[nd], x); if (s == e) return res; int mid = (s + e) / 2; long long sub; if (x <= mid) sub = query_min(nd*2, s, mid, x); else sub = query_min(nd*2+1, mid+1, e, x); return res < sub ? res : sub; } int main(void) { /* 예시: 3개 직선에서 최솟값 쿼리 */ add_line(1, 0, 100, (Line){2, 1}); // y = 2x + 1 add_line(1, 0, 100, (Line){-1, 20}); // y = -x + 20 add_line(1, 0, 100, (Line){0, 10}); // y = 10 for (int x = 0; x <= 20; x += 5) printf("x=%d: min=%lld\n", x, query_min(1, 0, 100, x)); return 0; }
