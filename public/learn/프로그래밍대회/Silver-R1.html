[문제] N개의 도시와 도시 간 이동 비용이 주어집니다.
모든 도시를 정확히 한 번씩 방문하고 출발 도시로 돌아오는 최소 비용 경로를 구하세요.
N ≤ 20, 비용은 양의 정수입니다.
불가능한 경로(비용 0)가 있을 수 있습니다.

핵심 아이디어: 외판원 순회(Traveling Salesman Problem)는 NP-hard 문제이지만,
비트마스크 DP를 사용하면 O(2N · N2) 시간에 풀 수 있습니다.
이는 N ≤ 20일 때 충분히 실행 가능한 시간 복잡도입니다.

상태 정의:
dp[mask][i] = 방문한 도시 집합이 mask이고, 현재 도시 i에 있을 때의 최소 비용
mask는 N비트 이진수로, j번째 비트가 1이면 j번 도시를 방문했다는 뜻입니다.
초기값: dp[1][0] = 0 (0번 도시에서 출발, 0번만 방문한 상태)
전이: mask에서 i에 있을 때, 아직 방문하지 않은 도시 j로 이동
dp[mask | (1 << j)][j] = min(dp[mask | (1 << j)][j], dp[mask][i] + cost[i][j])
정답: min(dp[(1 << N) - 1][i] + cost[i][0]) for all i

Ctsp_bitmask_dp.c
#include <stdio.h> #include <string.h> #define MAXN 20 #define INF 1000000000 int N; int cost[MAXN][MAXN]; int dp[1 << MAXN][MAXN]; /* dp[mask][i] */ int main(void) { scanf("%d", &N); for (int i = 0; i < N; i++) for (int j = 0; j < N; j++) scanf("%d", &cost[i][j]); /* 초기화: 모든 값을 INF로 */ memset(dp, 0x3f, sizeof(dp)); dp[1][0] = 0; /* 0번 도시 출발, mask=1(0번만 방문) */ int full = (1 << N) - 1; for (int mask = 1; mask <= full; mask++) { for (int i = 0; i < N; i++) { if (dp[mask][i] >= INF) continue; if (!(mask & (1 << i))) continue; for (int j = 0; j < N; j++) { if (mask & (1 << j)) continue; /* 이미 방문 */ if (cost[i][j] == 0) continue; /* 경로 없음 */ int nmask = mask | (1 << j); if (dp[mask][i] + cost[i][j] < dp[nmask][j]) dp[nmask][j] = dp[mask][i] + cost[i][j]; } } } /* 모든 도시 방문 후 0번으로 복귀 */ int ans = INF; for (int i = 1; i < N; i++) { if (dp[full][i] < INF && cost[i][0] > 0) if (dp[full][i] + cost[i][0] < ans) ans = dp[full][i] + cost[i][0]; } printf("%d\n", ans >= INF ? -1 : ans); return 0; }

프로필 DP (Broken Profile) — 격자 타일링 비트마스크
프로필 DP비트마스크격자 타일링Silver

[문제] N × M 격자를 1 × 2 또는 2 × 1 도미노로 완전히 채우는 방법의 수를 구하세요.
N ≤ 20, M ≤ 20 (단 min(N, M) ≤ 20).
격자에 빈 칸이 없어야 합니다.

핵심 아이디어: 프로필 DP(Broken Profile DP)는 격자를 열 단위(또는 행 단위)로 스캔하면서,
현재 열의 "프로필"(어떤 칸이 이미 채워져 있는지)을 비트마스크로 관리합니다.
열 j에서 열 j+1로 넘어갈 때, 현재 프로필에서 가능한 모든 타일 배치를 시도합니다.

상태 정의:
dp[j][mask] = j번째 열까지 채운 상태에서, j번째 열의 프로필이 mask일 때 방법 수
mask의 i번째 비트가 1이면: i행 j열이 오른쪽으로 돌출(j+1열까지 차지)
mask의 i번째 비트가 0이면: i행 j열이 완전히 채워짐(돌출 없음)

전이 과정 (열 j → 열 j+1):
현재 프로필 mask에서, 위에서 아래로 각 행을 순회하며:
1) 비트가 1이면(돌출): 다음 열에서 해당 행은 이미 채워짐 → 다음 프로필의 해당 비트 = 0
2) 비트가 0이면(빈 칸): 두 가지 선택
   a) 가로 타일 배치: 다음 프로필의 해당 비트 = 1 (오른쪽으로 돌출)
   b) 세로 타일 배치: 아래 행도 비트 0이어야 하며, 두 행을 함께 채움

Cprofile_dp_tiling.c
#include <stdio.h> #include <string.h> #define MAXM 21 int N, M; long long dp[MAXM][1 << 20]; /* row: 현재 처리 중인 행, cur_mask: 현재 열 프로필, nxt_mask: 다음 열 프로필 */ void solve(int col, int row, int cur_mask, int nxt_mask) { if (row == N) { dp[col + 1][nxt_mask] += dp[col][cur_mask]; return; } if (cur_mask & (1 << row)) { /* 이 칸은 이전 열에서 돌출됨 → 건너뜀 */ solve(col, row + 1, cur_mask, nxt_mask); } else { /* 선택 1: 가로 타일 → 다음 열로 돌출 */ solve(col, row + 1, cur_mask, nxt_mask | (1 << row)); /* 선택 2: 세로 타일 → 아래 행과 함께 채움 */ if (row + 1 < N && !(cur_mask & (1 << (row + 1)))) solve(col, row + 2, cur_mask, nxt_mask); } } int main(void) { scanf("%d %d", &N, &M); if (N > M) { int t = N; N = M; M = t; } /* N을 작은 쪽으로 */ memset(dp, 0, sizeof(dp)); dp[0][0] = 1; for (int col = 0; col < M; col++) { for (int mask = 0; mask < (1 << N); mask++) { if (dp[col][mask] == 0) continue; solve(col, 0, mask, 0); } } printf("%lld\n", dp[M][0]); return 0; }

[문제] 길이 2N인 배열 A가 주어집니다.
각 mask에 대해, mask의 모든 부분마스크 sub에 대한 A[sub]의 합을 구하세요.
즉, F[mask] = Σ A[sub] (sub ⊆ mask)
N ≤ 20.

무식한 방법: 각 mask에 대해 모든 부분마스크를 열거하면 O(3N)입니다.
for (sub = mask; sub > 0; sub = (sub - 1) & mask) 트릭으로 열거 가능하지만,
N=20이면 320 ≈ 35억으로 시간 초과입니다.

SOS DP 핵심:
비트 위치를 하나씩 처리합니다. i번째 비트를 "고려 완료"했을 때:
dp[mask] = mask에서 0~i번째 비트만 자유롭게 0으로 바꾼 부분마스크들의 A 합
전이: i번째 비트가 1이면 dp[mask] += dp[mask ^ (1 << i)]
i번째 비트가 0이면 변화 없음
최종적으로 모든 비트를 처리하면 dp[mask] = F[mask]가 됩니다.

Csos_dp.c
#include <stdio.h> #define MAXN 20 int N; long long A[1 << MAXN]; long long dp[1 << MAXN]; int main(void) { scanf("%d", &N); int sz = 1 << N; for (int i = 0; i < sz; i++) { scanf("%lld", &A[i]); dp[i] = A[i]; } /* SOS DP: 비트 i를 아래에서 위로 순회 */ for (int i = 0; i < N; i++) { for (int mask = 0; mask < sz; mask++) { if (mask & (1 << i)) dp[mask] += dp[mask ^ (1 << i)]; } } /* 출력: 각 mask에 대한 F[mask] */ for (int i = 0; i < sz; i++) printf("%lld ", dp[i]); printf("\n"); return 0; }

Knuth 최적화구간 DP단조성Silver

[문제] N개의 파일을 하나로 합치는 최소 비용을 구하세요.
인접한 두 파일을 합칠 때 비용은 두 파일 크기의 합입니다.
모든 파일을 하나로 합치는 최소 총 비용을 구하세요.
N ≤ 5000.

기본 구간 DP:
dp[i][j] = i번부터 j번 파일까지 합치는 최소 비용
dp[i][j] = min(dp[i][k] + dp[k+1][j]) + sum(i, j) for i ≤ k < j
이 풀이는 O(N3)이며, N=5000이면 약 1250억 연산으로 시간 초과입니다.

Knuth 최적화 조건:
최적 분할점 opt[i][j]가 단조성을 만족하면 적용 가능합니다:
opt[i][j-1] ≤ opt[i][j] ≤ opt[i+1][j]
이 조건이 성립하면, k의 탐색 범위를 opt[i][j-1]부터 opt[i+1][j]까지로 줄일 수 있어
전체 시간 복잡도가 O(N2)로 감소합니다.

단조성이 성립하는 충분조건 (사각 부등식):
비용 함수 C(i,j)가 다음을 만족하면 됩니다:
C(a,c) + C(b,d) ≤ C(a,d) + C(b,c) (a ≤ b ≤ c ≤ d)
파일 합치기 문제에서 C(i,j) = sum(i,j)이므로 이 조건이 성립합니다.

Cknuth_optimization.c
#include <stdio.h> #include <string.h> #define MAXN 5001 #define INF 0x3f3f3f3f3f3f3f3fLL int N; int a[MAXN]; long long psum[MAXN]; /* 누적합 */ long long dp[MAXN][MAXN]; int opt[MAXN][MAXN]; /* 최적 분할점 */ long long sum(int i, int j) { return psum[j] - psum[i - 1]; } int main(void) { scanf("%d", &N); for (int i = 1; i <= N; i++) { scanf("%d", &a[i]); psum[i] = psum[i - 1] + a[i]; } /* 초기화: 길이 1은 비용 0, opt[i][i] = i */ for (int i = 1; i <= N; i++) { dp[i][i] = 0; opt[i][i] = i; } /* 구간 길이 len = 2부터 N까지 */ for (int len = 2; len <= N; len++) { for (int i = 1; i + len - 1 <= N; i++) { int j = i + len - 1; dp[i][j] = INF; int lo = opt[i][j - 1]; int hi = opt[i + 1][j]; for (int k = lo; k <= hi && k < j; k++) { long long val = dp[i][k] + dp[k + 1][j] + sum(i, j); if (val < dp[i][j]) { dp[i][j] = val; opt[i][j] = k; } } } } printf("%lld\n", dp[1][N]); return 0; }

볼록 껍질 트릭 (CHT) — dp[j] = min(dp[i] + b[i]·a[j])
CHTConvex Hull TrickDP 최적화Silver

[문제] N개의 작업이 순서대로 주어집니다. 작업 i의 비용 계수는 a[i]와 b[i]입니다.
작업들을 연속 구간으로 나누어 처리할 때, 구간 [l, r]의 비용은 b[l] · a[r]입니다.
총 비용을 최소화하세요.
N ≤ 100,000.

DP 점화식:
dp[j] = min(dp[i] + b[i] · a[j]) for 0 ≤ i < j
이를 정리하면: dp[j] = min(b[i] · a[j] + dp[i])
이는 직선 y = b[i] · x + dp[i]에서 x = a[j]일 때의 최솟값을 구하는 문제입니다.

볼록 껍질 트릭 원리:
각 i에 대해 직선 y = b[i] · x + dp[i]를 관리합니다.
새 직선을 추가할 때, 불필요한 직선(다른 두 직선보다 항상 위에 있는 직선)을 제거합니다.
쿼리 시 이분 탐색 또는 포인터로 최솟값을 찾습니다.
기울기가 단조 감소이고 쿼리가 단조 증가이면 O(N), 아니면 O(N log N).

#include <stdio.h> #define MAXN 100001 typedef struct { long long m, b; } Line; /* y = m*x + b */ Line hull[MAXN]; int sz = 0; /* 마지막 직선이 불필요한지 판별 */ int bad(Line l1, Line l2, Line l3) { /* l2가 l1과 l3의 교점 사이에서 항상 위에 있는지 */ return (double)(l3.b - l1.b) * (l1.m - l2.m) <= (double)(l2.b - l1.b) * (l1.m - l3.m); } void addLine(long long m, long long b) { Line nl = {m, b}; while (sz >= 2 && bad(hull[sz - 2], hull[sz - 1], nl)) sz--; hull[sz++] = nl; } int ptr = 0; long long query(long long x) { /* 쿼리가 단조 증가일 때 포인터 방식 O(1) amortized */ while (ptr + 1 < sz && hull[ptr + 1].m * x + hull[ptr + 1].b <= hull[ptr].m * x + hull[ptr].b) ptr++; return hull[ptr].m * x + hull[ptr].b; } long long a[MAXN], b[MAXN], dp[MAXN]; int N; int main(void) { scanf("%d", &N); for (int i = 1; i <= N; i++) scanf("%lld", &a[i]); for (int i = 1; i <= N; i++) scanf("%lld", &b[i]); /* 직선: y = b[i]*x + dp[i-1] (기울기 b[i] 단조 감소 가정) */ dp[0] = 0; addLine(b[1], dp[0]); for (int j = 1; j <= N; j++) { dp[j] = query(a[j]); if (j < N) addLine(b[j + 1], dp[j]); } printf("%lld\n", dp[N]); return 0; }

[문제] N개의 물건을 K개의 연속 구간으로 나누어,
각 구간의 비용 합을 최소화하세요.
구간 [l, r]의 비용은 C(l, r)이며, 이 함수는 사각 부등식을 만족합니다.
N ≤ 100,000, K ≤ 50.

기본 DP:
dp[k][j] = k개의 구간으로 첫 j개 물건을 나누는 최소 비용
dp[k][j] = min(dp[k-1][i] + C(i+1, j)) for 0 ≤ i < j
O(KN2)으로 N=100,000이면 시간 초과입니다.

분할 정복 최적화 원리:
opt[k][j] = dp[k][j]를 최소로 만드는 i값이라 할 때,
opt[k][j] ≤ opt[k][j+1] (단조성)이 성립하면 적용 가능합니다.
j의 범위 [jl, jr]에서 중간값 jm의 최적점을 찾고,
[jl, jm-1]은 opt 범위 [il, opt_m], [jm+1, jr]은 [opt_m, ir]로 재귀합니다.
각 레벨에서 O(N) 작업이고 재귀 깊이가 O(log N)이므로 총 O(KN log N).

Cdivide_conquer_opt.c
#include <stdio.h> #include <string.h> #define MAXN 100001 #define MAXK 51 #define INF 0x3f3f3f3f3f3f3f3fLL int N, K; long long dp[MAXK][MAXN]; long long psum[MAXN]; int a[MAXN]; /* 비용 함수 예: 구간 [l,r] 내 원소들의 분산 관련 비용 */ long long C(int l, int r) { long long s = psum[r] - psum[l - 1]; return s * s; /* 예시: 구간 합의 제곱 */ } void solve(int k, int jl, int jr, int il, int ir) { if (jl > jr) return; int jm = (jl + jr) / 2; int best_i = il; dp[k][jm] = INF; int up = ir < jm - 1 ? ir : jm - 1; for (int i = il; i <= up; i++) { long long val = dp[k - 1][i] + C(i + 1, jm); if (val < dp[k][jm]) { dp[k][jm] = val; best_i = i; } } solve(k, jl, jm - 1, il, best_i); solve(k, jm + 1, jr, best_i, ir); } int main(void) { scanf("%d %d", &N, &K); for (int i = 1; i <= N; i++) { scanf("%d", &a[i]); psum[i] = psum[i - 1] + a[i]; } memset(dp, 0x3f, sizeof(dp)); dp[0][0] = 0; for (int j = 1; j <= N; j++) dp[1][j] = C(1, j); for (int k = 2; k <= K; k++) solve(k, k, N, k - 1, N - 1); printf("%lld\n", dp[K][N]); return 0; }

[문제 유형] N개의 정점으로 이루어진 그래프에서,
연결 그래프의 개수를 세거나, 그래프를 연결 요소별로 분할하는 문제입니다.
예: N개의 정점을 가진 라벨 있는 연결 그래프의 개수를 구하세요.
N ≤ 20.

핵심 아이디어:
전체 그래프 수에서 비연결 그래프 수를 빼는 포함-배제 방식입니다.
g[S] = 정점 집합 S에 대한 모든 그래프 수 (연결 여부 무관)
f[S] = 정점 집합 S에 대한 연결 그래프 수
관계식: g[S] = Σ f[T] · g[S \ T] (T는 S의 부분집합 중 특정 정점을 포함하는 것)

구체적 전이:
S에서 가장 작은 번호의 정점 v를 고정합니다.
v를 포함하는 연결 요소 T에 대해:
g[S] = Σ f[T] · g[S \ T] (v ∈ T ⊆ S)
따라서: f[S] = g[S] - Σ f[T] · g[S \ T] (T ⊊ S, v ∈ T)
이렇게 하면 f[S]를 구할 수 있습니다.

g[S] 계산: S 내부의 가능한 간선 수가 e(S)이면, g[S] = 2e(S)입니다.
e(S)는 S에 속하는 정점 쌍 (i, j) 중 간선이 존재할 수 있는 쌍의 수입니다.
라벨 있는 그래프에서 e(S) = |S| · (|S| - 1) / 2.

Cconnected_component_dp.c
#include <stdio.h> #include <string.h> #define MAXN 20 #define MOD 1000000007LL int N; long long g[1 << MAXN]; /* 모든 그래프 수 */ long long f[1 << MAXN]; /* 연결 그래프 수 */ long long pw2[MAXN * MAXN]; /* 2의 거듭제곱 */ int popcount(int x) { int c = 0; while (x) { c += x & 1; x >>= 1; } return c; } int lowest_bit(int x) { return x & (-x); } int main(void) { scanf("%d", &N); pw2[0] = 1; for (int i = 1; i < N * N; i++) pw2[i] = pw2[i - 1] * 2 % MOD; int full = (1 << N) - 1; for (int S = 1; S <= full; S++) { int cnt = popcount(S); int edges = cnt * (cnt - 1) / 2; g[S] = pw2[edges]; } for (int S = 1; S <= full; S++) { int v = lowest_bit(S); /* S에서 가장 작은 비트 */ f[S] = g[S]; /* v를 포함하는 진부분집합 T를 열거 */ for (int T = (S - 1) & S; T > 0; T = (T - 1) & S) { if (T & v) f[S] = (f[S] - f[T] % MOD * g[S ^ T] % MOD + MOD) % MOD; } } printf("%lld\n", f[full]); return 0; }

기댓값 DP 심화 — 흡수 마르코프 체인과 가우스 소거
기댓값 DP마르코프 체인가우스 소거Silver

[문제] N개의 상태가 있는 확률 게임입니다.
상태 0에서 시작하여 상태 N-1(흡수 상태)에 도달할 때까지의 기대 이동 횟수를 구하세요.
각 상태 i에서 상태 j로 이동할 확률 p[i][j]가 주어집니다.
N ≤ 200.

기댓값 DP의 한계:
일반적인 기댓값 DP는 DAG(사이클 없는 그래프)에서만 직접 적용 가능합니다.
E[i] = 1 + Σ p[i][j] · E[j] (j ≠ 흡수 상태)
사이클이 있으면 E[i]끼리 서로 참조하므로, 연립방정식을 풀어야 합니다.

연립방정식 구성:
E[N-1] = 0 (흡수 상태)
각 비흡수 상태 i에 대해:
E[i] = 1 + p[i][0]·E[0] + p[i][1]·E[1] + ... + p[i][N-2]·E[N-2]
정리하면:
E[i] - p[i][0]·E[0] - p[i][1]·E[1] - ... = 1
즉 (N-1)개의 미지수, (N-1)개의 방정식 → 가우스 소거법으로 풀기

Cexpected_value_gauss.c
#include <stdio.h> #include <math.h> #define MAXN 201 int N; double p[MAXN][MAXN]; /* 전이 확률 */ double A[MAXN][MAXN + 1]; /* 확장 행렬 [계수 | 상수] */ double E[MAXN]; /* 기댓값 결과 */ void gauss(int n) { for (int col = 0; col < n; col++) { /* 피벗 선택 (부분 피벗팅) */ int pivot = col; for (int row = col + 1; row < n; row++) if (fabs(A[row][col]) > fabs(A[pivot][col])) pivot = row; /* 행 교환 */ for (int j = col; j <= n; j++) { double tmp = A[col][j]; A[col][j] = A[pivot][j]; A[pivot][j] = tmp; } /* 소거 */ for (int row = 0; row < n; row++) { if (row == col) continue; double ratio = A[row][col] / A[col][col]; for (int j = col; j <= n; j++) A[row][j] -= ratio * A[col][j]; } } for (int i = 0; i < n; i++) E[i] = A[i][n] / A[i][i]; } int main(void) { scanf("%d", &N); for (int i = 0; i < N; i++) for (int j = 0; j < N; j++) scanf("%lf", &p[i][j]); /* 연립방정식 구성: E[i] - Σ p[i][j]*E[j] = 1 */ int m = N - 1; /* 미지수 개수 (흡수 상태 제외) */ for (int i = 0; i < m; i++) { for (int j = 0; j < m; j++) { A[i][j] = (i == j ? 1.0 : 0.0) - p[i][j]; } A[i][m] = 1.0; /* 상수항 */ } gauss(m); printf("%.6f\n", E[0]); return 0; }

[개요] DP로 최적값을 구한 뒤, 실제 답(경로, 부분 수열, 선택된 물건 등)을 복원하는 기법입니다.

이번 섹션에서는 세 가지 대표 문제의 역추적 코드를 모두 다룹니다.

1. LIS (최장 증가 부분 수열) 역추적:
dp[i] = a[i]를 마지막 원소로 하는 LIS 길이
prev[i] = LIS에서 a[i] 바로 앞에 오는 원소의 인덱스

#include <stdio.h> #define MAXN 100001 int a[MAXN], dp[MAXN], prev_idx[MAXN], result[MAXN]; int N; int main(void) { scanf("%d", &N); for (int i = 0; i < N; i++) scanf("%d", &a[i]); int lis_len = 0, lis_end = 0; for (int i = 0; i < N; i++) { dp[i] = 1; prev_idx[i] = -1; for (int j = 0; j < i; j++) { if (a[j] < a[i] && dp[j] + 1 > dp[i]) { dp[i] = dp[j] + 1; prev_idx[i] = j; } } if (dp[i] > lis_len) { lis_len = dp[i]; lis_end = i; } } /* 역추적: lis_end에서 prev를 따라감 */ int cnt = 0, cur = lis_end; while (cur != -1) { result[cnt++] = a[cur]; cur = prev_idx[cur]; } printf("LIS 길이: %d\nLIS: ", lis_len); for (int i = cnt - 1; i >= 0; i--) printf("%d ", result[i]); printf("\n"); return 0; }

2. LCS (최장 공통 부분 수열) 역추적:
dp[i][j] = A[1..i]와 B[1..j]의 LCS 길이
역추적: dp[i][j]에서 시작하여 어디서 왔는지 추적합니다.
A[i] == B[j]이면 이 문자를 LCS에 포함하고 (i-1, j-1)로 이동
아니면 dp[i-1][j]와 dp[i][j-1] 중 큰 쪽으로 이동

#include <stdio.h> #include <string.h> #define MAXL 1001 char A[MAXL], B[MAXL], lcs[MAXL]; int dp[MAXL][MAXL]; int main(void) { scanf("%s %s", A + 1, B + 1); int n = strlen(A + 1), m = strlen(B + 1); for (int i = 1; i <= n; i++) for (int j = 1; j <= m; j++) dp[i][j] = (A[i] == B[j]) ? dp[i-1][j-1] + 1 : (dp[i-1][j] > dp[i][j-1] ? dp[i-1][j] : dp[i][j-1]); /* 역추적 */ int len = dp[n][m], idx = len; int i = n, j = m; while (i > 0 && j > 0) { if (A[i] == B[j]) { lcs[--idx] = A[i]; i--; j--; } else if (dp[i-1][j] > dp[i][j-1]) { i--; } else { j--; } } printf("LCS 길이: %d\nLCS: %s\n", len, lcs); return 0; }

SECTION 411Silver | 트리 심화
Heavy-Light Decomposition (HLD) — 개념
HLD트리세그먼트 트리Silver

[문제 유형] N개의 정점으로 이루어진 트리에서,
경로 쿼리(두 정점 사이의 합/최대/최소)와 경로 업데이트를 효율적으로 처리합니다.
N ≤ 100,000, Q ≤ 100,000.

핵심 아이디어: 트리의 간선을 "무거운 간선"(heavy edge)과 "가벼운 간선"(light edge)으로 분류합니다.
무거운 간선: 부모 → 서브트리 크기가 가장 큰 자식
가벼운 간선: 나머지 자식으로의 간선
무거운 간선으로 연결된 정점들은 하나의 "체인"(chain)을 형성합니다.

핵심 성질:
임의의 정점에서 루트까지 경로에서 가벼운 간선의 수는 최대 O(log N)개입니다.
이유: 가벼운 간선을 따라 올라가면 서브트리 크기가 최소 2배가 됩니다.
따라서 경로를 O(log N)개의 체인 구간으로 분해할 수 있습니다.

세그먼트 트리 연결:
각 체인의 정점들에 연속된 세그먼트 트리 인덱스를 부여합니다.
같은 체인 내의 구간 쿼리/업데이트는 세그먼트 트리의 구간 연산으로 O(log N)에 처리합니다.
경로 전체는 O(log N)개 체인 × O(log N) 세그먼트 트리 = O(log²N)에 처리합니다.

DFS 2단계:
DFS 1 (dfs_size): 서브트리 크기 계산, 무거운 자식(heavy child) 결정
DFS 2 (dfs_hld): 체인 번호, 세그먼트 트리 위치(pos), 체인 시작점(head) 결정
무거운 자식을 먼저 방문하여, 같은 체인이 연속 번호를 갖도록 합니다.

개념설명예시 (7개 정점 트리)
무거운 자식서브트리 크기가 가장 큰 자식루트(1)의 무거운 자식: 크기 4인 자식(2)
체인무거운 간선으로 연결된 경로1→2→4→7 (크기 4인 체인)
head[v]v가 속한 체인의 최상단 정점head[7] = 1
pos[v]v의 세그먼트 트리 인덱스pos[1]=0, pos[2]=1, pos[4]=2, pos[7]=3
가벼운 간선 수루트까지 최대 O(log N)7→(chain)→1: 가벼운 간선 0개
TIP — HLD vs 오일러 투어 오일러 투어: 서브트리 쿼리에 적합 (서브트리 → 연속 구간)
HLD: 경로 쿼리에 적합 (경로 → O(log N)개 연속 구간)
두 기법을 결합하면 서브트리 + 경로 쿼리를 모두 처리할 수 있습니다.
WARNING HLD는 정점 기반과 간선 기반 두 가지 변형이 있습니다.
간선 기반일 때 LCA에서 한 칸 더 올라가지 않도록 주의하세요.
쿼리 시 두 정점이 같은 체인에 올 때까지 반복해야 합니다.

트리 DP 심화 — Rerooting Technique (모든 루트 답 O(N))
Rerooting트리 DP전환 기법Silver

[문제] N개의 정점으로 이루어진 트리에서,
각 정점을 루트로 했을 때의 DP 값을 모두 구하세요.
예: 각 정점에서 가장 먼 정점까지의 거리 (트리의 지름 관련).
N ≤ 200,000.

무식한 방법: 각 정점을 루트로 DFS를 N번 → O(N2). N=200,000이면 시간 초과.

Rerooting 핵심 아이디어:
1단계 (DFS 1): 임의의 루트(예: 1번)에서 일반적인 트리 DP를 수행
dp_down[v] = v를 루트로 하는 서브트리에서의 답
2단계 (DFS 2): 루트를 v에서 v의 자식 u로 "전환"할 때,
dp_up[u] = u의 부모 방향(위쪽)으로의 답
이를 합치면: ans[u] = combine(dp_down[u], dp_up[u])

루트 전환 공식:
v가 현재 루트, u가 v의 자식일 때:
dp_up[u] = combine(dp_up[v], dp_down[v에서 u를 제외한 나머지 자식들])
"u를 제외한 나머지"를 효율적으로 구하려면 접두사/접미사 합을 사용합니다.

#include <stdio.h> #define MAXN 200001 int head_e[MAXN], to[MAXN*2], nxt[MAXN*2], ecnt; void add_edge(int u, int v) { to[++ecnt]=v; nxt[ecnt]=head_e[u]; head_e[u]=ecnt; to[++ecnt]=u; nxt[ecnt]=head_e[v]; head_e[v]=ecnt; } int N; int dp_down[MAXN]; /* 서브트리에서 가장 먼 거리 */ int dp_up[MAXN]; /* 부모 방향에서 가장 먼 거리 */ int ans[MAXN]; /* 최종 답: max(dp_down, dp_up) */ int child_list[MAXN*2], cl_start[MAXN], cl_cnt[MAXN]; /* DFS 1: dp_down 계산 */ void dfs1(int u, int p) { dp_down[u] = 0; for (int e = head_e[u]; e; e = nxt[e]) { int v = to[e]; if (v == p) continue; dfs1(v, u); if (dp_down[v] + 1 > dp_down[u]) dp_down[u] = dp_down[v] + 1; } } /* DFS 2: dp_up 계산 (rerooting) */ void dfs2(int u, int p) { /* 자식들의 dp_down+1 중 최대값과 두 번째 최대값을 구함 */ int max1 = -1, max2 = -1; for (int e = head_e[u]; e; e = nxt[e]) { int v = to[e]; if (v == p) continue; int val = dp_down[v] + 1; if (val >= max1) { max2 = max1; max1 = val; } else if (val > max2) max2 = val; } for (int e = head_e[u]; e; e = nxt[e]) { int v = to[e]; if (v == p) continue; /* v를 제외한 u의 최대 깊이 */ int best_without_v; if (dp_down[v] + 1 == max1) best_without_v = max2; else best_without_v = max1; /* dp_up[v] = max(dp_up[u], best_without_v) + 1 */ dp_up[v] = (dp_up[u] > best_without_v ? dp_up[u] : best_without_v) + 1; dfs2(v, u); } } int main(void) { scanf("%d", &N); for (int i = 0; i < N - 1; i++) { int u, v; scanf("%d %d", &u, &v); add_edge(u, v); } dfs1(1, 0); dp_up[1] = 0; dfs2(1, 0); for (int i = 1; i <= N; i++) { ans[i] = dp_down[i] > dp_up[i] ? dp_down[i] : dp_up[i]; printf("%d ", ans[i]); } printf("\n"); return 0; }

트리 위 게임 이론 — Sprague-Grundy on Tree
게임 이론Sprague-Grundy트리Silver

[문제] 루트 있는 트리에서 두 플레이어가 번갈아 가며 리프 노드를 제거합니다.

선수(첫 번째 플레이어)가 이기는지 판별하세요.
N ≤ 100,000.

게임 이론 기초 — Sprague-Grundy 정리:
모든 공정 게임(impartial game)은 그런디 수(Grundy number)로 분석 가능합니다.
독립적인 게임들의 합은 XOR로 결합됩니다.
그런디 수가 0이면 후수 승, 0이 아니면 선수 승입니다.

트리에서의 그런디 수:
각 서브트리를 독립적인 게임으로 봅니다.
리프 노드의 그런디 수: 0 (이동 불가)
내부 노드의 그런디 수: 자식들의 그런디 수에 대한 연산
트리 게임의 종류에 따라 공식이 달라집니다.

Green Hackenbush on Tree:
트리의 각 간선이 하나의 게임입니다.
그런디 수 = 각 간선에서 루트까지 깊이의 XOR.
더 일반적으로: grundy[v] = XOR(grundy[child] + 1) (Nim-value)

[문제] N개의 정점으로 이루어진 루트 트리에서,
1) 정점 v의 서브트리에 있는 모든 정점의 값에 x를 더하기
2) 정점 v의 현재 값을 출력하기

오일러 투어 (Euler Tour):
DFS 진입 시각 tin[v]와 이탈 시각 tout[v]를 기록하면,
v의 서브트리는 배열에서 [tin[v], tout[v]] 연속 구간에 대응합니다.
서브트리 업데이트 → 구간 업데이트, 단일 정점 쿼리 → 점 쿼리.

Euler Tour LCA (ETL):
오일러 투어로 LCA도 구할 수 있습니다.
DFS 중 방문하는 정점을 순서대로 기록하면 (크기 2N-1 배열),
u와 v의 LCA = 이 배열에서 첫 등장 위치 사이의 최소 깊이 정점.
Sparse Table로 전처리하면 O(1) LCA 쿼리가 가능합니다.

#include <stdio.h> #include <string.h> #define MAXN 200001 int head_e[MAXN], to[MAXN*2], nxt[MAXN*2], ecnt; void add_edge(int u, int v) { to[++ecnt]=v; nxt[ecnt]=head_e[u]; head_e[u]=ecnt; to[++ecnt]=u; nxt[ecnt]=head_e[v]; head_e[v]=ecnt; } int N, Q; int tin[MAXN], tout[MAXN], timer; void dfs(int u, int p) { tin[u] = ++timer; for (int e = head_e[u]; e; e = nxt[e]) { int v = to[e]; if (v != p) dfs(v, u); } tout[u] = timer; } /* BIT (Fenwick Tree) for range update, point query */ long long bit[MAXN]; void bit_add(int i, long long v) { for (; i <= N; i += i & (-i)) bit[i] += v; } long long bit_sum(int i) { long long s = 0; for (; i > 0; i -= i & (-i)) s += bit[i]; return s; } /* 구간 [l, r]에 v를 더하기 */ void range_add(int l, int r, long long v) { bit_add(l, v); bit_add(r + 1, -v); } /* 점 i의 값 쿼리 */ long long point_query(int i) { return bit_sum(i); } int main(void) { scanf("%d %d", &N, &Q); for (int i = 0; i < N - 1; i++) { int u, v; scanf("%d %d", &u, &v); add_edge(u, v); } dfs(1, 0); while (Q--) { int type; scanf("%d", &type); if (type == 1) { int v; long long x; scanf("%d %lld", &v, &x); range_add(tin[v], tout[v], x); } else { int v; scanf("%d", &v); printf("%lld\n", point_query(tin[v])); } } return 0; }

Z 알고리즘 — 최장 공통 접두사 O(N)
Z 알고리즘문자열 매칭접두사Silver

[문제] 문자열 S가 주어집니다.
각 위치 i에 대해, S와 S[i..]의 최장 공통 접두사(LCP) 길이 Z[i]를 구하세요.
즉, Z[i] = S[0..Z[i]-1] == S[i..i+Z[i]-1]인 최대 Z[i].
|S| ≤ 1,000,000.

활용: 패턴 매칭에 사용 가능합니다.
패턴 P + 구분자 + 텍스트 T로 이어 붙인 문자열의 Z 배열을 구하면,
Z[i] == |P|인 위치가 패턴이 등장하는 위치입니다. KMP의 대안입니다.

Z 알고리즘 핵심 (Z-box):
[l, r]은 현재까지 발견된 가장 오른쪽 Z-box (S[l..r] == S[0..r-l])입니다.
새 위치 i에 대해:
1) i > r이면: 처음부터 비교하여 Z[i]를 구하고 [l, r] 갱신
2) i ≤ r이면: 대응 위치 k = i - l에서 Z[k]를 참조
   a) Z[k] < r - i + 1이면: Z[i] = Z[k] (확장 불필요)
   b) Z[k] ≥ r - i + 1이면: r+1부터 비교하여 확장

Cz_algorithm.c
#include <stdio.h> #include <string.h> #define MAXN 1000001 char S[MAXN]; int Z[MAXN]; void z_function(char *s, int n) { Z[0] = n; int l = 0, r = 0; for (int i = 1; i < n; i++) { if (i < r) Z[i] = (Z[i - l] < r - i) ? Z[i - l] : r - i; else Z[i] = 0; while (i + Z[i] < n && s[Z[i]] == s[i + Z[i]]) Z[i]++; if (i + Z[i] > r) { l = i; r = i + Z[i]; } } } int main(void) { scanf("%s", S); int n = strlen(S); z_function(S, n); for (int i = 0; i < n; i++) printf("%d ", Z[i]); printf("\n"); return 0; } /* 패턴 매칭 예: P + "$" + T의 Z 배열에서 Z[i] == |P|인 위치 찾기 */

[문제] 문자열 S에서 모든 위치를 중심으로 하는 최장 팰린드롬의 반지름을 구하세요.
홀수 길이와 짝수 길이 팰린드롬을 모두 처리합니다.

핵심 트릭 — '#' 삽입:
원본: abc → 변환: #a#b#c#
이렇게 하면 짝수 길이 팰린드롬도 '#'을 중심으로 하는 홀수 길이 팰린드롬이 됩니다.
변환 후 배열 길이: 2N+1.

Manacher 알고리즘:
P[i] = 변환 문자열에서 위치 i를 중심으로 하는 최장 팰린드롬 반지름
Z 알고리즘과 유사하게, 현재까지의 가장 오른쪽 팰린드롬 경계를 관리합니다.
[c, r]: 중심 c, 오른쪽 경계 r인 팰린드롬
새 위치 i에 대해:
1) i < r이면: 대칭 위치 mirror = 2c - i의 P[mirror]를 참조
   P[i] = min(P[mirror], r - i) 에서 시작하여 확장
2) i ≥ r이면: P[i] = 0에서 시작하여 확장
3) 확장 후 i + P[i] > r이면 [c, r] 갱신

Suffix Array O(N log N) — SA + LCP (Kasai)
Suffix ArrayLCPKasaiSilver

[문제] 문자열 S가 주어집니다.
모든 접미사를 사전순으로 정렬한 Suffix Array와,
인접 접미사 간의 최장 공통 접두사(LCP) 배열을 구하세요.
|S| ≤ 200,000.

Suffix Array란:
S의 모든 접미사를 사전순 정렬한 결과의 시작 인덱스 배열입니다.
예: S = "banana" → 접미사: banana, anana, nana, ana, na, a
사전순: a, ana, anana, banana, na, nana → SA = [5, 3, 1, 0, 4, 2]

O(N log N) 구축 (Radix Sort 기반):
1) 길이 1 기준으로 정렬 → 순위(rank) 배분
2) 길이 2, 4, 8, ... 기준으로 반복 정렬
   각 단계에서 (rank[i], rank[i + len]) 쌍으로 정렬
3) O(log N)번의 O(N) radix sort → 총 O(N log N)

Csuffix_array.c
#include <stdio.h> #include <string.h> #define MAXN 200001 char S[MAXN]; int sa[MAXN], rk[MAXN], tmp[MAXN], lcp[MAXN]; int cnt[MAXN], buf[MAXN]; int N; void build_sa() { int m = 256; /* 알파벳 크기 */ for (int i = 0; i < N; i++) rk[i] = S[i]; for (int i = 0; i < N; i++) sa[i] = i; for (int len = 1; len < N; len *= 2) { /* 2번째 키 기준 counting sort */ memset(cnt, 0, (m + 1) * sizeof(int)); for (int i = 0; i < N; i++) cnt[i + len < N ? rk[i + len] + 1 : 0]++; for (int i = 1; i <= m; i++) cnt[i] += cnt[i-1]; for (int i = N - 1; i >= 0; i--) buf[--cnt[sa[i] + len < N ? rk[sa[i] + len] + 1 : 0]] = sa[i]; /* 1번째 키 기준 counting sort */ memset(cnt, 0, (m + 1) * sizeof(int)); for (int i = 0; i < N; i++) cnt[rk[buf[i]] + 1]++; for (int i = 1; i <= m; i++) cnt[i] += cnt[i-1]; for (int i = N - 1; i >= 0; i--) sa[--cnt[rk[buf[i]] + 1]] = buf[i]; /* 새 순위 계산 */ tmp[sa[0]] = 0; for (int i = 1; i < N; i++) { tmp[sa[i]] = tmp[sa[i-1]]; if (rk[sa[i]] != rk[sa[i-1]] || (sa[i]+len < N ? rk[sa[i]+len] : -1) != (sa[i-1]+len < N ? rk[sa[i-1]+len] : -1)) tmp[sa[i]]++; } memcpy(rk, tmp, N * sizeof(int)); m = rk[sa[N-1]] + 1; if (m == N) break; } } /* Kasai's Algorithm: LCP 배열 O(N) */ void build_lcp() { int k = 0; for (int i = 0; i < N; i++) rk[sa[i]] = i; for (int i = 0; i < N; i++) { if (rk[i] == 0) { k = 0; continue; } int j = sa[rk[i] - 1]; while (i + k < N && j + k < N && S[i + k] == S[j + k]) k++; lcp[rk[i]] = k; if (k > 0) k--; } } int main(void) { scanf("%s", S); N = strlen(S); build_sa(); build_lcp(); printf("SA: "); for (int i = 0; i < N; i++) printf("%d ", sa[i]); printf("\nLCP: "); for (int i = 0; i < N; i++) printf("%d ", lcp[i]); printf("\n"); return 0; }

Suffix Automaton — 접미사 오토마톤과 부분 문자열 개수
Suffix AutomatonDAG문자열Silver

[문제] 문자열 S에 대해 서로 다른 부분 문자열의 개수를 구하세요.
|S| ≤ 200,000.

Suffix Automaton이란:
문자열 S의 모든 부분 문자열을 인식하는 최소 DFA(결정적 유한 오토마톤)입니다.
상태 수: 최대 2N-1개, 전이 수: 최대 3N-4개.
온라인(한 글자씩 추가)으로 O(N) 시간에 구축 가능합니다.

핵심 구조:
각 상태 s는 S의 부분 문자열 집합 endpos(s)와 대응합니다.
endpos(s) = s에 대응하는 모든 부분 문자열이 끝나는 위치의 집합
link(s) = suffix link, endpos가 s를 진포함하는 가장 긴 상태
len(s) = s에 대응하는 가장 긴 부분 문자열의 길이

서로 다른 부분 문자열 개수:
각 상태 s가 나타내는 부분 문자열 수 = len(s) - len(link(s))
총 개수 = Σ (len(s) - len(link(s))) (초기 상태 제외)

Csuffix_automaton.c
#include <stdio.h> #include <string.h> #define MAXN 400002 /* 최대 2*|S| 상태 */ #define ALPHA 26 struct State { int len, link; int next[ALPHA]; } st[MAXN]; int sz, last; void sa_init() { st[0].len = 0; st[0].link = -1; memset(st[0].next, -1, sizeof(st[0].next)); sz = 1; last = 0; } void sa_extend(int c) { int cur = sz++; st[cur].len = st[last].len + 1; memset(st[cur].next, -1, sizeof(st[cur].next)); int p = last; while (p != -1 && st[p].next[c] == -1) { st[p].next[c] = cur; p = st[p].link; } if (p == -1) { st[cur].link = 0; } else { int q = st[p].next[c]; if (st[p].len + 1 == st[q].len) { st[cur].link = q; } else { int clone = sz++; st[clone] = st[q]; st[clone].len = st[p].len + 1; while (p != -1 && st[p].next[c] == q) { st[p].next[c] = clone; p = st[p].link; } st[q].link = clone; st[cur].link = clone; } } last = cur; } int main(void) { char S[200001]; scanf("%s", S); int n = strlen(S); sa_init(); for (int i = 0; i < n; i++) sa_extend(S[i] - 'a'); /* 서로 다른 부분 문자열 개수 */ long long ans = 0; for (int i = 1; i < sz; i++) ans += st[i].len - st[st[i].link].len; printf("%lld\n", ans); return 0; }
