// 내림차순 정렬 qsort(w, n, sizeof(int), cmp_desc); int ans = 0; for (int i = 0; i < n; i++) { int val = w[i] * (i + 1); // w[i]가 최소, i+1개 사용 if (val > ans) ans = val; }

입력
