랜덤 알고리즘이란?
결정론적(deterministic) 알고리즘과 달리, 난수(random number)를 활용하여 동작하는 알고리즘입니다.
최악의 경우를 확률적으로 회피할 수 있어, 평균적으로 매우 좋은 성능을 보장합니다.
대표적으로 랜덤 피벗 퀵소트, 랜덤 해싱, Miller-Rabin 소수 판별 등이 있습니다.

랜덤 피벗 퀵소트
일반 퀵소트는 이미 정렬된 배열에서 O(N²)이 됩니다.
피벗을 랜덤으로 선택하면, 어떤 입력이든 기대 시간 복잡도 O(N log N)이 됩니다.
대회에서 anti-quicksort 데이터를 방어하는 가장 간단한 방법입니다.

랜덤 해싱
해시 함수의 base와 mod를 랜덤으로 선택합니다.
이렇게 하면 해시 충돌을 의도적으로 만드는 anti-hash 테스트 케이스를 방어할 수 있습니다.
특히 문자열 해싱에서 유용합니다.

Crandom_quicksort.c
#include <stdio.h> #include <stdlib.h> #include <time.h> void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; } /* 랜덤 피벗 선택 후 파티션 */ int partition(int arr[], int lo, int hi) { /* 핵심: lo~hi 중 랜덤 인덱스를 피벗으로 */ int ri = lo + rand() % (hi - lo + 1); swap(&arr[ri], &arr[hi]); int pivot = arr[hi]; int i = lo - 1; for (int j = lo; j < hi; j++) { if (arr[j] <= pivot) { i++; swap(&arr[i], &arr[j]); } } swap(&arr[i + 1], &arr[hi]); return i + 1; } void quicksort(int arr[], int lo, int hi) { if (lo < hi) { int p = partition(arr, lo, hi); quicksort(arr, lo, p - 1); quicksort(arr, p + 1, hi); } } int main(void) { srand(time(NULL)); int arr[] = {5, 3, 8, 1, 9, 2, 7, 4, 6}; int n = sizeof(arr) / sizeof(arr[0]); quicksort(arr, 0, n - 1); for (int i = 0; i < n; i++) printf("%d ", arr[i]); printf("\n"); return 0; }

근사 알고리즘이란?
NP-hard 문제는 다항 시간 내에 최적해를 구할 수 없다고 믿어집니다.
근사 알고리즘은 최적해에 가까운 해를 다항 시간에 구합니다.
α-근사란: 알고리즘의 해가 최적해의 α배 이내임을 보장하는 것입니다.

정점 커버 2-근사 알고리즘
정점 커버: 모든 간선의 양 끝점 중 하나 이상이 포함되는 정점 집합의 최소 크기를 구합니다.
2-근사 알고리즘:
1) 아무 간선 (u, v)를 선택합니다.
2) u와 v를 모두 커버에 넣습니다.
3) u 또는 v에 연결된 모든 간선을 제거합니다.
4) 간선이 남아 있으면 1)로 돌아갑니다.
이 알고리즘은 최적해의 2배 이내를 보장합니다.

#include <stdio.h> #include <string.h> #define MAXN 1001 #define MAXM 10001 int eu[MAXM], ev[MAXM]; /* 간선의 양 끝점 */ int inCover[MAXN]; /* 커버에 포함 여부 */ int used[MAXM]; /* 간선 제거 여부 */ int main(void) { int n, m; scanf("%d %d", &n, &m); for (int i = 0; i < m; i++) scanf("%d %d", &eu[i], &ev[i]); memset(inCover, 0, sizeof(inCover)); memset(used, 0, sizeof(used)); for (int i = 0; i < m; i++) { if (used[i]) continue; /* 간선 (eu[i], ev[i])의 양 끝점을 커버에 추가 */ inCover[eu[i]] = 1; inCover[ev[i]] = 1; /* 이 두 정점에 연결된 모든 간선 제거 */ for (int j = i; j < m; j++) { if (eu[j] == eu[i] || eu[j] == ev[i] || ev[j] == eu[i] || ev[j] == ev[i]) used[j] = 1; } } int coverSize = 0; printf("Vertex Cover: "); for (int i = 1; i <= n; i++) { if (inCover[i]) { printf("%d ", i); coverSize++; } } printf("\nSize: %d\n", coverSize); return 0; }

TSP 2-근사 (MST 기반)
삼각 부등식을 만족하는 완전 그래프에서:
1) MST를 구합니다.
2) MST를 DFS로 순회합니다 (preorder).
3) 방문 순서대로 정점을 나열하면 해밀턴 경로가 됩니다.
이 경로의 비용은 최적 TSP 투어의 2배 이내입니다.

Ctsp_2approx.c
#include <stdio.h> #include <stdlib.h> #include <math.h> #define MAXN 101 double x[MAXN], y[MAXN]; double dist[MAXN][MAXN]; int parent[MAXN], visited[MAXN], tour[MAXN]; int adj[MAXN][MAXN], deg[MAXN]; int n, tourLen; double calcDist(int a, int b) { double dx = x[a] - x[b], dy = y[a] - y[b]; return sqrt(dx*dx + dy*dy); } /* Prim MST */ void primMST(void) { double key[MAXN]; int inMST[MAXN]; for (int i = 0; i < n; i++) { key[i] = 1e18; inMST[i] = 0; parent[i] = -1; } key[0] = 0; for (int cnt = 0; cnt < n; cnt++) { int u = -1; for (int v = 0; v < n; v++) if (!inMST[v] && (u == -1 || key[v] < key[u])) u = v; inMST[u] = 1; for (int v = 0; v < n; v++) if (!inMST[v] && dist[u][v] < key[v]) { key[v] = dist[u][v]; parent[v] = u; } } /* MST 인접 리스트 구성 */ for (int i = 1; i < n; i++) { adj[parent[i]][deg[parent[i]]++] = i; adj[i][deg[i]++] = parent[i]; } } /* DFS preorder 순회 */ void dfs(int u) { visited[u] = 1; tour[tourLen++] = u; for (int i = 0; i < deg[u]; i++) if (!visited[adj[u][i]]) dfs(adj[u][i]); } int main(void) { scanf("%d", &n); for (int i = 0; i < n; i++) scanf("%lf %lf", &x[i], &y[i]); for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) dist[i][j] = calcDist(i, j); primMST(); tourLen = 0; dfs(0); double total = 0; for (int i = 0; i < tourLen; i++) total += dist[tour[i]][tour[(i+1) % tourLen]]; printf("Tour: "); for (int i = 0; i < tourLen; i++) printf("%d ", tour[i]); printf("\nCost: %.2f\n", total); return 0; }

시뮬레이티드 어닐링(Simulated Annealing)이란?
금속을 서서히 냉각(annealing)하면 결정이 안정된 구조를 이루는 물리 현상에서 영감을 받았습니다.
현재 해 근처에서 이웃 해를 생성하고, 더 좋으면 무조건 수락합니다.
더 나쁘더라도 일정 확률로 수락합니다 (온도가 높을수록 확률 높음).
온도를 서서히 낮추면서 반복하면 지역 최적해(local optimum)를 탈출해 전역 최적해에 근접합니다.

핵심 파라미터
1) 초기 온도 T0: 높을수록 탐색 범위가 넓습니다 (보통 1000~10000).
2) 냉각률 α: 매 반복마다 T ← T × α (보통 0.995~0.9999).
3) 종료 온도 Tend: 이 이하면 종료합니다 (보통 1e-8~1e-4).
4) 수락 확률: ΔE < 0이면 무조건 수락. ΔE ≥ 0이면 확률 e-ΔE/T로 수락.

Csa_tsp.c
#include <stdio.h> #include <stdlib.h> #include <math.h> #include <time.h> #define MAXN 201 int n; double px[MAXN], py[MAXN]; int route[MAXN], best[MAXN]; double d(int a, int b) { double dx = px[a]-px[b], dy = py[a]-py[b]; return sqrt(dx*dx + dy*dy); } double tourCost(int r[]) { double c = 0; for (int i = 0; i < n; i++) c += d(r[i], r[(i+1)%n]); return c; } double frand(void) { return (double)rand() / RAND_MAX; } void sa(void) { double T = 10000.0; /* 초기 온도 */ double alpha = 0.9995; /* 냉각률 */ double Tend = 1e-8; /* 종료 온도 */ double curCost = tourCost(route); double bestCost = curCost; for (int i = 0; i < n; i++) best[i] = route[i]; while (T > Tend) { /* 이웃 해 생성: 2-opt (i, j 구간 반전) */ int i = rand() % n, j = rand() % n; if (i > j) { int t = i; i = j; j = t; } if (i == j) { T *= alpha; continue; } /* 구간 [i, j] 반전 */ for (int a = i, b = j; a < b; a++, b--) { int t = route[a]; route[a] = route[b]; route[b] = t; } double newCost = tourCost(route); double delta = newCost - curCost; if (delta < 0 || frand() < exp(-delta / T)) { /* 수락 */ curCost = newCost; if (curCost < bestCost) { bestCost = curCost; for (int k = 0; k < n; k++) best[k] = route[k]; } } else { /* 거부: 원래대로 복원 */ for (int a = i, b = j; a < b; a++, b--) { int t = route[a]; route[a] = route[b]; route[b] = t; } } T *= alpha; } printf("Best cost: %.2f\n", bestCost); } int main(void) { srand(time(NULL)); scanf("%d", &n); for (int i = 0; i < n; i++) { scanf("%lf %lf", &px[i], &py[i]); route[i] = i; } sa(); printf("Tour: "); for (int i = 0; i < n; i++) printf("%d ", best[i]); printf("\n"); return 0; }

유전 알고리즘(Genetic Algorithm, GA)이란?
자연 선택과 유전학의 원리를 모방한 메타 휴리스틱 최적화 기법입니다.
해(solution)를 "개체(individual)"로, 해의 집합을 "모집단(population)"으로 표현합니다.
세대(generation)를 반복하며 더 좋은 해를 진화시킵니다.

GA의 핵심 연산
1) 선택(Selection): 적합도(fitness)가 높은 개체를 부모로 선택합니다.
   - 룰렛 휠, 토너먼트 선택 등이 있습니다.
2) 교차(Crossover): 두 부모의 유전자를 조합해 자식을 생성합니다.
   - 1-point, 2-point, uniform crossover 등이 있습니다.
3) 돌연변이(Mutation): 일정 확률로 유전자를 변형합니다.
   - 지역 최적해 탈출에 기여합니다.
4) 대체(Replacement): 새로운 세대를 구성합니다.

Cgenetic_algorithm.c
#include <stdio.h> #include <stdlib.h> #include <time.h> /* 간단한 GA: f(x) = -(x-50)^2 + 2500 최대화 (0 ≤ x ≤ 100) */ #define POP 20 /* 모집단 크기 */ #define GEN 100 /* 세대 수 */ #define MUT_RATE 0.1 /* 돌연변이 확률 */ int pop[POP], newpop[POP]; int fitness(int x) { return -(x - 50) * (x - 50) + 2500; } /* 토너먼트 선택: 2개 중 더 좋은 것 */ int select_parent(void) { int a = rand() % POP, b = rand() % POP; return fitness(pop[a]) > fitness(pop[b]) ? pop[a] : pop[b]; } /* 교차: 평균값 */ int crossover(int p1, int p2) { return (p1 + p2) / 2; } /* 돌연변이: ±5 범위 내 변형 */ int mutate(int x) { if ((double)rand() / RAND_MAX < MUT_RATE) { x += (rand() % 11) - 5; /* -5 ~ +5 */ if (x < 0) x = 0; if (x > 100) x = 100; } return x; } int main(void) { srand(time(NULL)); /* 초기 모집단: 랜덤 생성 */ for (int i = 0; i < POP; i++) pop[i] = rand() % 101; for (int g = 0; g < GEN; g++) { /* 새 세대 생성 */ for (int i = 0; i < POP; i++) { int p1 = select_parent(); int p2 = select_parent(); int child = crossover(p1, p2); newpop[i] = mutate(child); } for (int i = 0; i < POP; i++) pop[i] = newpop[i]; } /* 최적 개체 출력 */ int bestX = pop[0], bestF = fitness(pop[0]); for (int i = 1; i < POP; i++) { int f = fitness(pop[i]); if (f > bestF) { bestF = f; bestX = pop[i]; } } printf("Best x = %d, f(x) = %d\n", bestX, bestF); return 0; }

병렬 이분 탐색(Parallel Binary Search)이란?
Q개의 쿼리가 있고, 각 쿼리의 답을 이분 탐색으로 구해야 할 때 사용합니다.
각 쿼리를 독립적으로 이분 탐색하면 O(Q × N log N)이 되지만,
모든 쿼리를 동시에 이분 탐색하면 O((N + Q) log N)으로 줄일 수 있습니다.

핵심 아이디어
1) 모든 쿼리의 이분 탐색 구간 [lo, hi]를 관리합니다.
2) 각 반복에서 mid = (lo + hi) / 2를 구합니다.
3) 시간 축에서 mid까지의 이벤트를 처리한 후, 각 쿼리의 조건을 확인합니다.
4) 조건 만족 시 hi = mid, 아니면 lo = mid + 1로 갱신합니다.
5) 모든 쿼리가 수렴할 때까지 반복합니다 (최대 log N 라운드).

CDQ 분할 정복이란?
Chen Dan Qi가 제안한 오프라인 분할 정복 기법입니다.
3차원 편향 순서(partial order) 문제에 주로 사용됩니다.
(ai ≤ aj) ∧ (bi ≤ bj) ∧ (ci ≤ cj)를 만족하는 쌍 (i, j)의 개수를 구합니다.

핵심 아이디어
1) 1차원(a)은 정렬로 처리합니다.
2) 2차원(b)은 CDQ 분할 정복의 머지 단계에서 처리합니다.
3) 3차원(c)은 BIT로 처리합니다.
시간 복잡도: O(N log² N)

분할 정복 구조
구간 [lo, hi]를 처리할 때:
1) [lo, mid]를 재귀적으로 처리합니다.
2) [mid+1, hi]를 재귀적으로 처리합니다.
3) 왼쪽 구간의 원소가 오른쪽 구간의 원소에 미치는 영향을 계산합니다.
   - 양쪽을 b 기준으로 정렬합니다.
   - 투 포인터로 b 조건을 처리하며 BIT에 c 값을 추가/쿼리합니다.

프랙셔널 캐스케이딩(Fractional Cascading)이란?
K개의 정렬된 리스트에서 같은 값을 동시에 이분 탐색해야 할 때 사용합니다.
나이브하게 하면 O(K log N)이지만, 프랙셔널 캐스케이딩을 적용하면 O(log N + K)로 줄일 수 있습니다.

핵심 아이디어
1) K번째(마지막) 리스트부터 시작하여 (K-1)번째 리스트로 올라갑니다.
2) 각 리스트에 다음 리스트 원소의 절반(every other element)을 삽입합니다.
3) 각 원소에 "다음 리스트에서의 위치 포인터"를 저장합니다.
4) 첫 번째 리스트에서 이분 탐색 1회 후, 나머지는 포인터를 따라가며 O(1)에 처리합니다.

응용
- 2D 범위 탐색 (range tree + fractional cascading)
- 다층 자료구조에서 이분 탐색 반복 최적화
- 기하 알고리즘 (polygon 안에 점 포함 여부 등)

HLD + 세그먼트 트리 Lazy 조합
Heavy-Light Decomposition(HLD)은 트리를 체인으로 분할하여,
경로 쿼리를 세그먼트 트리 구간 연산으로 바꿔줍니다.
Lazy Propagation을 결합하면 경로에 대한 구간 업데이트 + 구간 쿼리를 O(log² N)에 처리할 수 있습니다.

핵심 구조
1) DFS로 subtree 크기를 구하고, heavy child를 결정합니다.
2) heavy child를 우선으로 DFS 순서(오일러 투어)를 매깁니다.
3) 같은 체인의 노드들이 연속된 구간이 됩니다.
4) 경로 쿼리는 LCA까지 체인을 올라가며 세그먼트 트리 쿼리/업데이트를 합니다.

Chld_segtree_lazy.c
#include <stdio.h> #include <string.h> #define MAXN 100001 #define MAXE 200002 /* 인접 리스트 */ int head[MAXN], nxt[MAXE], to[MAXE], ecnt; void addEdge(int u, int v) { nxt[++ecnt] = head[u]; head[u] = ecnt; to[ecnt] = v; nxt[++ecnt] = head[v]; head[v] = ecnt; to[ecnt] = u; } /* HLD 변수 */ int par[MAXN], dep[MAXN], sz[MAXN], heavy[MAXN]; int chainTop[MAXN], pos[MAXN], curPos; int n; long long val[MAXN]; void dfs1(int u, int p, int d) { par[u] = p; dep[u] = d; sz[u] = 1; heavy[u] = -1; int maxSz = 0; for (int e = head[u]; e; e = nxt[e]) { int v = to[e]; if (v == p) continue; dfs1(v, u, d + 1); sz[u] += sz[v]; if (sz[v] > maxSz) { maxSz = sz[v]; heavy[u] = v; } } } void dfs2(int u, int top) { chainTop[u] = top; pos[u] = ++curPos; if (heavy[u] != -1) dfs2(heavy[u], top); for (int e = head[u]; e; e = nxt[e]) { int v = to[e]; if (v != par[u] && v != heavy[u]) dfs2(v, v); } } /* 세그먼트 트리 + Lazy */ long long tree[MAXN * 4], lazy[MAXN * 4]; void pushDown(int nd, int l, int r) { if (lazy[nd]) { int mid = (l + r) / 2; tree[nd*2] += lazy[nd] * (mid - l + 1); tree[nd*2+1] += lazy[nd] * (r - mid); lazy[nd*2] += lazy[nd]; lazy[nd*2+1] += lazy[nd]; lazy[nd] = 0; } } void update(int nd, int l, int r, int ql, int qr, long long v) { if (ql > r || qr < l) return; if (ql <= l && r <= qr) { tree[nd] += v * (r - l + 1); lazy[nd] += v; return; } pushDown(nd, l, r); int mid = (l + r) / 2; update(nd*2, l, mid, ql, qr, v); update(nd*2+1, mid+1, r, ql, qr, v); tree[nd] = tree[nd*2] + tree[nd*2+1]; } long long query(int nd, int l, int r, int ql, int qr) { if (ql > r || qr < l) return 0; if (ql <= l && r <= qr) return tree[nd]; pushDown(nd, l, r); int mid = (l + r) / 2; return query(nd*2, l, mid, ql, qr) + query(nd*2+1, mid+1, r, ql, qr); } /* HLD 경로 업데이트: u ~ v 경로에 +v_add */ void pathUpdate(int u, int v, long long addVal) { while (chainTop[u] != chainTop[v]) { if (dep[chainTop[u]] < dep[chainTop[v]]) { int t = u; u = v; v = t; } update(1, 1, n, pos[chainTop[u]], pos[u], addVal); u = par[chainTop[u]]; } if (dep[u] > dep[v]) { int t = u; u = v; v = t; } update(1, 1, n, pos[u], pos[v], addVal); } /* HLD 경로 쿼리: u ~ v 경로 합 */ long long pathQuery(int u, int v) { long long res = 0; while (chainTop[u] != chainTop[v]) { if (dep[chainTop[u]] < dep[chainTop[v]]) { int t = u; u = v; v = t; } res += query(1, 1, n, pos[chainTop[u]], pos[u]); u = par[chainTop[u]]; } if (dep[u] > dep[v]) { int t = u; u = v; v = t; } res += query(1, 1, n, pos[u], pos[v]); return res; } int main(void) { int q; scanf("%d %d", &n, &q); for (int i = 1; i <= n; i++) scanf("%lld", &val[i]); for (int i = 0; i < n - 1; i++) { int u, v; scanf("%d %d", &u, &v); addEdge(u, v); } dfs1(1, 0, 0); curPos = 0; dfs2(1, 1); /* 초기값 세팅 */ for (int i = 1; i <= n; i++) update(1, 1, n, pos[i], pos[i], val[i]); while (q--) { int op, u, v; scanf("%d %d %d", &op, &u, &v); if (op == 1) { long long w; scanf("%lld", &w); pathUpdate(u, v, w); } else { printf("%lld\n", pathQuery(u, v)); } } return 0; }

TIP — HLD 구현 체크리스트 1) dfs1: subtree 크기, 부모, 깊이, heavy child 계산
2) dfs2: 오일러 투어 순서, 체인 톱 설정
3) 경로 쿼리: 체인 톱이 다르면 깊은 쪽을 올림
4) 같은 체인이 되면 깊이 순서로 구간 쿼리

dep[u]를 비교하면 안 됩니다!

부분 점수(subtask)란?
많은 대회에서 한 문제를 여러 서브태스크로 나누어 각각 독립적으로 채점합니다.

정해를 못 구해도 서브태스크 1, 2만 풀어 40점을 확보할 수 있습니다!

서브태스크별 풀이 전략
N ≤ 20: 비트마스크 완전 탐색 O(2N)
N ≤ 100: O(N³) 알고리즘 (플로이드, 행렬 곱셈 등)
N ≤ 5000: O(N²) DP
N ≤ 105: O(N log N) (정렬, 세그먼트 트리, 이분 탐색)
N ≤ 106: O(N) (투 포인터, 선형 DP, 스택)
