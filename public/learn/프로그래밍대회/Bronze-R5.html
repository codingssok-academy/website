SECTION 301Bronze | DP 입문
DP 개념 — 메모이제이션 vs 타뷸레이션
DPmemoizationtabulationBronze

동적 프로그래밍(Dynamic Programming, DP)은 큰 문제를 작은 부분 문제로 나누어 풀되, 이미 계산한 결과를 저장(캐싱)하여 중복 계산을 피하는 기법입니다. DP가 적용되려면 두 가지 조건이 필요합니다:

1. 최적 부분 구조(Optimal Substructure): 큰 문제의 최적해가 작은 부분 문제의 최적해로 구성됩니다.
2. 중복 부분 문제(Overlapping Subproblems): 같은 부분 문제가 여러 번 반복 계산됩니다.

메모이제이션(Top-Down): 재귀 + 캐싱. 필요한 부분 문제만 계산합니다.
타뷸레이션(Bottom-Up): 반복문으로 작은 문제부터 채워 나갑니다. 스택 오버플로 걱정이 없습니다.

C피보나치 — 재귀 vs 메모이제이션 vs 타뷸레이션
#include <stdio.h> #include <string.h> // 1. 단순 재귀 O(2^n) — 매우 느림 int fib_naive(int n) { if (n <= 1) return n; return fib_naive(n - 1) + fib_naive(n - 2); } // 2. 메모이제이션 (Top-Down) O(n) int memo[50]; int fib_memo(int n) { if (n <= 1) return n; if (memo[n] != -1) return memo[n]; // 이미 계산됨 memo[n] = fib_memo(n-1) + fib_memo(n-2); return memo[n]; } // 3. 타뷸레이션 (Bottom-Up) O(n) int fib_tab(int n) { int dp[50]; dp[0] = 0; dp[1] = 1; for (int i = 2; i <= n; i++) dp[i] = dp[i-1] + dp[i-2]; return dp[n]; } int main() { memset(memo, -1, sizeof(memo)); printf("fib(10) = %d\n", fib_tab(10)); return 0; }

TIP메모이제이션은 재귀 구조를 유지하므로 점화식이 직관적입니다. 타뷸레이션은 반복문이라 스택 오버플로가 없고 상수 시간이 빠릅니다. 대회에서는 타뷸레이션을 더 자주 사용합니다.

SECTION 302Bronze | DP 입문
1D DP — 계단 오르기
DP1D점화식Bronze

N개의 계단이 있을 때, 한 번에 1칸 또는 2칸을 올라갈 수 있습니다. 꼭대기에 도달하는 방법의 수를 구하세요.

점화식: dp[i] = dp[i-1] + dp[i-2]
i번째 계단에 도달하려면 (i-1)에서 1칸 오르거나, (i-2)에서 2칸 오르면 됩니다.

C계단 오르기
#include <stdio.h> int main() { int n; scanf("%d", &n); int dp[46]; // n <= 45 dp[0] = 1; // 시작점: 1가지 dp[1] = 1; // 1칸: 1가지 for (int i = 2; i <= n; i++) { dp[i] = dp[i-1] + dp[i-2]; } printf("%d\n", dp[n]); return 0; }

계단 점수 합 — 연속 3계단 불가
DP조건부 점화식BOJ 2579Bronze

각 계단에 점수가 있고, 연속으로 3개의 계단을 밟을 수 없습니다. 마지막 계단은 반드시 밟아야 합니다. 얻을 수 있는 최대 점수를 구하세요.

상태 정의: dp[i][j] = i번째 계단까지 왔고, 연속 j번째로 밟은 경우의 최대 점수.
dp[i][1] = max(dp[i-2][1], dp[i-2][2]) + score[i]
dp[i][2] = dp[i-1][1] + score[i]

C계단 점수 합 (BOJ 2579)
#include <stdio.h> int score[301], dp[301][3]; int max(int a, int b) { return a > b ? a : b; } int main() { int n; scanf("%d", &n); for (int i = 1; i <= n; i++) scanf("%d", &score[i]); dp[1][1] = score[1]; if (n >= 2) { dp[2][1] = score[2]; dp[2][2] = score[1] + score[2]; } for (int i = 3; i <= n; i++) { dp[i][1] = max(dp[i-2][1], dp[i-2][2]) + score[i]; dp[i][2] = dp[i-1][1] + score[i]; } printf("%d\n", max(dp[n][1], dp[n][2])); return 0; }

1로 만들기 — 최소 연산 횟수
DP최솟값BOJ 1463Bronze

정수 N이 주어질 때, 다음 3가지 연산을 사용하여 1로 만드는 최소 연산 횟수를 구하세요:
1) 3으로 나누어 떨어지면 3으로 나누기   2) 2로 나누어 떨어지면 2로 나누기   3) 1 빼기

C1로 만들기
#include <stdio.h> int dp[1000001]; int min(int a, int b) { return a < b ? a : b; } int main() { int n; scanf("%d", &n); dp[1] = 0; // 1은 이미 1 for (int i = 2; i <= n; i++) { dp[i] = dp[i - 1] + 1; // -1 연산 if (i % 2 == 0) dp[i] = min(dp[i], dp[i/2] + 1); if (i % 3 == 0) dp[i] = min(dp[i], dp[i/3] + 1); } printf("%d\n", dp[n]); return 0; }

SECTION 305Bronze | DP 입문
LIS — 최장 증가 부분수열 O(n²)
DPLIS대회 필수Bronze

LIS(Longest Increasing Subsequence)는 주어진 수열에서 순서를 유지하면서 순 증가하는 가장 긴 부분수열의 길이를 구하는 문제입니다.

점화식: dp[i] = max(dp[j] + 1) (j < i이고 a[j] < a[i]인 모든 j에 대해)

CLIS O(n²)
#include <stdio.h> int main() { int n; scanf("%d", &n); int a[1001], dp[1001]; for (int i = 0; i < n; i++) scanf("%d", &a[i]); int ans = 1; for (int i = 0; i < n; i++) { dp[i] = 1; // 최소 자기 자신 for (int j = 0; j < i; j++) { if (a[j] < a[i] && dp[j] + 1 > dp[i]) dp[i] = dp[j] + 1; } if (dp[i] > ans) ans = dp[i]; } printf("%d\n", ans); return 0; }

SECTION 306Bronze | DP 입문
LIS O(n log n) — 이분 탐색 최적화
DP이분 탐색LISBronze

O(n²) LIS는 n이 클 때 느립니다. tail 배열과 이분 탐색을 사용하면 O(n log n)에 해결합니다.

아이디어: tail[i] = 길이 (i+1)인 증가 부분수열의 마지막 원소의 최솟값. 새 원소 x가 올 때, x가 tail의 마지막보다 크면 추가, 아니면 lower_bound 위치에 교체합니다.

CLIS O(n log n)
#include <stdio.h> int tail[100001], len = 0; // x 이상인 첫 위치를 이분 탐색 int lower_bound(int x) { int lo = 0, hi = len; while (lo < hi) { int mid = (lo + hi) / 2; if (tail[mid] < x) lo = mid + 1; else hi = mid; } return lo; } int main() { int n; scanf("%d", &n); for (int i = 0; i < n; i++) { int x; scanf("%d", &x); int pos = lower_bound(x); tail[pos] = x; if (pos == len) len++; // LIS 길이 증가 } printf("%d\n", len); return 0; }

650추가[10, 20, 30, 50]4
TIPtail 배열은 실제 LIS가 아닙니다! 길이만 정확합니다. 실제 수열 복원이 필요하면 별도의 역추적을 해야 합니다.
WARN비감소(non-decreasing) 수열을 구하려면 lower_bound 대신 upper_bound(x보다 큰 첫 위치)를 사용하세요.

SECTION 307Bronze | DP 입문
0/1 배낭 문제 — Knapsack
DPknapsack대회 필수Bronze

0/1 배낭 문제: N개의 물건이 있고 각각 무게 w[i]와 가치 v[i]를 가집니다. 배낭의 최대 용량이 W일 때, 넣을 수 있는 물건의 가치 합의 최댓값을 구하세요. 각 물건은 최대 1개만 넣을 수 있습니다.

점화식: dp[i][j] = max(dp[i-1][j], dp[i-1][j-w[i]] + v[i])
i번째 물건을 안 넣거나(dp[i-1][j]) / 넣거나(dp[i-1][j-w[i]]+v[i]) 중 최댓값.

C0/1 배낭 문제
#include <stdio.h> int n, W; int w[101], v[101]; int dp[101][10001]; int max(int a, int b) { return a > b ? a : b; } int main() { scanf("%d %d", &n, &W); for (int i = 1; i <= n; i++) scanf("%d %d", &w[i], &v[i]); for (int i = 1; i <= n; i++) { for (int j = 0; j <= W; j++) { dp[i][j] = dp[i-1][j]; // i번째 물건 안 넣기 if (j >= w[i]) dp[i][j] = max(dp[i][j], dp[i-1][j-w[i]] + v[i]); } } printf("%d\n", dp[n][W]); return 0; }

SECTION 308Bronze | DP 입문
동전 교환 — 최소 동전 개수
DPunbounded knapsack동전Bronze

N종류의 동전이 있고, 각 동전은 무한히 사용 가능합니다. 금액 K를 만드는 데 필요한 최소 동전 개수를 구하세요.

점화식: dp[j] = min(dp[j], dp[j - coin[i]] + 1)
이것은 완전 배낭(Unbounded Knapsack)의 변형입니다. j를 정순으로 순회하여 같은 동전을 여러 번 사용할 수 있게 합니다.

C동전 교환
#include <stdio.h> #include <string.h> int dp[10001]; int min(int a, int b) { return a < b ? a : b; } int main() { int n, k; int coin[101]; scanf("%d %d", &n, &k); for (int i = 0; i < n; i++) scanf("%d", &coin[i]); memset(dp, 0x3f, sizeof(dp)); // 큰 값으로 초기화 dp[0] = 0; // 0원: 동전 0개 for (int i = 0; i < n; i++) { for (int j = coin[i]; j <= k; j++) { dp[j] = min(dp[j], dp[j - coin[i]] + 1); } } if (dp[k] >= 0x3f3f3f3f) printf("-1\n"); else printf("%d\n", dp[k]); return 0; }

출력

15 = 12 + 1 + 1 + 1 (3개). 5+5+5=15(3개)도 가능하지만, 12+1+1+1도 3개로 같습니다.

NOTE불가능한 경우를 처리하기 위해 dp를 큰 값(INF)으로 초기화합니다. 결과가 INF 이상이면 -1을 출력합니다.

SECTION 309Bronze | DP 입문
연속 부분합 최대 — Kadane 알고리즘
DPKadane부분합Bronze

N개의 정수로 이루어진 수열에서 연속된 부분수열의 합 중 최댓값을 구하세요. (모든 수가 음수일 수도 있음)

DP 관점: dp[i] = max(a[i], dp[i-1] + a[i])
i번째 원소를 끝으로 하는 연속 부분합의 최댓값. 이전 합이 음수이면 현재부터 새로 시작합니다.
공간 최적화: dp 배열 대신 변수 하나로 충분합니다 (Kadane's Algorithm).

CKadane 알고리즘 (DP 관점)
#include <stdio.h> int main() { int n; scanf("%d", &n); int a[100001]; for (int i = 0; i < n; i++) scanf("%d", &a[i]); // dp[i] = i번째 원소를 끝으로 하는 최대 부분합 int dp = a[0], ans = a[0]; for (int i = 1; i < n; i++) { // 이전까지의 합에 현재를 더하거나, 현재부터 새로 시작 dp = (dp + a[i] > a[i]) ? dp + a[i] : a[i]; if (dp > ans) ans = dp; } printf("%d\n", ans); return 0; }

SECTION 310Bronze | DP 입문
Debug Hunt — DP 버그 잡기
debugDP초기화Bronze

DP 코드에서 자주 발생하는 실수를 찾아보세요. 각 코드에는 하나의 버그가 있습니다.

BUG #1 — 배열 크기 부족
C피보나치 — 버그를 찾으세요
#include <stdio.h> int dp[100]; // BUG: 크기 부족! n이 100이면 dp[100] 접근 불가 int main() { int n = 100; dp[0] = 0; dp[1] = 1; for (int i = 2; i <= n; i++) dp[i] = dp[i-1] + dp[i-2]; printf("%d\n", dp[n]); return 0; }

힌트: dp[100]에 접근할 수 있나요? 배열 크기가 100이면 인덱스 0~99만 사용 가능합니다.

수정: int dp[101];로 변경. ≤ 조건일 때 배열 크기는 n+1 이상이어야 합니다.

BUG #2 — 초기값 누락

행렬 연쇄 곱셈(Matrix Chain Multiplication): N개의 행렬을 곱할 때, 괄호를 어떻게 묶느냐에 따라 곱셈 횟수가 달라집니다. 최소 곱셈 횟수를 구하세요.

SECTION 316Bronze | DP 응용
비트마스크 DP — 외판원 문제 (TSP)
DPbitmaskTSPBronze

비트마스크 DP는 집합의 상태를 비트로 표현합니다. N개 원소의 부분집합은 N비트 정수로 나타낼 수 있습니다.

외판원 문제(TSP): N개 도시를 모두 방문하고 돌아오는 최소 비용. N ≤ 16이면 비트마스크 DP로 O(2n · n²)에 해결합니다.

상태: dp[mask][u] = 방문한 도시 집합이 mask이고 현재 u에 있을 때의 최소 비용.
전이: dp[mask | (1<<v)][v] = min(..., dp[mask][u] + dist[u][v])

CTSP (비트마스크 DP)
#include <stdio.h> #include <string.h> #define INF 1000000000 #define MAXN 16 int n; int dist[MAXN][MAXN]; int dp[1 << MAXN][MAXN]; int min(int a, int b) { return a < b ? a : b; } int main() { scanf("%d", &n); for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) scanf("%d", &dist[i][j]); memset(dp, 0x3f, sizeof(dp)); dp[1][0] = 0; // 시작: 0번 도시, 방문={0} for (int mask = 1; mask < (1 << n); mask++) { for (int u = 0; u < n; u++) { if (!(mask & (1 << u))) continue; if (dp[mask][u] >= INF) continue; for (int v = 0; v < n; v++) { if (mask & (1 << v)) continue; // 이미 방문 int nmask = mask | (1 << v); dp[nmask][v] = min(dp[nmask][v], dp[mask][u] + dist[u][v]); } } } int full = (1 << n) - 1, ans = INF; for (int u = 0; u < n; u++) ans = min(ans, dp[full][u] + dist[u][0]); printf("%d\n", ans); return 0; }
입력
4 0 10 15 20 10 0 35 25 15 35 0 30 20 25 30 0
출력
80
TIP비트 연산 정리: mask & (1<<i) = i번 비트 확인, mask | (1<<i) = i번 비트 켜기, mask ^ (1<<i) = i번 비트 토글.
WARN216 × 16 = 약 100만이므로 N ≤ 16까지 가능합니다. N > 20이면 다른 방법(근사 알고리즘)을 사용해야 합니다.

트리 DP: 트리 구조에서 DFS(후위 순회)하며 자식의 DP 값을 부모에 합산합니다.

독립 집합: 인접한 노드를 동시에 선택하지 않으면서 가중치 합을 최대화합니다.
dp[v][0] = Σ max(dp[u][0], dp[u][1]) (v 미선택: 자식은 선택 가능)

C트리 DP — 독립 집합
#include <stdio.h> #define MAXN 100001 int head[MAXN], nxt[2*MAXN], to[2*MAXN], ecnt; int dp[MAXN][2]; // dp[v][0]: v 미선택, dp[v][1]: v 선택 int w[MAXN]; void add_edge(int u, int v) { to[ecnt] = v; nxt[ecnt] = head[u]; head[u] = ecnt++; } int max(int a, int b) { return a > b ? a : b; } void dfs(int v, int parent) { dp[v][0] = 0; dp[v][1] = w[v]; for (int e = head[v]; e != -1; e = nxt[e]) { int u = to[e]; if (u == parent) continue; dfs(u, v); dp[v][0] += max(dp[u][0], dp[u][1]); dp[v][1] += dp[u][0]; // v 선택 시 자식 미선택 } } int main() { int n; scanf("%d", &n); memset(head, -1, sizeof(head)); for (int i = 1; i <= n; i++) scanf("%d", &w[i]); for (int i = 0; i < n-1; i++) { int u, v; scanf("%d %d", &u, &v); add_edge(u, v); add_edge(v, u); } dfs(1, 0); printf("%d\n", max(dp[1][0], dp[1][1])); return 0; }

BUG #2 — LCS 역추적 인덱스
CLCS 역추적 — 경계 오류
// LCS 역추적에서 인덱스 오류 int i = n, j = m; while (i >= 0 && j >= 0) { // BUG: i>0, j>0이어야 함! if (s1[i-1] == s2[j-1]) { // i=0이면 s1[-1] 접근! lcs[--idx] = s1[i-1]; i--; j--; } }

문제: i >= 0이면 i=0일 때 s1[i-1] = s1[-1]로 잘못된 메모리에 접근합니다.

수정: i > 0 && j > 0으로 변경하세요.

BUG #3 — 비트마스크 상태 누락
C비트마스크 DP — 유효성 검사 누락
// 비트마스크 DP — 상태 누락 for (int mask = 0; mask < (1 << n); mask++) { for (int u = 0; u < n; u++) { // BUG: mask에 u가 포함되어 있는지 확인 안 함! for (int v = 0; v < n; v++) { if (mask & (1 << v)) continue; dp[mask | (1 << v)][v] = dp[mask][u] + dist[u][v]; } } }

문제: mask에 u가 포함되지 않은 상태에서 dp[mask][u]를 사용하면 초기화되지 않은 값을 사용합니다.

수정: if (!(mask & (1 << u))) continue;를 u 루프 첫 줄에 추가하세요.

WARN응용 DP 디버깅 3대 포인트: 1) 구간 범위(len 시작), 2) 역추적 경계(0 vs 1), 3) 비트마스크 유효 상태 확인.

SECTION 321Bronze | DP 실전
타일링 문제 모음 — 2×N, 3×N
DP타일링점화식 유도Bronze

2×N 타일링: 2×N 직사각형을 1×2 타일로 채우는 방법의 수.
dp[n] = dp[n-1] + dp[n-2] (세로 1개 놓기 + 가로 2개 놓기)

3×N 타일링: 3×N 직사각형을 1×2 타일로 채우는 방법의 수. N이 홀수이면 불가능합니다.
dp[n] = 4×dp[n-2] - dp[n-4] (점화식 유도가 까다로움)

C2×N, 3×N 타일링
#include <stdio.h> long long dp2[1001], dp3[1001]; int main() { int n; scanf("%d", &n); // 2×N 타일링 (1×2, 2×1 타일) dp2[1] = 1; dp2[2] = 2; for (int i = 3; i <= n; i++) dp2[i] = (dp2[i-1] + dp2[i-2]) % 10007; // 3×N 타일링 (n이 홀수이면 불가능) dp3[0] = 1; dp3[2] = 3; for (int i = 4; i <= n; i += 2) dp3[i] = (dp3[i-2] * 4 - dp3[i-4] + 10007) % 10007; printf("2x%d: %lld\n", n, dp2[n]); if (n % 2 == 0) printf("3x%d: %lld\n", n, dp3[n]); else printf("3x%d: 0 (impossible)\n", n); return 0; }

6
출력
2x6: 13 3x6: 41

SECTION 322Bronze | DP 실전
파도반 수열 — P(n) = P(n-2) + P(n-3)
DP파도반long longBronze

파도반 수열: P(1)=P(2)=P(3)=1, P(n) = P(n-2) + P(n-3).
정삼각형을 나선형으로 붙여 나가는 수열입니다. 수열: 1, 1, 1, 2, 2, 3, 4, 5, 7, 9, ...

C파도반 수열 (BOJ 9461)
#include <stdio.h> long long dp[101]; int main() { int T, n; scanf("%d", &T); // 파도반 수열: P(n) = P(n-2) + P(n-3) dp[1] = 1; dp[2] = 1; dp[3] = 1; dp[4] = 2; dp[5] = 2; for (int i = 6; i <= 100; i++) dp[i] = dp[i-2] + dp[i-3]; while (T--) { scanf("%d", &n); printf("%lld\n", dp[n]); } return 0; }
입력
3 6 12 100
출력
3 16 888855064897
n12345678910
P(n)1112234579
WARNP(100)은 약 8.9×1011로 int 범위를 초과합니다! 반드시 long long을 사용하세요. int는 약 21억까지만 저장 가능합니다.
TIP비슷한 패턴: 피보나치(dp[i-1]+dp[i-2]), 트리보나치(dp[i-1]+dp[i-2]+dp[i-3]), 파도반(dp[i-2]+dp[i-3]). 점화식의 계수와 참조 위치만 다릅니다.

SECTION 323Bronze | DP 실전
대회 기출 DP — RGB 거리
DPRGBBOJ 1149Bronze

RGB 거리: N개의 집을 R/G/B 중 하나로 칠하되, 인접한 집은 다른 색이어야 합니다. 각 집마다 색별 비용이 다를 때, 최소 비용을 구하세요.

점화식:
dp[i][0] = min(dp[i-1][1], dp[i-1][2]) + cost[i][0] (R)
dp[i][1] = min(dp[i-1][0], dp[i-1][2]) + cost[i][1] (G)
dp[i][2] = min(dp[i-1][0], dp[i-1][1]) + cost[i][2] (B)

CRGB 거리 (BOJ 1149)
#include <stdio.h> int cost[1001][3]; // 0:R, 1:G, 2:B int dp[1001][3]; int min(int a, int b) { return a < b ? a : b; } int main() { int n; scanf("%d", &n); for (int i = 1; i <= n; i++) scanf("%d %d %d", &cost[i][0], &cost[i][1], &cost[i][2]); dp[1][0] = cost[1][0]; dp[1][1] = cost[1][1]; dp[1][2] = cost[1][2]; for (int i = 2; i <= n; i++) { dp[i][0] = min(dp[i-1][1], dp[i-1][2]) + cost[i][0]; dp[i][1] = min(dp[i-1][0], dp[i-1][2]) + cost[i][1]; dp[i][2] = min(dp[i-1][0], dp[i-1][1]) + cost[i][2]; } int ans = min(min(dp[n][0], dp[n][1]), dp[n][2]); printf("%d\n", ans); return 0; }

3 26 40 83 49 60 57 13 89 99
출력
96

SECTION 324Bronze | DP 실전
대회 기출 DP 2 — 포도주 시식
DPBOJ 2156대회 기출Bronze

포도주 시식: N잔의 포도주가 있고, 연속 3잔을 마실 수 없습니다. 최대로 마실 수 있는 포도주의 양을 구하세요.

점화식:
dp[i] = max(dp[i-1], dp[i-2]+a[i], dp[i-3]+a[i-1]+a[i])
1) i번째를 안 마시기 2) 한 잔 건너뛰고 마시기 3) 연속 2잔 (i-1, i) 마시기

C포도주 시식 (BOJ 2156)
#include <stdio.h> int a[10001], dp[10001]; int max(int a, int b) { return a > b ? a : b; } int max3(int a, int b, int c) { return max(a, max(b, c)); } int main() { int n; scanf("%d", &n); for (int i = 1; i <= n; i++) scanf("%d", &a[i]); dp[1] = a[1]; dp[2] = a[1] + a[2]; for (int i = 3; i <= n; i++) { dp[i] = max3( dp[i-1], // i번째 안 마시기 dp[i-2] + a[i], // 한 잔 건너뛰고 마시기 dp[i-3] + a[i-1] + a[i] // 연속 2잔 (i-1, i) ); } printf("%d\n", dp[n]); return 0; }
