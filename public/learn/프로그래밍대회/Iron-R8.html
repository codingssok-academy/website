서로 다른 키가 같은 해시값을 가지면 충돌(collision)이 발생합니다. 체이닝(Chaining)은 같은 버킷에 연결 리스트로 여러 값을 저장합니다.

C체이닝 해시 테이블
#define M 1007 typedef struct Node { int key; struct Node *next; } Node; Node *table[M]; // 전부 NULL로 초기화 void insert(int key) { int h = ((key % M) + M) % M; Node *nd = (Node*)malloc(sizeof(Node)); nd->key = key; nd->next = table[h]; table[h] = nd; } int search(int key) { int h = ((key % M) + M) % M; for (Node *p = table[h]; p; p = p->next) if (p->key == key) return 1; return 0; }

개방 주소법 — 빈 자리 찾아 저장
hash tableopen addressinglinear probingIron

개방 주소법(Open Addressing)은 충돌 시 다른 빈 슬롯을 찾아 저장합니다. 가장 단순한 방식은 선형 탐사(Linear Probing): 한 칸씩 이동하며 빈 자리를 찾습니다.

C선형 탐사
#define M 10007 #define EMPTY -1 int table[M]; void init() { for(int i=0;i<M;i++) table[i]=EMPTY; } void insert(int key) { int h = ((key % M) + M) % M; while (table[h] != EMPTY && table[h] != key) h = (h + 1) % M; // 다음 칸 table[h] = key; } int search(int key) { int h = ((key % M) + M) % M; while (table[h] != EMPTY) { if (table[h] == key) return 1; h = (h + 1) % M; } return 0; }
체이닝 vs 개방 주소법 체이닝: 구현 간단, 메모리 동적 할당 필요
개방 주소법: 배열만으로 구현 가능, 클러스터링 주의
대회에서는 보통 카운팅 배열이나 체이닝을 더 많이 씁니다.

문자열을 숫자로 바꾸면 비교를 O(1)에 할 수 있습니다. 롤링 해시는 한 칸 슬라이드할 때 이전 해시값을 재활용합니다.

C간단한 문자열 해시
#define BASE 31 #define MOD 1000000007LL long long str_hash(const char *s, int len) { long long h = 0, pw = 1; for (int i = 0; i < len; i++) { h = (h + (s[i] - 'a' + 1) * pw) % MOD; pw = pw * BASE % MOD; } return h; } // 같은 해시 → 같은 문자열 (높은 확률) // 다른 해시 → 100% 다른 문자열
해시 충돌해시가 같아도 실제 문자열이 다를 수 있습니다(해시 충돌). 대회에서는 MOD를 크게 잡거나, 이중 해시(두 개의 다른 MOD)를 사용합니다.

C문자열 치환
void replace_all(char *result, const char *src, const char *old, const char *new_s) { int olen = strlen(old), nlen = strlen(new_s); int pos = 0; const char *p = src; while (*p) { char *found = strstr(p, old); if (!found) { strcpy(result + pos, p); return; } int gap = found - p; memcpy(result + pos, p, gap); pos += gap; memcpy(result + pos, new_s, nlen); pos += nlen; p = found + olen; } result[pos] = '\0'; }

C힌트 코드
int count = 0; for (int mask = 1; mask < (1 << n); mask++) { int sum = 0; for (int i = 0; i < n; i++) if (mask & (1 << i)) sum += a[i]; if (sum == S) count++; }
